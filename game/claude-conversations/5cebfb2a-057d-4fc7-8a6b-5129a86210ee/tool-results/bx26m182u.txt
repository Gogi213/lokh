	public static bool ConnectionsExist
	{
		get
		{
			if (!roomDataLoaded || !RoomPlaced(startRoom))
			{
				return true;
			}
			foreach (List<IntVec2> value in priorityConnDict.Values)
			{
				if (value != null && value.Count > 0)
				{
					return true;
				}
			}
			foreach (List<IntVec2> value2 in connectionDict.Values)
			{
				if (value2 != null && value2.Count > 0)
				{
					return true;
				}
			}
			return false;
		}
	}

	private static bool AllRoomsCreated => roomsRemaining <= 0;

	private static bool RoomPlaced(RoomStub givenRoom)
	{
		return createdRoomsList.Contains(givenRoom);
	}

	public static void StartGenerator(MonoBehaviour givenMono, TileGrid tileGrid)
	{
		isGenerating = true;
		currentMonobehaviour = givenMono;
		currentTileGrid = tileGrid;
		SetGenerationVars();
		currentMonobehaviour.StartCoroutine(LoadRoomData(currentMonobehaviour));
		currentMonobehaviour.StartCoroutine(GenerateLayout());
	}

	private static void SetGenerationVars()
	{
		mapSize = new IntVec2(currentTileGrid.RowCount, currentTileGrid.ColCount);
		mapSizePadded = new IntVec2(mapSize.x - 8 + 4, mapSize.y - 8 + 4);
		mapCenter = mapSize * 0.5f;
		mapStart = mapCenter;
		roomsRemaining = ((!GameController.coopOn) ? 10 : Mathf.FloorToInt(12f));
		architectActive = false;
		sceneName = Application.loadedLevelName;
		if (roomContainer != null && roomContainer.gameObject != null)
		{
			Object.Destroy(roomContainer.gameObject);
		}
		if (hallContainer != null && hallContainer.gameObject != null)
		{
			Object.Destroy(hallContainer.gameObject);
		}
		roomContainer = new GameObject("RoomContainer").transform;
		roomContainer.transform.SetParent(Level.levelContainer);
		hallContainer = new GameObject("HallContainer").transform;
		hallContainer.transform.SetParent(Level.levelContainer);
		ClearLists();
		disableConnections = false;
		createdRoomsList.Remove(startRoom);
		createdRoomsList.Remove(exitRoom);
		createdRoomsList.Remove(skillStoreRoom);
		createdRoomsList.Remove(itemStoreRoom);
		createdRoomsList.Remove(cursedStoreRoom);
		treasureRoomCreated = false;
	}

	public static void ClearLists()
	{
		priorityRoomGenList.Clear();
		Level.CurrentRoomList.Clear();
		Level.CurrentHallList.Clear();
		Level.GridRoomTupleList.Clear();
		Level.GridHallTupleList.Clear();
		foreach (List<IntVec2> value in connectionDict.Values)
		{
			value.Clear();
		}
		foreach (List<IntVec2> value2 in priorityConnDict.Values)
		{
			value2.Clear();
		}
	}

	public static IEnumerator LoadRoomData(MonoBehaviour givenMono)
	{
		while (roomDataLoading)
		{
			yield return null;
		}
		if (roomDataLoaded)
		{
			yield break;
		}
		roomDataLoading = true;
		mainRoomDict = GetMapDirectionsDict<RoomStub>();
		mainHallwayDict = GetMapDirectionsDict<RoomStub>();
		mainFloorPlanDict = GetMapDirectionsDict<RoomStub>();
		mainEndRoomDict = GetMapDirectionsDict<RoomStub>();
		List<RoomStub> roomStubList = new List<RoomStub>();
		yield return givenMono.StartCoroutine(RoomStub.LoadAll(roomStubList));
		foreach (RoomStub rs in roomStubList)
		{
			switch (rs.type)
			{
			case RoomType.Start:
				startRoom = rs;
				break;
			case RoomType.Exit:
				exitRoom = rs;
				break;
			case RoomType.ItemStore:
				itemStoreRoom = rs;
				break;
			case RoomType.SkillStore:
				skillStoreRoom = rs;
				break;
			case RoomType.Misc:
				if (rs.name == "CursedStore")
				{
					cursedStoreRoom = rs;
					break;
				}
				mainMiscList.Add(rs);
				switch (rs.name)
				{
				case "SlotsRoom":
					shufflerRoom = rs;
					break;
				case "NurseRoom":
					doctorRoom = rs;
					break;
				case "TailorRoom":
					tailorRoom = rs;
					break;
				case "CollectorRoom":
					collectorRoom = rs;
					break;
				case "BankerRoom":
					bankerRoom = rs;
					break;
				case "PinataRoom":
					pinataRoom = rs;
					break;
				}
				break;
			case RoomType.FloorPlan:
				AddToRoomDict(rs, ref mainFloorPlanDict);
				break;
			case RoomType.Hallway:
				switch (rs.name)
				{
				case "ge_NS_stVertHall":
					nsHallway = rs;
					break;
				case "ge_EW_stHoriHall":
					ewHallway = rs;
					break;
				}
				AddToRoomDict(rs, ref mainHallwayDict);
				break;
			case RoomType.End:
				AddToRoomDict(rs, ref mainEndRoomDict);
				break;
			case RoomType.Party:
				partyRoom = rs;
				break;
			default:
				AddToRoomDict(rs, ref mainRoomDict);
				break;
			}
			yield return null;
		}
		roomDataLoading = false;
		roomDataLoaded = true;
	}

	public static IEnumerator GenerateLayout()
	{
		forceQuitGenerate = false;
		while (!roomDataLoaded)
		{
			yield return null;
		}
		if (Level.resetUsedRooms)
		{
			Level.resetUsedRooms = false;
			ClearLists();
			createdRoomsList.Clear();
			availableMiscList = new List<RoomStub>(mainMiscList);
		}
		Level.startPosition = new Vector2(mapStart.y, -mapStart.x) + new Vector2(12f, -14f);
		if (Level.isBossRushMode)
		{
			PlaceRoom(mapStart, MapDirections.None, exitRoom);
			yield return null;
			ConnectionInfo[] rushConns = new ConnectionInfo[4]
			{
				GetDirectionConnection(MapDirections.West),
				GetDirectionConnection(MapDirections.North),
				GetDirectionConnection(MapDirections.East),
				GetDirectionConnection(MapDirections.South)
			};
			foreach (List<IntVec2> value in connectionDict.Values)
			{
				value.Clear();
			}
			foreach (List<IntVec2> value2 in priorityConnDict.Values)
			{
				value2.Clear();
			}
			PlaceHallway(rushConns[0].position, rushConns[0].direction, ewHallway);
			PlaceHallway(rushConns[1].position, rushConns[1].direction, nsHallway);
			PlaceHallway(rushConns[2].position, rushConns[2].direction, ewHallway);
			PlaceHallway(rushConns[3].position, rushConns[3].direction, nsHallway);
			ref ConnectionInfo reference = ref rushConns[0];
			reference = GetDirectionConnection(MapDirections.West);
			ref ConnectionInfo reference2 = ref rushConns[1];
			reference2 = GetDirectionConnection(MapDirections.North);
			ref ConnectionInfo reference3 = ref rushConns[2];
			reference3 = GetDirectionConnection(MapDirections.East);
			ref ConnectionInfo reference4 = ref rushConns[3];
			reference4 = GetDirectionConnection(MapDirections.South);
			Level.ClearRoomPositions();
			PlaceRoom(rushConns[0].position, rushConns[0].direction, itemStoreRoom);
			yield return null;
			selectedMiscRoom = GetMiscRoom(cursedDisabled: true);
			availableMiscList.Remove(selectedMiscRoom);
			PlaceRoom(rushConns[1].position, rushConns[1].direction, selectedMiscRoom);
			yield return null;
			PlaceRoom(rushConns[2].position, rushConns[2].direction, skillStoreRoom);
			yield return null;
			PlaceRoom(rushConns[3].position, rushConns[3].direction, cursedStoreRoom);
			yield return null;
			Level.startPosition += Vector2.down * 3f;
			Level.tileGrid = currentTileGrid;
			isGenerating = false;
			yield break;
		}
		PlaceRoom(mapStart, MapDirections.None, startRoom);
		ConnectionInfo connInfo = GetRandomConnection();
		currentMonobehaviour.StartCoroutine(StartArchitect(connInfo.position, connInfo.direction));
		while (!AllRoomsCreated || architectActive)
		{
			if (forceQuitGenerate)
			{
				yield break;
			}
			yield return null;
		}
		if (forceQuitGenerate)
		{
			yield break;
		}
		disableConnections = true;
		Level.ClearRoomPositions();
		priorityRoomGenList.Clear();
		priorityRoomGenList.Add(itemStoreRoom);
		priorityRoomGenList.Add(skillStoreRoom);
		connInfo = GetRandomConnection();
		currentMonobehaviour.StartCoroutine(StartArchitect(connInfo.position, connInfo.direction));
		while (architectActive || !RoomPlaced(itemStoreRoom) || !RoomPlaced(skillStoreRoom))
		{
			if (forceQuitGenerate)
			{
				yield break;
			}
			yield return null;
		}
		selectedMiscRoom = GetMiscRoom();
		priorityRoomGenList.Clear();
		priorityRoomGenList.Add(selectedMiscRoom);
		connInfo = GetRandomConnection();
		currentMonobehaviour.StartCoroutine(StartArchitect(connInfo.position, connInfo.direction));
		while ((selectedMiscRoom != null && architectActive) || !RoomPlaced(selectedMiscRoom))
		{
			if (forceQuitGenerate)
			{
				yield break;
			}
			yield return null;
		}
		priorityRoomGenList.Clear();
		priorityRoomGenList.Add(exitRoom);
		connInfo = GetFurthestConnectionInfo(mapStart);
		currentMonobehaviour.StartCoroutine(StartArchitect(connInfo.position, connInfo.direction));
		while (architectActive || !RoomPlaced(exitRoom))
		{
			if (forceQuitGenerate)
			{
				yield break;
			}
			yield return null;
		}
		if (!createdRoomsList.Contains(partyRoom) && Random.value < 0.02f)
		{
			priorityRoomGenList.Clear();
			priorityRoomGenList.Add(partyRoom);
			connInfo = GetRandomConnection();
			currentMonobehaviour.StartCoroutine(StartArchitect(connInfo.position, connInfo.direction));
			while (architectActive || !RoomPlaced(partyRoom))
			{
				if (forceQuitGenerate)
				{
					yield break;
				}
				yield return null;
			}
		}
		AddDeadEndRooms(priorityConnDict);
		FillMissedConnections();
		RemoveExtraConnectionsFromSectionList(Level.GridRoomTupleList, new IntVec2(4, 2));
		availableMiscList.Remove(selectedMiscRoom);
		Level.tileGrid = currentTileGrid;
		isGenerating = false;
	}

	private static IEnumerator StartArchitect(IntVec2 startPosition, MapDirections startDirection)
	{
		architectActive = true;
		RoomStub architectStub = null;
		IntVec2 position = startPosition;
		MapDirections direction = startDirection;
		yield return null;
		while ((!AllRoomsCreated || priorityRoomGenList.Count > 0 || architectStub != null) && !forceQuitGenerate)
		{
			if (priorityRoomGenList.Count > 0 && architectStub == null)
			{
				architectStub = priorityRoomGenList[0];
				priorityRoomGenList.RemoveAt(0);
			}
			if ((Random.value < 0.75f || architectStub != null) && PlaceRoom(position, direction, architectStub, Level.doubleEnemies || Random.value < 0.3f))
			{
				architectStub = null;
				priorityConnDict[direction].Remove(position);
				connectionDict[direction].Remove(position);
				yield return null;
			}
			if (PlaceHallway(position, direction))
			{
				priorityConnDict[direction].Remove(position);
				connectionDict[direction].Remove(position);
				yield return null;
			}
			ConnectionInfo connInfo = GetRandomConnection();
			direction = connInfo.direction;
			position = connInfo.position;
			yield return null;
		}
		architectActive = false;
	}

	private static bool PlaceRoom(IntVec2 startPosition, MapDirections givenDirection, RoomStub givenRoom = null, bool isRandom = false)
	{
		incomingPlaceRoomDir = GetOppositeDirection(givenDirection);
		selectedPlaceRoom = givenRoom ?? GetAvailableRoom(startPosition, incomingPlaceRoomDir, isRandom) ?? GetAvailableRoom(startPosition, incomingPlaceRoomDir, isRandom: true);
		if (selectedPlaceRoom == null)
		{
			Debug.Log("Error with available room list!! Catastrophic disaster!!");
			return false;
		}
		loadedPlaceRoom = GetRoomFromLevelDict(selectedPlaceRoom.name);
		if (!PlaceTileGrid(startPosition, incomingPlaceRoomDir, selectedPlaceRoom, loadedPlaceRoom.tileGrid, isRoom: true))
		{
			return false;
		}
		createdRoomsList.Add(selectedPlaceRoom);
		Level.CurrentRoomList.Add(loadedPlaceRoom);
		roomsRemaining--;
		return true;
	}

	private static bool PlaceHallway(IntVec2 startPosition, MapDirections givenDirection, RoomStub givenHall = null)
	{
		RoomStub roomStub = givenHall;
		MapDirections oppositeDirection = GetOppositeDirection(givenDirection);
		if (roomStub == null)
		{
			List<RoomStub> list = mainHallwayDict[oppositeDirection];
			roomStub = list[Random.Range(0, list.Count)];
		}
		if (roomStub == null)
		{
			Debug.Log("Error with hallway dictionary!! Catastrophic disaster!!");
			return false;
		}
		Room roomFromLevelDict = GetRoomFromLevelDict(roomStub.name);
		if (!PlaceTileGrid(startPosition, oppositeDirection, roomStub, roomFromLevelDict.tileGrid, isRoom: false))
		{
			return false;
		}
		roomFromLevelDict.randomEnemies = false;
		roomFromLevelDict.randomInteractables = false;
		Level.CurrentHallList.Add(roomFromLevelDict);
		return true;
	}

	private static bool PlaceTileGrid(IntVec2 startPosition, MapDirections givenDirection, RoomStub givenStub, TileGrid givenGrid, bool isRoom)
	{
		if (roomContainer == null || hallContainer == null)
		{
			return false;
		}
		placedGrid = new TileGrid(givenGrid);
		foundConnDict.Clear();
		usedConnList.Clear();
		LayoutGeneratorHelper.GetAdjustedTileGrid(ref placedGrid, givenStub, ref foundConnDict, isRoom);
		LayoutGeneratorHelper.PaveConnectionPaths(ref placedGrid, givenDirection, ref foundConnDict, ref usedConnList);
		if (usedConnList.Count == 0 && givenStub.type != RoomType.Start && !Level.isBossRushMode)
		{
			Debug.LogError(givenStub.name + " doesn't have connections!! " + startPosition.y + " | " + startPosition.x + " || " + foundConnDict.Count);
		}
		queueLoadRoom = true;
		roomConnOffset = startPosition - ((usedConnList.Count <= 0) ? IntVec2.zero : usedConnList[Random.Range(0, usedConnList.Count)]);
		switch (LayoutGeneratorHelper.CanPlaceGrid(placedGrid, currentTileGrid, roomConnOffset))
		{
		case 1:
			queueLoadRoom = false;
			break;
		case 2:
			return false;
		}
		if (queueLoadRoom)
		{
			if (isRoom)
			{
				Level.GridRoomTupleList.Add(new Tuple<TileGridSection, Room>(new TileGridSection(Level.tileGrid, roomConnOffset.x, roomConnOffset.y, placedGrid.RowCount, placedGrid.ColCount), Room.Load(givenStub.name, instantiate: true, roomContainer, new Vector3(roomConnOffset.y + 2, -roomConnOffset.x - 4))));
			}
			else
			{
				Level.GridHallTupleList.Add(new Tuple<TileGridSection, Room>(new TileGridSection(Level.tileGrid, roomConnOffset.x - 4, roomConnOffset.y - 2, placedGrid.RowCount + 8, placedGrid.ColCount + 4), Room.Load(givenStub.name, instantiate: true, hallContainer, new Vector3(roomConnOffset.y, -roomConnOffset.x))));
			}
		}
		for (int i = 0; i < placedGrid.RowCount; i++)
		{
			for (int j = 0; j < placedGrid.ColCount; j++)
			{
				if (placedGrid[i, j] != null)
				{
					if (Level.removeAllPits && placedGrid[i, j].terrain == Tile.TerrainType.Pit)
					{
						placedGrid[i, j].terrain = Tile.TerrainType.Floor;
						placedGrid[i, j].pitOverrideType = Tile.FloorType.Center;
						placedGrid[i, j].layers["Floor"].hasDoodad = true;
						placedGrid[i, j].layers["FloorDoodad"].hasDoodad = true;
						placedGrid[i, j].layers["FloorDoodad2"].hasDoodad = true;
						placedGrid[i, j].layers["FloorDoodad3"].hasDoodad = true;
						placedGrid[i, j].layers["Floor"].flipX = Random.value > 0.5f;
					}
					currentTileGrid[roomConnOffset.x + i, roomConnOffset.y + j] = placedGrid[i, j];
				}
			}
		}
		RoomType type = givenStub.type;
		if (type == RoomType.Treasure)
		{
			treasureRoomCreated = true;
		}
		if (isRoom && disableConnections)
		{
			return true;
		}
		List<MapDirections> mapDirectionsList = GetMapDirectionsList();
		mapDirectionsList.Remove(givenDirection);
		for (int k = 0; k < connQueueBuilds.Length; k++)
		{
			connQueueBuilds[k] = false;
		}
		calcConnectionSpawnChance = 1f;
		for (int l = 0; l < mapDirectionsList.Count; l++)
		{
			if (!(Random.value > calcConnectionSpawnChance))
			{
				calcConnectionSpawnChance *= 0.75f;
				selectedConnDirection = mapDirectionsList[Random.Range(0, mapDirectionsList.Count)];
				switch (selectedConnDirection)
				{
				case MapDirections.North:
					connQueueBuilds[0] = true;
					break;
				case MapDirections.South:
					connQueueBuilds[1] = true;
					break;
				case MapDirections.East:
					connQueueBuilds[2] = true;
					break;
				case MapDirections.West:
					connQueueBuilds[3] = true;
					break;
				}
				mapDirectionsList.Remove(selectedConnDirection);
			}
		}
		foreach (KeyValuePair<IntVec2, Tile> item in foundConnDict)
		{
			if (usedConnList.Contains(item.Key))
			{
				continue;
			}
			connBaseCoord = item.Key;
			connStartCoord = item.Key + roomConnOffset;
			connTypeStr = item.Value.connection.ToString();
			connIsRequired = connTypeStr.Contains("Required");
			connDirectionIndex = ((!connTypeStr.Contains("Top")) ? (connTypeStr.Contains("Bottom") ? 1 : (connTypeStr.Contains("Right") ? 2 : ((!connTypeStr.Contains("Left")) ? (-1) : 3))) : 0);
			if (connDirectionIndex == -1)
			{
				continue;
			}
			connDirection = connMapDirections[connDirectionIndex];
			connStartCoord += connDirOffsets[connDirectionIndex];
			connLength = 0;
			while (foundConnDict.ContainsKey(connBaseCoord))
			{
				usedConnList.Add(connBaseCoord);
				connBaseCoord += connLengthOffsets[connDirectionIndex];
				connLength++;
			}
			if (connLength >= 4)
			{
				connPosCount = Mathf.FloorToInt(connLength / 4);
				connStartIndex = 4 * Random.Range(0, connPosCount);
				connStartCoord += connLengthOffsets[connDirectionIndex] * connStartIndex;
				connectionDict[connDirection].Add(connStartCoord);
				if (connIsRequired || connQueueBuilds[connDirectionIndex])
				{
					priorityConnDict[connDirection].Add(connStartCoord);
				}
			}
		}
		return true;
	}

	private static void AddDeadEndRooms(Dictionary<MapDirections, List<IntVec2>> givenDict)
	{
		foreach (KeyValuePair<MapDirections, List<IntVec2>> item in givenDict)
		{
			MapDirections oppositeDirection = GetOppositeDirection(item.Key);
			List<RoomStub> source = new List<RoomStub>(mainEndRoomDict[oppositeDirection]);
			source = source.Where((RoomStub p) => p.name.Contains("_ge_") || p.name.Contains(tierPrefixDict[sceneName])).ToList();
			int count = source.Count;
			foreach (IntVec2 item2 in item.Value)
			{
				int num = Random.Range(0, source.Count);
				for (int num2 = 0; num2 < count; num2++)
				{
					num = (num + 1) % count;
					RoomStub givenRoom = source[num];
					if (PlaceRoom(item2, item.Key, givenRoom))
					{

	private static void FillMissedConnections()
	{
		List<IntVec2> list = new List<IntVec2>();
		int num = currentTileGrid.RowCount - 8;
		int num2 = currentTileGrid.ColCount - 8;
		for (int i = 8; i < num; i += 4)
		{
			for (int j = 10; j < num2; j += 4)
			{
				if (currentTileGrid[i, j] == null)
				{
					bool flag = IsFloorTile(currentTileGrid[i - 4, j]);
					bool flag2 = IsFloorTile(currentTileGrid[i + 4, j]);
					bool flag3 = IsFloorTile(currentTileGrid[i, j + 4]);
					bool flag4 = IsFloorTile(currentTileGrid[i, j - 4]);
					if ((flag && flag2 && !flag3 && !flag4) || (flag3 && flag4 && !flag && !flag2) || (flag && flag3 && !IsFloorTile(currentTileGrid[i - 4, j + 4])) || (flag && flag4 && !IsFloorTile(currentTileGrid[i - 4, j - 4])) || (flag2 && flag3 && !IsFloorTile(currentTileGrid[i + 4, j + 4])) || (flag2 && flag4 && !IsFloorTile(currentTileGrid[i + 4, j - 4])))
					{
						list.Add(new IntVec2(i, j));
					}
				}
			}
		}
		RoomStub roomStub = new RoomStub(RoomType.Hallway);
		int count = list.Count;
		for (int k = 0; k < count; k++)
		{
			IntVec2 intVec = list[k];
			bool flag5 = true;
			for (int l = 0; l < 4; l++)
			{
				for (int m = 0; m < 4; m++)
				{
					if (currentTileGrid[intVec.x + l, intVec.y + m] != null)
					{
						flag5 = false;
					}
				}
			}
			if (!flag5)
			{
				continue;
			}
			for (int n = 0; n < 4; n++)
			{
				for (int num3 = 0; num3 < 4; num3++)
				{
					currentTileGrid[intVec.x + n, intVec.y + num3] = new Tile(Tile.TerrainType.Floor);
					currentTileGrid[intVec.x + n, intVec.y + num3].roomStub = roomStub;
				}
			}
		}
	}

	private static bool IsFloorTile(Tile givenTile)
	{
		return givenTile != null && givenTile.terrain == Tile.TerrainType.Floor;
	}

	private static void RemoveExtraConnectionsFromSectionList(List<Tuple<TileGridSection, Room>> givenList, IntVec2 givenOffset)
	{
		IntVec2 intVec = new IntVec2(givenOffset.x, givenOffset.y);
		foreach (Tuple<TileGridSection, Room> given in givenList)
		{
			IntVec2 intVec2 = intVec + new IntVec2(given.First.rowOffset, given.First.colOffset);
			given.Second.SetConnectionInfo();
			foreach (Tile allConnection in given.Second.allConnections)
			{
				IntVec2 intVec3 = intVec2 + new IntVec2(allConnection.row, allConnection.col);
				switch (currentTileGrid[intVec3.x, intVec3.y].connection)
				{
				case Tile.ConnectionType.TopOptional:
					intVec3.x--;
					break;
				case Tile.ConnectionType.TopRequired:
					intVec3.x--;
					break;
				case Tile.ConnectionType.BottomOptional:
					intVec3.x++;
					break;
				case Tile.ConnectionType.BottomRequired:
					intVec3.x++;
					break;
				case Tile.ConnectionType.LeftOptional:
					intVec3.y--;
					break;
				case Tile.ConnectionType.LeftRequired:
					intVec3.y--;
					break;
				case Tile.ConnectionType.RightOptional:
					intVec3.y++;
					break;
				case Tile.ConnectionType.RightRequired:
					intVec3.y++;
					break;
				}
				if (currentTileGrid[intVec3.x, intVec3.y] == null || (currentTileGrid[intVec3.x, intVec3.y].terrain != Tile.TerrainType.Floor && currentTileGrid[intVec3.x, intVec3.y].terrain != Tile.TerrainType.Pit))
				{
					allConnection.connection = Tile.ConnectionType.None;
				}
			}
		}
	}

	private static MapDirections GetOppositeDirection(MapDirections dir)
	{
		return dir switch
		{
			MapDirections.North => MapDirections.South, 
			MapDirections.South => MapDirections.North, 
			MapDirections.East => MapDirections.West, 
			MapDirections.West => MapDirections.East, 
			_ => MapDirections.None, 
		};
	}

	private static ConnectionInfo GetRandomConnection()
	{
		List<MapDirections> mapDirectionsList = GetMapDirectionsList();
		int count = mapDirectionsList.Count;
		int num = 0;
		if (Random.value < 0.75f)
		{
			for (int i = 0; i < count; i++)
			{
				MapDirections mapDirections = mapDirectionsList[Random.Range(0, mapDirectionsList.Count)];
				num = priorityConnDict[mapDirections].Count;
				if (num > 0)
				{
					return new ConnectionInfo(mapDirections, priorityConnDict[mapDirections][Random.Range(0, num)]);
				}
				mapDirectionsList.Remove(mapDirections);
			}
		}
		mapDirectionsList = GetMapDirectionsList();
		count = mapDirectionsList.Count;
		for (int j = 0; j < count; j++)
		{
			MapDirections mapDirections = mapDirectionsList[Random.Range(0, mapDirectionsList.Count)];
			num = connectionDict[mapDirections].Count;
			if (num > 0)
			{
				return new ConnectionInfo(mapDirections, connectionDict[mapDirections][Random.Range(0, num)]);
			}
			mapDirectionsList.Remove(mapDirections);
		}
		return new ConnectionInfo(MapDirections.None, IntVec2.zero);
	}

	private static ConnectionInfo GetDirectionConnection(MapDirections givenDir)
	{
		int count = priorityConnDict[givenDir].Count;
		if (count > 0)

			return new ConnectionInfo(givenDir, priorityConnDict[givenDir][Random.Range(0, count)]);
		}
		count = connectionDict[givenDir].Count;
		if (count > 0)
		{
			return new ConnectionInfo(givenDir, connectionDict[givenDir][Random.Range(0, count)]);
		}
		return new ConnectionInfo(MapDirections.None, IntVec2.zero);
	}

	private static ConnectionInfo GetFurthestConnectionInfo(IntVec2 givenPosition)
	{
		ConnectionInfo result = new ConnectionInfo(MapDirections.None, IntVec2.zero);
		float num = 0f;
		foreach (KeyValuePair<MapDirections, List<IntVec2>> item in priorityConnDict)
		{
			List<IntVec2> value = item.Value;
			int count = value.Count;
			for (int i = 0; i < count; i++)
			{
				float num2 = IntVec2.Distance(givenPosition, value[i]);
				if (num2 > num)
				{
					num = num2;
					result = new ConnectionInfo(item.Key, value[i]);
				}
			}
		}
		foreach (KeyValuePair<MapDirections, List<IntVec2>> item2 in connectionDict)
		{
			List<IntVec2> value = item2.Value;
			int count = value.Count;
			for (int j = 0; j < count; j++)
			{
				float num2 = IntVec2.Distance(givenPosition, value[j]);
				if (num2 > num)
				{
					num = num2;
					result = new ConnectionInfo(item2.Key, value[j]);
				}
			}
		}

	}
