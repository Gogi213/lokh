		public void LoadPlayer(Player givenPlayer, bool win = false)
		{
			if (givenPlayer == null)
			{
				return;
			}
			outfitImage.sprite = IconManager.GetWizardSprite((!win) ? "WizardDeadDown8" : "WizardJokePose0");
			outfitImage.material.SetFloat("_PaletteIndex", givenPlayer.outfitColorIndex);
			outfitImage.enabled = true;
			outfitSparkle.enabled = givenPlayer.outfitEnhanced;
			for (int i = 0; i < 6; i++)
			{
				SetSkillIcon(i, givenPlayer.assignedSkills[i]);
			}
			SetSkillIcon(6, givenPlayer.stockSkill);
			List<string> itemIDList = givenPlayer.inventory.GetItemIDList();
			int count = itemIDList.Count;
			for (int j = 0; j < 12; j++)
			{
				if (j < count)
				{
					itemIcons[j].sprite = IconManager.GetItemIcon(itemIDList[j]);
					itemIcons[j].enabled = true;
					itemBGs[j].enabled = true;
					if (givenPlayer.inventory.GetItem(itemIDList[j]).isCursed)
					{
						itemBGs[j].color = cursedItemBGColor;
						itemBGs[j].material.SetColor("_OutlineColor", cursedItemOutlineColor);
					}
					else
					{
						itemBGs[j].color = itemBGColor;
						itemBGs[j].material.SetColor("_OutlineColor", itemOutlineColor);
					}
				}
				else
				{
					itemIcons[j].enabled = false;
					itemBGs[j].enabled = false;
				}
			}

		private void SetSkillIcon(int givenIndex, Player.SkillState givenSkill)

	private IEnumerator ShowDeathSummary(float fadeInOverride = -1f)
	{
		gameWon = !GameController.AllPlayersDead();
		if (fadeInOverride > 0f)
		{
			FadeInImage(bgImage, fadeInOverride);
			yield return new WaitForSeconds(fadeInOverride);
		}
		else
		{
			FadeInImage(bgImage, bgImageFadeInTime);
			yield return new WaitForSeconds(bgImageFadeInTime + 0.5f);
		}
		statTopDivider.enabled = true;
		stats["title"].transform.parent.gameObject.SetActive(value: true);
		stats["title"].text = ((!gameWon) ? TextManager.GetUIText("deathSummary-title") : TextManager.GetUIText("bossRoom-victory"));
		statBotDivider.enabled = true;
		yield return waitForDisplayDelay;
		stats["stage"].transform.parent.gameObject.SetActive(value: true);
		string tempStageStr = GameController.CurrentStageStr;
		if (GameController.startingDifficulty > 0)
		{
			tempStageStr = "+ " + tempStageStr;
		}
		if (RunData.superSuraDefeated)
		{
			tempStageStr = "★ " + tempStageStr;
		}
		stats["stage"].text = tempStageStr;
		yield return waitForDisplayDelay;
		stats["plat"].transform.parent.gameObject.SetActive(value: true);
		stats["plat"].text = GameController.totalPlat.ToString();
		yield return waitForDisplayDelay;
		stats["time"].transform.parent.gameObject.SetActive(value: true);
		stats["time"].text = Globals.TimeToString(GameController.totalTime);
		yield return waitForDisplayDelay;
		if (Level.isBossRushMode)
		{
			stats["breakTitle"].transform.parent.gameObject.SetActive(value: true);
			stats["breakValue"].transform.parent.gameObject.SetActive(value: true);
			stats["breakTitle"].text = string.Empty;
			stats["breakValue"].text = string.Empty;
			yield return waitForShortDelay;
			for (int i = 0; i < RunData.timestampList.Count; i++)
			{
				Text text = stats["breakTitle"];
				string text2 = text.text;
				text.text = text2 + (i + 1) + ". " + RunData.timestampList[i].First + "\n";
				Text text3 = stats["breakValue"];
				text3.text = text3.text + RunData.timestampList[i].Second + "\n";
				yield return waitForShortDelay;
			}
		}
		playerRefs[0].LoadPlayer(GameController.playerScripts[0], gameWon);
		yield return waitForDisplayDelay;
		playerRefs[1].LoadPlayer(GameController.playerScripts[1], gameWon);
		yield return waitForDisplayDelay;
		confirmButtonImage.sprite = GameUI.GetInputSprite(GameController.activePlayers[0].inputDevice.inputScheme);
		FadeInImage(confirmButtonImage, 0.5f);
		stats["confirm"].transform.parent.gameObject.SetActive(value: true);
		yield return waitForDisplayDelay;
		if (!inCreditsScene)
		{
			cancelButtonImage.sprite = GameUI.GetInputSprite(GameController.activePlayers[0].inputDevice.inputScheme, "Cancel");
			FadeInImage(cancelButtonImage, 0.5f);
			stats["cancel"].transform.parent.gameObject.SetActive(value: true);
			yield return waitForDisplayDelay;
		}
		int optionSelected = 0;
		while (optionSelected == 0)
		{
			Player[] activePlayers = GameController.activePlayers;
			foreach (Player player in activePlayers)
			{
				if (!(player == null))
				{
					if (player.inputDevice.GetButtonDown("Confirm"))
					{
						optionSelected = 1;
						break;
					}
					if (!inCreditsScene && player.inputDevice.GetButtonDown("Cancel"))
					{
						optionSelected = 2;
						break;
					}
				}
			}
			yield return null;
		}
		Deactivate();
		if (optionSelected == 1)
		{
			if (inCreditsScene)
			{
				GameController.LoadLevel("Epilogue");
			}
			else
			{
				GameController.LoadLevel("PlayerRoom");
			}
		}
		else
		{
			GameController.LoadLevel("TitleScreen");
		}
		ResetStats();
		GameController.ClearLevelStats(fullReset: true);
		playerRefs[0].SetVisible(givenStatus: false);
		playerRefs[1].SetVisible(givenStatus: false);
	}
	private void FadeInImage(Image img, float fadeInTime = 1f)
	{
		img.canvasRenderer.SetAlpha(0.01f);
		img.enabled = true;
		img.CrossFadeAlpha(1f, fadeInTime, ignoreTimeScale: true);
	}
	private void ResetStats()
	{
		stats["stage"].text = string.Empty;
		stats["plat"].text = string.Empty;
		stats["time"].text = string.Empty;
		foreach (Text value in stats.Values)
		{
			value.transform.parent.gameObject.SetActive(value: false);
		}
		inCreditsScene = false;
	}

using System.Collections.Generic;
using UnityEngine;
public static class CheckpointGenerator
{
	public static void GenerateCheckpoints(TileGrid tileGrid, Transform parentTrans = null)
	{
		GameObject gameObject = Globals.FindOrCreateGameObject("Colliders", parentTrans);
		GameObject gameObject2 = Globals.FindOrCreateGameObject("Checkpoints", parentTrans);
		gameObject2.transform.SetParent(gameObject.transform);
		DestroyOldCheckpoints(gameObject2);
		List<Tile> list = new List<Tile>();
		for (int i = 0; i < tileGrid.RowCount; i++)
		{
			for (int j = 0; j < tileGrid.ColCount; j++)
			{
				if (tileGrid[i, j] == null || list.Contains(tileGrid[i, j]) || tileGrid[i, j].connection == Tile.ConnectionType.None)
				{
					continue;
				}
				Tile tile = tileGrid[i, j];
				Vector2 vector = new Vector2(j, -i);
				Vector2 vector2 = vector;
				list.Add(tile);
				int num = -1;
				float x = 1f;
				float y = 1f;
				if (tile.connection == Tile.ConnectionType.TopOptional || tile.connection == Tile.ConnectionType.TopRequired || tile.connection == Tile.ConnectionType.BottomOptional || tile.connection == Tile.ConnectionType.BottomRequired)
				{
					for (num = j + 1; num < tileGrid.ColCount && tileGrid[i, num] != null && tileGrid[i, num].connection == tile.connection; num++)
					{
						vector2 = new Vector2(num, -i);
						list.Add(tileGrid[i, num]);
					}
					Vector2 zero = Vector2.zero;
					zero = ((tile.connection != Tile.ConnectionType.TopOptional && tile.connection != Tile.ConnectionType.TopRequired) ? new Vector2(0.5f, -1.5f) : new Vector2(0.5f, 0.5f));
					vector += zero;
					vector2 += zero;
					x = Vector2.Distance(vector, vector2) + 1f;
				}
				else
				{
					for (num = i + 1; num < tileGrid.RowCount && tileGrid[num, j] != null && tileGrid[num, j].connection == tile.connection; num++)
					{
						vector2 = new Vector2(j, -num);
						list.Add(tileGrid[num, j]);
					}
					Vector2 zero2 = Vector2.zero;
					zero2 = ((tile.connection != Tile.ConnectionType.LeftOptional && tile.connection != Tile.ConnectionType.LeftRequired) ? new Vector2(1.5f, -0.5f) : new Vector2(-0.5f, -0.5f));
					vector += zero2;
					vector2 += zero2;
					y = Vector2.Distance(vector, vector2) + 1f;
				}
				GameObject gameObject3 = new GameObject();
				gameObject3.name = "Checkpoint";
				gameObject3.transform.SetParent(gameObject2.transform);
				gameObject3.transform.position = (vector + vector2) / 2f;
				gameObject3.layer = ChaosCollisions.layerPlayerDetector;
				BoxCollider2D boxCollider2D = gameObject3.AddComponent<BoxCollider2D>();
				boxCollider2D.size = new Vector2(x, y);
				boxCollider2D.isTrigger = true;
				gameObject3.AddComponent<Checkpoint>();
			}
		}
	}
	public static void DestroyOldCheckpoints(GameObject checkpoints)
	{
		Checkpoint component = checkpoints.GetComponent<Checkpoint>();
		if (component != null)
		{
			Object.DestroyImmediate(component);
		}
		List<GameObject> list = new List<GameObject>();
		Transform[] componentsInChildren = checkpoints.GetComponentsInChildren<Transform>();
		foreach (Transform transform in componentsInChildren)
		{
			if (transform != checkpoints.transform)
			{
				list.Add(transform.gameObject);
			}
		}
		list.ForEach((GameObject child) =>
		{
			Object.DestroyImmediate(child);
		});
		List<BoxCollider2D> list2 = new List<BoxCollider2D>();
		BoxCollider2D[] components = checkpoints.GetComponents<BoxCollider2D>();
		foreach (BoxCollider2D item in components)
		{
			list2.Add(item);
		}
		list2.ForEach((BoxCollider2D child) =>
		{
			Object.DestroyImmediate(child);
		});
	}

using UnityEngine;
public class ExitPortal : MonoBehaviour
{
	[HideInInspector]
	public Animator anim;
	[HideInInspector]
	public NextLevelLoader nextLevelLoader;
	private Animator arcanaAnim;
	private AnimEffect tpAnimEffect;
	private bool tpStarted;
	private bool tpStartedSkippedFrame;
	private bool playersHidden;
	public static float tpEffectHideTime = 0.1f;
	public bool queueEndGameSummary;
	public bool disableMiniArcana;
	public bool hideArcanaOnSpawn;
	private AudioSource prepareAudio;
	public void Awake()
	{
		anim = GetComponent<Animator>();
		if (!disableMiniArcana)
		{
			GameObject gameObject = Object.Instantiate(ChaosBundle.Get("Assets/Prefabs/Items/ArcanaMini.prefab"), Vector3.zero, Quaternion.identity, transform);
			gameObject.transform.localPosition = new Vector3(0f, 1.0625f, 0f);
			arcanaAnim = gameObject.GetComponent<Animator>();
			if (hideArcanaOnSpawn)
			{
				arcanaAnim.Play("Hidden");
			}
		}
	}
	public void Start()
	{
		nextLevelLoader = transform.Find("NextLevelTrigger").GetComponent<NextLevelLoader>();
		nextLevelLoader.showLevelSummary &= !Level.isBossRushMode;
		Globals.AddItemRepeller(nextLevelLoader.gameObject, 2f, "ItemRepeller");
	}
	public void Update()
	{
		if (tpStarted && !tpStartedSkippedFrame)
		{
			tpStartedSkippedFrame = true;
		}
		else
		{
			if (!tpStarted)
			{
				return;
			}
			float animTime = tpAnimEffect.GetAnimTime();
			if (!playersHidden && (animTime >= tpEffectHideTime || !tpAnimEffect.playing))
			{
				Player[] activePlayers = GameController.activePlayers;
				foreach (Player player in activePlayers)
				{
					if (player != null)
					{
						player.Hide();
					}
				}
				playersHidden = true;
			}
			else if (animTime >= 1f || !tpAnimEffect.playing)
			{
				tpStarted = false;
				LoadNextLevel();
			}
		}
	}
	public virtual void Close()
	{
		tpStarted = true;
		GameController.PauseAllPlayers();
		GameController.allies.Clear();
		DustEmitter poolItem = PoolManager.GetPoolItem<DustEmitter>();
		int particleCount = 200;
		Vector3? emitPosition = transform.position;
		poolItem.EmitCircle(particleCount, 1f, -1f, -1f, emitPosition);
		anim.Play("Teleport", -1, 0f);
		tpAnimEffect = PoolManager.GetPoolItem<AnimEffect>("TeleportEffect");
		tpAnimEffect.Play(transform.position, string.Empty);
	}
	public void LoadNextLevel()
	{
		if (queueEndGameSummary)
		{
			GameUI.deathSummary.Activate();
		}
		else
		{
			nextLevelLoader.LoadNextLevel();
		}
	}
	public void Prepare()
	{
		anim.Play("Prepare");
		if (!disableMiniArcana)
		{
			arcanaAnim.Play("Spin", -1, 0f);
		}
		prepareAudio = SoundManager.PlayIfNotPlaying("PortalShine", prepareAudio, transform, useStdPitchOverride: false);
	}
	public void Deactivate()
	{
		if (!tpStarted)
		{
			anim.Play("Deactivate", -1, 0f);
			if (!disableMiniArcana)
			{
				arcanaAnim.Play("Idle", -1, 0f);
			}
			if (prepareAudio != null)
			{
				prepareAudio.Stop();
				prepareAudio = null;
			}
		}
	}
}

	private SurvivalRoom CreateSurvivalRoom(float givenValue)
	{
		currentSRoom = Globals.ChaosInst<SurvivalRoom>(SurvivalRoom.Prefab, transform, centerSpawnLocation + Vector2.down);
		BoxCollider2D[] components = currentSRoom.GetComponents<BoxCollider2D>();
		foreach (BoxCollider2D boxCollider2D in components)
		{
			boxCollider2D.enabled = false;
		}
		miniBossSelectedIndex = (miniBossSelectedIndex + Random.Range(1, miniBossGroupCount)) % miniBossGroupCount;
		currentEnemyList = miniBossGroupList[miniBossSelectedIndex];
		initSpawnTime = 1.5f;
		CreateEnemyFactory(currentEnemyList[0], centerSpawnLocation + Vector2.down, initSpawnTime, currentSRoom.transform);
		if (useBlessedActivation)
		{
			return currentSRoom;
		}
		if (useCursedActivation)
		{
			elementSwapChance = 1f;
			int num = 0;
			int count = eeDict.Count;
			foreach (Enemy.EName value in eeDict.Values)
			{
				CreateEnemyFactory(value, centerSpawnLocation + Globals.GetCirclePositionVector(count + 1, num + 1) * enemySpawnRange, initSpawnTime + Random.Range(0.125f, 0.25f), currentSRoom.transform);
				num++;
			}
			return currentSRoom;
		}
		int num2 = ((!(givenValue > 0.7f)) ? ((givenValue > 0.4f) ? 1 : ((!(givenValue > 0.1f)) ? 3 : 2)) : 0);
		int num3 = currentEnemyList.Count - num2 - 1;
		currentEnemyList = new List<Enemy.EName>(currentEnemyList.GetRange(1, Mathf.Max(1, num3)));
		currentEnemyList.Shuffle();
		for (int j = 0; j < num3; j++)
		{
			CreateEnemyFactory(currentEnemyList[j], centerSpawnLocation + Globals.GetCirclePositionVector(num3 + 1, j + 1) * enemySpawnRange, initSpawnTime + Random.Range(0.1f, 0.5f), currentSRoom.transform);
		}
		return currentSRoom;
	}

		earthBoss.anim.Play("Intro", -1, 0f);
	}
	private IEnumerator SpawnIceBoss()
	{
		iceBossObj.SetActive(value: true);
		iceBoss.anim.Play("Intro", -1, 0f);
		yield return null;
		while (!iceBoss.anim.AnimPlayed())
		{
			yield return null;
		}
		iceWheelAnim.Play("Fold", -1, 1f);
	}
	private IEnumerator SpawnLightningBoss()
	{
		lightBossAObj.SetActive(value: true);
		lightBossBObj.SetActive(value: true);
		lightBossA.anim.Play("Land", -1, 0f);
		Animator tpAnim = Globals.ChaosInst<Animator>(ChaosBundle.Get("Assets/Prefabs/Effects/Lightning/LightningTeleportEffect.prefab"));
		tpAnim.transform.position = lightBossA.transform.position;
		tpAnim.Play("Land", -1, 0f);
		Globals.ChaosInst<LightningStrike>(LightningStrike.Prefab, null, lightBossA.transform.position);
		SoundManager.PlayWithDistAndSPR("ThunderClap", lightBossA.transform.position);
		lightBossB.anim.Play("DashEnd");
		LightningBoss.EmitStatueMetalBreak(lightBossB.transform.position);
		yield return Globals.waitForOne;
		Object.Destroy(tpAnim.gameObject);
	}
	public IEnumerator SpawnBosses()
	{
		CameraController camController = GameController.cameraController;
		Vector3 originalCameraPosition = ((!(cameraLockPosition == Vector3.zero)) ? cameraLockPosition : camController.transform.position);
		bool horiOnly = ((!overrideCameraLocks) ? camController.HoriMoveOnly : (!overrideHoriLockValue));
		bool vertOnly = ((!overrideCameraLocks) ? camController.VertMoveOnly : (!overrideVertLockValue));
		GameController.PauseAllPlayers();
		Vector3? targetLoc = finalBossObj.transform.position + bossCameraOffset;
		camController.OverrideAndMoveCameraToLocation(null, targetLoc, 2f, followPlayer: false, setLocks: true, vertLockStatus: true, horiLockStatus: true);
		yield return Globals.waitForQuarter;
		finalBoss.anim.Play("Shield", -1, 0f);
		yield return Globals.waitForHalf;
		StartCoroutine(SpawnAirBoss());
		StartCoroutine(SpawnEarthBoss());
		yield return Globals.waitForHalf;
		StartCoroutine(SpawnFireBoss());
		StartCoroutine(SpawnIceBoss());
		yield return Globals.waitForHalf;
		StartCoroutine(SpawnLightningBoss());
		yield return new WaitForSeconds(1.5f);
		playAllDialog = true;
		currentChaosID = LootManager.GetLockedChaosArcana();
		allowDialogSkip = Item.IsUnlocked(PlayerWinItem.staticID) && currentChaosID == string.Empty;
		currentDialogID = "FinalBoss-Ending0";
		while (IsInDialog)
		{
			yield return null;
		}
		if (playAllDialog)
		{
			yield return Globals.waitForOne;
			finalBoss.anim.Play("Summon", -1, 0f);
			playerWinItemTrans.position = finalBossObj.transform.position + Vector3.up * 2f;
			playerWinItemAnim.Play("Spawn", -1, 0f);
			ParticleEffect sparkleEmitter = PoolManager.GetPoolItem<ParticleEffect>("SparkleEffect");
			sparkleEmitter.followTransform = playerWinItemSpriteTrans;
			sparkleEmitter.Play(3f);
			yield return null;
			ChaosQuickStopwatch chimeStopwatch = new ChaosQuickStopwatch(0.5f);
			while (!playerWinItemAnim.AnimPlayed())
			{
				if (!chimeStopwatch.IsRunning)
				{
					chimeStopwatch.SetDelay(Random.Range(0.35f, 0.5f), stopCurrent: false, autoStart: true);
					SoundManager.PlayAudio("Chime", Random.Range(1f, 1.5f));
				}
				yield return null;
			}
			yield return Globals.waitForOne;
			currentDialogID = "FinalBoss-Ending1";
			while (IsInDialog)
			{
				yield return null;
			}
			playerWinItemAnim.Play("Hidden", -1, 0f);
			sparkleEmitter.Stop();
			PlayerRunData prd = RunData.GetWinnerData();
			GameController.activePlayers[(GameController.coopOn && prd != null) ? prd.playerID : 0].newItemNoticeUI.Display(TextManager.GetItemName(PlayerWinItem.staticID), IconManager.GetItemIcon(PlayerWinItem.staticID));
			SoundManager.PlayAudio("Powerup");
			if (!Item.IsUnlocked(PlayerWinItem.staticID))
			{
				Item.IsUnlocked(PlayerWinItem.staticID, setUnlocked: true);
			}
			finalBoss.anim.Play("SummonRecovery", -1, 0f);
			yield return new WaitForSeconds(1.5f);
		}
		else
		{
			finalBoss.anim.Play("SummonRecovery", -1, 0.35f);
		}
		if (playAllDialog && !allowDialogSkip)
		{
			currentDialogID = "FinalBoss-Ending2";
			while (IsInDialog)
			{
				yield return null;
			}
			GameController.activePlayers[0].HandleSkillUnlock(currentChaosID, isSignature: true);
			AchievementManager.UnlockAchievement(Achievement.UnlockThreeChaos);
			yield return new WaitForSeconds(1.5f);
		}
		if (playAllDialog)
		{
			currentDialogID = "FinalBoss-Ending3";
			while (IsInDialog)
			{
				yield return null;
			}
		}
		targetLoc = originalCameraPosition;
		camController.OverrideAndMoveCameraToLocation(null, targetLoc, 1f, followPlayer: false, setLocks: true, horiOnly, vertOnly);
		while (camController.overrideCameraUpdate)
		{
			yield return null;
		}
		yield return Globals.waitFor0p15;
		GameController.PauseAllPlayers(state: false);
	}
