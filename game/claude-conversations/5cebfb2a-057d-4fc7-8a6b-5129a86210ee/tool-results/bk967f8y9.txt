if (parentPlayer.assignedSkills[i] != null)

maxEleCount = eleCountDict[currentElement];
}
}
}
if (maxEleCount >= 2)
{
damageMod.modValue = (float)(maxEleCount - 1) * 0.04f;
ActivatePowerup();
}
else if (activated)
{
damageMod.modValue = 0f;
DeactivatePowerup();

=== DoubleAttackChance
public class DoubleAttackChance : Item
{
public static string staticID = "DoubleAttackChance";
private string itemAtkID = string.Empty;
private AttackInfo itemAtkInfo;
private WaitForSeconds waitForDelay = new WaitForSeconds(0.1f);
public DoubleAttackChance()
{
ID = staticID;
category = Category.Offense;
}
public override void Activate()
{
SetEventHandlers(givenStatus: true);
}
public override void Deactivate()
{
SetEventHandlers(givenStatus: false);
}
private void SetEventHandlers(bool givenStatus)
{
if (SetParentAsPlayer())
{
itemAtkID = parentSkillCategory + staticID;
Health.globalTakeDamageEnterHandlers = (Health.GlobalOnTakeDamageHandler)Delegate.Remove(Health.globalTakeDamageEnterHandlers, new Health.GlobalOnTakeDamageHandler(HandleDoubleAttack));
if (givenStatus)
{
Health.globalTakeDamageEnterHandlers = (Health.GlobalOnTakeDamageHandler)Delegate.Combine(Health.globalTakeDamageEnterHandlers, new Health.GlobalOnTakeDamageHandler(HandleDoubleAttack));
}
}
}
private void HandleDoubleAttack(AttackInfo givenInfo, Entity atkEntity, Entity hurtEntity)
{
if (!(parentPlayer == null) && !(hurtEntity == null) && !(atkEntity != parentPlayer) && !(givenInfo.attacker == itemAtkID) && parentPlayer.skillsDict.ContainsKey(givenInfo.skillID) && parentPlayer.skillsDict[givenInfo.skillID].isBasic)
{
parentPlayer.StartCoroutine(ExecuteDoubleAttack(givenInfo, hurtEntity));
}
}
private IEnumerator ExecuteDoubleAttack(AttackInfo givenInfo, Entity hurtEntity)
{
yield return waitForDelay;
if (hurtEntity != null)
{
itemAtkInfo = new AttackInfo(givenInfo);
itemAtkInfo.damage = Mathf.CeilToInt((float)itemAtkInfo.damage * 0.25f);
itemAtkInfo.knockbackVector = Vector2.zero;
itemAtkInfo.attacker = itemAtkID;
hurtEntity.health.TakeDamage(itemAtkInfo, parentEntity);
}
}
