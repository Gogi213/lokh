using Chaos.AnimatorExtensions;
using UnityEngine;
public class Dove : Enemy
{
	public class DoveIdleState : State<Dove>
	{
		public DoveIdleState(string stateName, FSM parentFSM, Dove parentEntity)
			: base(stateName, parentFSM, parentEntity)
		{
		}
		public override void OnEnter()
		{
			parent.anim.Play("Idle");
		}
		public override void Update()
		{
			parent.closestTargetInfo = Entity.GetClosestTarget(parent.targetGroup, parent.transform.position, 4.5f);
			if (parent.closestTargetInfo != null)
			{
				parent.fsm.ChangeState("Fly");
			}
			else if (!parent.disableRoam && ChaosStopwatch.Check(parent.hopStopwatchID) && Random.value <= 0.25f)
			{
				parent.hopCounter = Random.Range(1, 4);
				parent.movement.moveVector = parent.GetRandomMoveVector();
				parent.FaceTarget(parent.transform.position + (Vector3)parent.movement.moveVector, 2);
				parent.fsm.ChangeState("Roam");
			}
			else
			{
				if (parent.disableRandomTurn || !ChaosStopwatch.Check(parent.turnStopwatchID))
				{
					return;
				}
				parent.turnStopwatchID = ChaosStopwatch.Begin(parent.turnCooldown);
				if (Random.value <= 0.2f)
				{
					if (parent.facingDirection == FacingDirection.Right)
					{
						parent.facingDirection = FacingDirection.Left;
					}
					else
					{
						parent.facingDirection = FacingDirection.Right;
					}
				}
			}
		}
	}
	public class DoveRoamState : State<Dove>
	{
		private bool firstFrame;
		public DoveRoamState(string stateName, FSM parentFSM, Dove parentEntity)
			: base(stateName, parentFSM, parentEntity)
		{
		}
		public override void OnEnter()
		{
			parent.anim.Play("Hop", -1, 0f);
			firstFrame = true;
		}
		public override void Update()
		{
			if (firstFrame)
			{
				firstFrame = false;
			}
			else if (parent.anim.AnimPlayed())
			{
				parent.hopCounter--;
				if (parent.hopCounter <= 0)
				{
					parent.hopStopwatchID = ChaosStopwatch.Begin(parent.hopCooldown);
					parent.fsm.ChangeState("Idle");
				}
				else
				{
					parent.fsm.ChangeState("Roam", allowSelfTransition: true);
				}
			}
			else
			{
				parent.movement.MoveToMoveVector();
			}
		}
		public override void OnExit()
		{
			parent.movement.EndMovement();
		}
	}

			parent.spriteRenderer.sortingLayerName = "OverlayDoodad";

				{
					num = -1f;
				}
				if (parent.transform.position.x < parent.closestTargetInfo.transform.position.x)
				{
					num2 = -1f;
				}
			}
			else
			{
				num = ((!(Random.value > 0.5f)) ? 1f : (-1f));
				num2 = ((!(Random.value > 0.5f)) ? 1f : (-1f));
			}
			parent.movement.moveVector = new Vector3(Random.Range(0.75f, 1f) * num2, Random.Range(0.25f, 0.66f) * num);
			parent.movement.moveSpeedStat.AddMod(flyMoveSpeedMod);
			parent.FaceTarget(parent.transform.position + (Vector3)parent.movement.moveVector, 2);
			flyTimer = new ChaosQuickStopwatch(maxFlyTime);
			flyTimer.IsRunning = true;
		}
		public override void Update()
		{
			float num = 18f;
			if (parent.closestTargetInfo != null)
			{
				num = ((!(parent.closestTargetInfo.transform != null)) ? Vector3.Distance(parent.closestTargetInfo.position, parent.transform.position) : Vector3.Distance(parent.closes
			}
			if (!flyTimer.IsRunning || num >= 18f)
			{
				Object.Destroy(parent.gameObject);
			}
			else
			{
				parent.movement.MoveToMoveVector();
			}
		}
		public override void OnExit()
		{
			base.OnExit();
			parent.movement.moveSpeedStat.RemoveMod(flyMoveSpeedMod.ID);
		}

public class Rat : Enemy
{
	public class RatRoamState(string stateName, FSM parentFSM, Rat parentEntity) : State<Rat>(stateName, parentFSM, parentEntity)
	{
		private int roamStopwatchID;
		private float roamMinTime = 0.5f;
		private float roamMaxTime = 5f;
		private float obstructionCheckTime = 0.2f;
		private int obstructionStopwatchID;
		private Vector2 moveVector;
		public override void OnEnter()
		{
			SetRandomRoamTime();
			SetRandomMoveVector();
		}
		public override void Update()
		{
			if (ChaosStopwatch.Check(roamStopwatchID))
			{
				SetRandomRoamTime();
				SetRandomMoveVector();
			}
			if (ChaosStopwatch.Check(obstructionStopwatchID))
			{
				obstructionStopwatchID = ChaosStopwatch.Begin(obstructionCheckTime);
				if (IsMoveVectorObstructed())
				{
					SetRandomRoamTime();
					SetRandomMoveVector();
				}
			}
			parent.movement.moveVector = moveVector;
			parent.movement.MoveToMoveVector();
		}
		public bool IsMoveVectorObstructed()
		{
			return Entity.IsPathObstructed(parent.transform.position, moveVector, 0.5f);
		}
		public void SetRandomRoamTime()
		{
			roamStopwatchID = ChaosStopwatch.Begin(Random.Range(roamMinTime, roamMaxTime));
		}
		public void SetRandomMoveVector()
		{
			Vector2 availableNearbyLocation = Entity.GetAvailableNearbyLocation(parent.transform.position, excludeOrigin: true);
			moveVector = (availableNearbyLocation - (Vector2)parent.transform.position).normalized;
			moveVector = new Vector2(moveVector.x + Random.Range(-0.2f, 0.2f), moveVector.y + Random.Range(-0.2f, 0.2f));
			parent.FaceTarget(parent.transform.position + (Vector3)moveVector);
			parent.anim.PlayDirectional("Run" + parent.facingDirection);
		}
	}
	public override void Awake()
	{
		base.Awake();
		health.SetInvulnerabilityDuration(0.25f, showMsg: false);
	}
	public override void InitAudio()
	{
		minPitch = 0.8f;
		maxPitch = 1.1f;
	}
	public override void InitFSM()
	{
		fsm = new FSM();
		fsm.defaultStateName = "Roam";
		fsm.AddState(new RatRoamState("Roam", fsm, this));
		fsm.AddState(new EnemyDeadState("Dead", fsm, this));
		fsm.AddState(new EnemyFallState("Fall", fsm, this));
		fsm.ChangeState(fsm.defaultStateName);
	}
	public override void Start()
	{
		base.Start();
		GameController.enemies.Remove(gameObject);
	}
	public void OnTriggerEnter2D(Collider2D col)
	{
		if ((col.gameObject.name.Contains("FloorContact") && col.gameObject.tag == "Player") || col.gameObject.tag == "Enemy")
		{
			AttackInfo attackInfo = new AttackInfo();
			attackInfo.skillID = "Step";
			attackInfo.damage = 1;
			health.TakeDamage(attackInfo);
		}
	}
}

	public class ContestantChaseState : ChaseState
	{
		public new Contestant parent;

			: base(stateName, parentFSM, parentEntity)
		{
			parent = parentEntity;
			projectileCheck = new Collider2D[10];
		}
		public override void OnEnter()
		{
			parent.alerted = true;
			parent.anim.PlayDirectional("Run" + parent.facingDirection);
			calculatedAvoidance = Random.Range(0f, 1f);
			deviatePath = false;
			chaseStopwatchID = ChaosStopwatch.Begin(maxChaseTime);
		}
		public override void Update()
		{
			parent.closestTargetInfo = Entity.GetClosestTarget(parent.targetGroup, parent.transform.position, parent.escapeRadius, parent.onlyTargetPlayers);
			if (parent.closestTargetInfo == null || parent.closestTargetInfo.hurtBoxTransform == null)
			{
				fsm.ChangeState("GoBack");
			}
			else
			{
				if (parent.HandleExtraSkills())
				{
					return;
				}
				if (Vector2.Distance(parent.closestTargetInfo.position, parent.transform.position) < parent.alertRadius && Entity.HasLineOfSight(parent.transform.position, parent
				{
					parent.requestPathStopwatchID = -1;
					Physics2D.OverlapCircleNonAlloc(parent.transform.position, parent.alertRadius * 0.5f, projectileCheck);
					for (int i = 0; i < projectileCheck.Length; i++)
					{
						if (!(projectileCheck[i] != null) || !(projectileCheck[i].transform.parent != null))
						{
							continue;
						}
						otherProjectile = projectileCheck[i].transform.parent.gameObject.GetComponent<Projectile>();
						if (otherProjectile != null && otherProjectile.attackBox.atkInfo.skillCategory.Contains("Player"))
						{
							if (parent.tornadoReady)
							{
								fsm.ChangeState("Tornado");
								return;
							}
							if (parent.dashAvailable && parent.CalculateDashVector(dashingTowards: false, otherProjectile.transform))
							{
								fsm.ChangeState("Dash");
								return;
							}
						}
					}
					if (!deviatePath)
					{
						parent.PlayAnimOnFaceDirChange("Run", parent.closestTargetInfo.position);
					}
					distanceToTarget = parent.closestTargetInfo.distance;
					if (distanceToTarget < parent.useTornadoRange && parent.tornadoReady)
					{
						fsm.ChangeState("Tornado");
						return;
					}
					if (distanceToTarget < parent.useFireBallMaxRange && parent.fireballReady)
					{
						fsm.ChangeState("Fireball");
						return;
					}
					if (distanceToTarget > parent.useSwordThrowMinRange && distanceToTarget < parent.useSwordThrowMaxRange && parent.swordThrowReady)
					{
						fsm.ChangeState("SwordThrow");
						return;
					}
					if (distanceToTarget > 2f * parent.attackRadius && parent.dashAvailable)
					{
						avoid = calculatedAvoidance > avoidPercentage;
						if (parent.CalculateDashVector(avoid))
						{
							fsm.ChangeState("Dash");
							return;
						}
					}
					if (deviatePath)
					{
						if (!ChaosStopwatch.Check(deviateStopwatchID))
						{
							deviationDirection = parent.transform.position + (Vector3)deviationOffset;
							parent.PlayAnimOnFaceDirChange("Run", deviationDirection);
							parent.movement.MoveToTarget(deviationDirection);
							return;
						}
						deviatePath = false;
						chaseStopwatchID = ChaosStopwatch.Begin(maxChaseTime);
					}
					if (!parent.movement.MoveToTarget(parent.closestTargetInfo.transform.position, parent.attackRadius))
					{
						if (ChaosStopwatch.Check(chaseStopwatchID))
						{
							deviatePath = true;
							deviateStopwatchID = ChaosStopwatch.Begin(deviateTime);
							deviationOffset = new Vector2(Random.Range(-1, 2), Random.Range(-1, 2));
							parent.movement.EndMovement();
						}
					}
					else
					{
						fsm.ChangeState("Attack");
					}
				}
				else if (parent.DashOverPits())
				{
					parent.requestPathStopwatchID = -1;
				}
				else if (ChaosStopwatch.Check(parent.requestPathStopwatchID))
				{
					parent.requestPathStopwatchID = ChaosStopwatch.Begin(parent.requestPathTime);
					parent.followingPath = false;
					if (parent.FindAndTraversePath(parent.traversableNodeMap, parent.transform.position, parent.closestTargetInfo.position, parent.minNodeDist, "Run", appendFacingDi
					{
						fsm.ChangeState("GoBack");
					}
					else
					{
						parent.followingPath = true;
					}
				}
				else if (parent.followingPath && parent.FindAndTraversePath(parent.traversableNodeMap, parent.transform.position, parent.closestTargetInfo.position, parent.minNod
				{
					parent.followingPath = false;
					parent.HandleExitTransitions();
				}
			}
		}
	}

	public new class AttackState : State<Contestant>
	{
		private float distanceToTarget;
		private bool slash1;
		public AttackState(string stateName, FSM parentFSM, Contestant parentEntity)
			: base(stateName, parentFSM, parentEntity)
		{
		}
		public override void OnEnter()
		{
			parent.closestTargetInfo = Entity.GetClosestTarget(parent.targetGroup, parent.transform.position, parent.escapeRadius, parent.onlyTargetPlayers);
			if (parent.closestTargetInfo == null)
			{
				fsm.ChangeState("GoBack");
				return;
			}
			slash1 = false;
			parent.FaceTarget(parent.closestTargetInfo.position);
			parent.anim.PlayDirectional("Slash1" + parent.facingDirection);
			SoundManager.PlayAudioWithDistance("LightSlash", parent.transform.position);
			parent.attack.SetAttackInfo("Player", "WindSlashBasic", 2);
			parent.movement.EndMovement();
		}
		public override void Update()
		{
			float animTime = parent.anim.GetAnimTime();
			if (!slash1 && animTime >= 0.6f)
			{
				slash1 = true;
				parent.anim.PlayDirectional("Slash2" + parent.facingDirection);
				parent.attack.ChangeAttackInfoLevel(3);
				SoundManager.PlayAudioWithDistance("HeavySlash", parent.transform.position);
			}
			else
			{
				if (!slash1)
				{
					return;
				}
				if (animTime >= 0.6f && animTime < 0.9f)
				{
					parent.closestTargetInfo = Entity.GetClosestTarget(parent.targetGroup, parent.transform.position, parent.escapeRadius, parent.onlyTargetPlayers);
					if (parent.closestTargetInfo == null)
					{
						fsm.ChangeState("GoBack");
						return;
					}
					parent.FaceTarget(parent.closestTargetInfo.position);
					if (parent.swordThrowReady)
					{
						fsm.ChangeState("SwordThrow");
						return;
					}
					if (parent.fireballReady)
					{
						fsm.ChangeState("Fireball");
						return;
					}
					if (parent.tornadoReady)
					{
						fsm.ChangeState("Tornado");
						return;
					}
				}
				if (animTime >= 0.9f)
				{
					parent.HandleExitTransitions();
				}
			}
		}
		public override void OnExit()
		{
			parent.movement.EndMovement();
		}
	}

			: base(stateName, parentFSM, parentEntity)
		{
			parent = parentEntity;
		}
		public override void OnEnter()
		{
			parent.healthBar.gameObject.SetActive(value: false);
			if (parent.targetEnemyGroup)
			{
				GameController.allies.Remove(parent.gameObject);
			}
			destroyStopwatchID = ChaosStopwatch.Begin(destroyTime);

		}

		{

			{
				base.Update();
				parent.DestroyContestant();
			}
		}

	{
		base.Start();

		{

		}
		base.transform.Find("EnemyHurtBox").gameObject.name = "AllyHurtBox";
		base.transform.Find("EnemyFloorContact").gameObject.name = "AllyFloorContact";
		skillCategory = "EnemyOnly";
		foreach (Transform item in base.transform)
		{
			if (item.gameObject.layer == LayerMask.NameToLayer("Enemy"))
			{
				item.gameObject.layer = LayerMask.NameToLayer("Player");
			}
			if (item.gameObject.layer == LayerMask.NameToLayer("EnemyFloorContact"))
			{
				item.gameObject.layer = LayerMask.NameToLayer("AllyFloorContact");
			}
		}
		targetGroup = GameController.enemies;
		GameController.enemies.Remove(gameObject);
		GameController.allies.Add(gameObject);
		dropLoot = false;
	}

	private void HandleCooldowns()
	{
		if (!tornadoReady)
		{
			tornadoReady = ChaosStopwatch.Check(tornadoStopwatchID);
		}
		if (!swordThrowReady)
		{
			swordThrowReady = ChaosStopwatch.Check(swordThrowStopwatchID);
		}
		if (!fireballReady)
		{
			fireballReady = ChaosStopwatch.Check(fireballStopwatchID);
		}
		if (dashCharges < maxDashCharges - 1 && ChaosStopwatch.Check(dashStopwatchID))
		{
			dashStopwatchID = ChaosStopwatch.Begin(dashCooldown);
			dashCharges++;
		}
		if (!dashAvailable && dashCharges > 0 && ChaosStopwatch.Check(dashPauseStopwatchID))
		{
			dashAvailable = true;
		}
	}
	private bool CalculateDashVector(bool dashingTowards = true, Transform targetTransform = null)
	{
		Vector2 vector = new Vector2(Random.Range(-1f, 1f), Random.Range(-0.75f, 0.75f));
		Transform transform;
		if (targetTransform == null)
		{
			if (closestTargetInfo == null || !(closestTargetInfo.transform != null))
			{
				movement.dashVector = vector;
				return true;
			}
			transform = closestTargetInfo.transform;
		}
		else
		{
			transform = targetTransform;
		}
		if (dashingTowards)
		{
			movement.dashVector = (transform.position - base.transform.position + (Vector3)vector).normalized;
		}
		else
		{
			movement.dashVector = (base.transform.position - transform.position + (Vector3)vector).normalized;
		}
		return Entity.HasLineOfSight(movement.dashVector * 4.8f + (Vector2)base.transform.position, movement.dashVector * 6.3f + (Vector2)base.transform.position, 8f, Chaos
	}

	private void TrackFollowTransform()
	{
		if (followTransform != null && Vector2.Distance(transform.position, followTransform.position) > 3f)
		{
			destinationVector = followTransform.position;
			fsm.ChangeState("GoBack");
		}
	}
