=== Item
using System;

public abstract class Item
{
	public enum Category
	{
		Offense,
		Defense,
		Misc
	}

	public string ID = "UnnamedItem";

	public Player parentPlayer;

	public Entity parentEntity;

	public string parentSkillCategory = string.Empty;

	public string parentCombineID = string.Empty;

	public string parentGroupID = string.Empty;

	public Category category = Category.Misc;

	public bool isCursed;

	public bool hubOnly;

	public bool notForSale;

	public bool combineOnly;

	public bool isGroupItem;

	public bool destroyOnDrop;

	public bool useSimpleInfo;

	protected virtual string ExtraInfo => string.Empty;

	public virtual string FormattedExtraInfo
	{
		get
		{
			string result;
			if (ExtraInfo == string.Empty)
			{
				result = string.Empty;
			}
			else
			{
				result = ((!useSimpleInfo) ? ("\n<color=#009999>( </color><color=#00dddd>" + ExtraInfo + "</color><color=#009999> )</color>") : ("\n<color=#00dddd>" + ExtraInfo + "</color>"));
			}
			return result;
		}
	}

	public Item()
	{
	}

	protected string PercentToStr(float givenValue, string prefix = "")
	{
		return Globals.PercentToStr(givenValue, prefix);
	}

	protected string PercentToStr(NumVarStatMod givenMod, string prefix = "")
	{
		return Globals.PercentToStr(givenMod, prefix);
	}

	public virtual void Activate()
	{
	}

	public virtual void Deactivate()
	{
	}

	public bool GroupItemCreated(string givenID)
	{
		if (SetParentAsPlayer() && !parentPlayer.inventory.ContainsItem(givenID) && GroupItemManager.groupsDict.ContainsKey(givenID) && GroupItemManager.groupsDict[givenID].Contains(ID))
		{
			foreach (string item in GroupItemManager.groupsDict[givenID])
			{
				if (item != ID && parentPlayer.inventory.ContainsItem(item))
				{
					parentPlayer.inventory.AddItem(givenID, showNotice: false, ignoreMax: true);
					return true;
				}
			}
		}
		return false;
	}

	public static bool IsCursedItem(string itemID)
	{
		return LootManager.completeItemDict.ContainsKey(itemID) && LootManager.completeItemDict[itemID].isCursed;
	}

	public void SetParentSkillCategory()
	{
		if (parentEntity != null)
		{
			parentSkillCategory = parentEntity.skillCategory;
		}
	}

	public virtual bool SetParentAsPlayer()
	{
		return (parentPlayer = parentEntity as Player) != null;
	}

	public virtual void UpdateItemBar(ItemStatusBar.ItemState givenState)
	{
		if (SetParentAsPlayer())
		{
			ItemStatusBar.UpdateItem(parentPlayer.playerID, ID, givenState);
		}
	}

	public virtual void RemoveFromItemBar()
	{
		if (SetParentAsPlayer())
		{
			ItemStatusBar.RemoveItem(parentPlayer.playerID, ID);
		}
	}

	public static Item CreateItem(string givenID)
	{
		return (Item)Activator.CreateInstance(LootManager.completeItemDict[givenID].GetType());
	}

	public static bool IsUnlocked(string givenID, bool setUnlocked = false)
	{
		if (setUnlocked)
		{
			if (!GameDataManager.gameData.CheckItemDataEntry(givenID))
			{
				GameDataManager.gameData.UpdateItemDataEntry(givenID);
				GameDataManager.Save();
				AchievementManager.UnlockAchievement(Achievement.UnlockHalfRelics);
				AchievementManager.UnlockAchievement(Achievement.UnlockAllRelics);
			}
			return true;
		}
		return GameDataManager.gameData.CheckItemDataEntry(givenID);
	}

	public bool IgnoreStatusConditional(StatData givenStatData)
	{
		return !givenStatData.GetValue<bool>(StatData.isStatStr);
	}

	public bool BasicAndStatusConditional(StatData givenStatData)
	{
		return !givenStatData.GetValue<string>(StatData.idStr).Contains("Basic") && !givenStatData.GetValue<bool>(StatData.isStatStr);
	}

	public bool IgnoreBasicDashAndStatusCond(StatData givenStatData)
	{
		return BasicAndStatusConditional(givenStatData) && !givenStatData.GetValue<string>(StatData.idStr).Contains("Dash");
	}

	public static bool IsEmpowered(Player.SkillState givenSkill)
	{
		return givenSkill?.IsEmpowered ?? false;
	}

	public static bool IsNotEmpowered(Player.SkillState givenSkill)
	{
		return givenSkill != null && !givenSkill.IsEmpowered;
	}

	public static bool IsBasicSkill(Player.SkillState givenSkill)
	{
		return givenSkill != null && givenSkill.isBasic && givenSkill is Player.MeleeAttackState;
	}

	public static bool IsDashSkill(Player.SkillState givenSkill)
	{
		return givenSkill?.isDash ?? false;
	}

	public bool IsMovementSkill(Player.SkillState givenSkill)
	{
		return givenSkill?.isMovementSkill ?? false;
	}

	public bool IsJumpSkill(Player.SkillState givenSkill)
	{
		return givenSkill?.isJumpSkill ?? false;
	}

	public static bool IsMeleeSkill(Player.SkillState givenSkill)
	{
		return givenSkill?.isMeleeSkill ?? false;
	}

	public static bool IsProjectileSkill(Player.SkillState givenSkill)
	{
		return givenSkill?.isProjectileSkill ?? false;
	}

	public static bool IsSummonSkill(Player.SkillState givenSkill)
	{
		return givenSkill?.isSummonSkill ?? false;
	}

	public bool IsNegBasicSkill(Player.SkillState givenSkill)
	{
		return givenSkill is Player.NegMeleeState;
	}

	public static bool IsMinionSkill(Player.SkillState givenSkill)
	{
		return givenSkill != null && givenSkill is Player.UseBaseMinion;
	}
}

using System;
using UnityEngine;

public class OnBasicUseItem : Item
{
	protected Player.MeleeAttackState currentState;

	protected float activationChance;

	protected Vector2 currentPosition;

	protected Vector2 currentAtkPosition;

	protected Vector2 currentVector;

	public OnBasicUseItem(string givenID, float givenChance)
	{
		ID = givenID;
		category = Category.Offense;
		activationChance = givenChance;
		parentGroupID = OnBasicUseGroupItem.staticID;
	}

	public override void Activate()
	{
		SetEventHandlers(givenStatus: true);
	}

	public override void Deactivate()
	{
		SetEventHandlers(givenStatus: false);
	}

	private void SetEventHandlers(bool givenStatus)
	{
		if ((!givenStatus || !GroupItemCreated(OnBasicUseGroupItem.staticID)) && SetParentAsPlayer())
		{
			Player player = parentPlayer;
			player.skillExecEventHandlers = (Player.SkillEventHandler)Delegate.Remove(player.skillExecEventHandlers, new Player.SkillEventHandler(OnSkillExec));
			if (givenStatus)
			{
				Player player2 = parentPlayer;
				player2.skillExecEventHandlers = (Player.SkillEventHandler)Delegate.Combine(player2.skillExecEventHandlers, new Player.SkillEventHandler(OnSkillExec));
			}
		}
	}

	private void OnSkillExec(Player.SkillState givenState)
	{
		currentState = givenState as Player.MeleeAttackState;
		if (CheckActivationChance(currentState))
		{
			CreateItemEffect(currentState);
		}
	}

	public virtual bool CheckActivationChance(Player.MeleeAttackState givenState)
	{
		return givenState != null && UnityEngine.Random.value < activationChance / ((float)givenState.maxComboCount + 0.5f);
	}

	public virtual void CreateItemEffect(Player.MeleeAttackState givenState)
	{
		currentPosition = parentPlayer.transform.position;
		currentAtkPosition = parentPlayer.attackOriginTrans.position;
		currentVector = givenState.inputVector;
	}
}

using System;
using System.Collections.Generic;

public class OnCCItem : Item
{
	public delegate bool CheckForCCEvent();

	public Entity currentHurtEntity;

	public List<CheckForCCEvent> ccEventList = new List<CheckForCCEvent>();

	public bool checkCleared;

	public override void Activate()
	{
		SetEventHandlers(givenStatus: true);
	}

	public override void Deactivate()
	{
		SetEventHandlers(givenStatus: false);
	}

	public virtual void SetEventHandlers(bool givenStatus)
	{
		ccEventList.Clear();
		Health.globalTakeDamageEnterHandlers = (Health.GlobalOnTakeDamageHandler)Delegate.Remove(Health.globalTakeDamageEnterHandlers, new Health.GlobalOnTakeDamageHandler(OnGlobalTakeDamage));
		if (givenStatus)
		{
			SetCCEventList();
			Health.globalTakeDamageEnterHandlers = (Health.GlobalOnTakeDamageHandler)Delegate.Combine(Health.globalTakeDamageEnterHandlers, new Health.GlobalOnTakeDamageHandler(OnGlobalTakeDamage));
		}
	}

	public virtual void OnGlobalTakeDamage(AttackInfo givenInfo, Entity atkEntity, Entity hurtEntity)
	{
		if (hurtEntity == null || hurtEntity.fsm == null || hurtEntity.dotManager == null || givenInfo.skillCategory != parentSkillCategory || givenInfo.isStatusEffect)
		{
			return;
		}
		currentHurtEntity = hurtEntity;
		checkCleared = false;
		for (int num = ccEventList.Count - 1; num >= 0; num--)
		{
			if (ccEventList[num] != null && ccEventList[num]())
			{
				checkCleared = true;
				break;
			}
		}
		if (checkCleared)
		{
			ApplyEffect(givenInfo, atkEntity, hurtEntity);
		}
	}

	public virtual void SetCCEventList()
	{
	}

	public virtual void ApplyEffect(AttackInfo givenInfo, Entity atkEntity, Entity hurtEntity)
	{
	}

	public bool BurnCheck()
	{
		return currentHurtEntity.dotManager.HasDot(BurnDoTEffect.burnStatusStr);
	}

	public bool SlowCheck()
	{
		return currentHurtEntity.dotManager.HasDot(SlowDotEffect.slowStatusStr);
	}

	public bool PoisonCheck()
	{
		return currentHurtEntity.dotManager.HasDot(PoisonDotEffect.poisonStatusStr);
	}

	public bool ShockCheck()
	{
		return currentHurtEntity.dotManager.HasDot(ShockDotEffect.shockStatusStr);
	}

	public bool FrozenCheck()
	{
		return currentHurtEntity.fsm.currentStateName.Contains("Frozen");
	}

	public bool ChaosCheck()
	{
		return currentHurtEntity.dotManager.HasDot(ChaosDotEffect.chaosStatusStr);
	}
}

using System;

public class PreventEnemySkills : Item
{
	private static BoolVarStat inEffect = new BoolVarStat(newBaseVal: false);

	protected BoolVarStatMod preventSkillMod = new BoolVarStatMod("PreventSkillItem", newModValue: true);

	protected Type[] preventTypes;

	public override void Activate()
	{
		SetEventHandlers(givenStatus: true);
	}

	public override void Deactivate()
	{
		SetEventHandlers(givenStatus: false);
	}

	private void SetEventHandlers(bool givenStatus)
	{
		SetModStatus(givenStatus);
		GameController.levelLoadEventHandlers = (GameController.LevelLoadEventHandler)Delegate.Remove(GameController.levelLoadEventHandlers, new GameController.LevelLoadEventHandler(OnLevelLoad));
		if (givenStatus)
		{
			GameController.levelLoadEventHandlers = (GameController.LevelLoadEventHandler)Delegate.Combine(GameController.levelLoadEventHandlers, new GameController.LevelLoadEventHandler(OnLevelLoad));
		}
	}

	private void OnLevelLoad(string a, string b)
	{
		SetModStatus(givenStatus: true);
	}

	private void SetModStatus(bool givenStatus)
	{
		if (preventTypes == null)
		{
			return;
		}
		if (givenStatus)
		{
			if (!inEffect.CurrentValue)
			{
				inEffect.AddMod(preventSkillMod);
				for (int i = 0; i < preventTypes.Length; i++)
				{
					Enemy.PreventSkills(preventTypes[i], preventSkillMod, givenStatus: true);
				}
			}
		}
		else
		{
			inEffect.RemoveMod(preventSkillMod);
			for (int j = 0; j < preventTypes.Length; j++)
			{
				Enemy.PreventSkills(preventTypes[j], preventSkillMod, givenStatus: false);
			}
		}
	}
}
=== ModWithHP
using System;

public class ModWithHP : Item
{
	protected NumVarStatMod statMod;

	protected bool activated;

	protected IntComparisonType comparisonType;

	protected float HPThreshold = 0.1f;

	protected override string ExtraInfo => PercentToStr(statMod, "+");

	public override void Activate()
	{
		if (SetParentAsPlayer())
		{
			CheckActivatePowerup();
			NumVarStat healthStat = parentPlayer.health.healthStat;
			healthStat.statChangedEventHandlers = (VarStat<float>.StatChangedEventHandler)Delegate.Remove(healthStat.statChangedEventHandlers, new VarStat<float>.StatChangedEventHandler(CheckActivatePowerup));
			NumVarStat healthStat2 = parentPlayer.health.healthStat;
			healthStat2.statChangedEventHandlers = (VarStat<float>.StatChangedEventHandler)Delegate.Combine(healthStat2.statChangedEventHandlers, new VarStat<float>.StatChangedEventHandler(CheckActivatePowerup));
		}
	}

	public override void Deactivate()
	{
		if (SetParentAsPlayer())
		{
			DeactivatePowerup();
			NumVarStat healthStat = parentPlayer.health.healthStat;
			healthStat.statChangedEventHandlers = (VarStat<float>.StatChangedEventHandler)Delegate.Remove(healthStat.statChangedEventHandlers, new VarStat<float>.StatChangedEventHandler(CheckActivatePowerup));
		}
	}

	public virtual void CheckActivatePowerup()
	{
		if (parentPlayer == null || parentPlayer.health == null)
		{
			return;
		}
		if (activated)
		{
			if (!CheckHPThreshold())
			{
				DeactivatePowerup();
			}
		}
		else if (CheckHPThreshold())
		{
			ActivatePowerup();
		}
	}

	public virtual bool CheckHPThreshold()
	{
		return comparisonType switch
		{
			IntComparisonType.Equals => parentPlayer.health.HealthPercentage == HPThreshold, 
			IntComparisonType.NotEquals => parentPlayer.health.HealthPercentage != HPThreshold, 
			IntComparisonType.LessThan => parentPlayer.health.HealthPercentage < HPThreshold, 
			IntComparisonType.LessThanEquals => parentPlayer.health.HealthPercentage <= HPThreshold, 
			IntComparisonType.GreaterThan => parentPlayer.health.HealthPercentage > HPThreshold, 
			IntComparisonType.GreaterThanEquals => parentPlayer.health.HealthPercentage >= HPThreshold, 
			_ => false, 
		};
	}

	public virtual void ActivatePowerup()
	{
		activated = true;
	}

	public virtual void DeactivatePowerup()
	{
		activated = false;
	}
}

using System;
using System.Collections.Generic;
using UnityEngine;

public class ODIncreaseOnUse : Item
{
	public static string staticID = "ODIncreaseOnUse";

	private static Dictionary<int, float> idDamageDict = new Dictionary<int, float>();

	private NumVarStatMod damageMod;

	private int parentID = -1;

	private Player.SkillState signatureSkill;

	private float DamageModValue
	{
		get
		{
			return (!idDamageDict.ContainsKey(parentID)) ? 0f : idDamageDict[parentID];
		}
		set
		{
			idDamageDict[parentID] = value;
			damageMod.modValue = value;
		}
	}

	protected override string ExtraInfo => PercentToStr(DamageModValue, "+");

	public ODIncreaseOnUse()
	{
		ID = staticID;
		category = Category.Offense;
		RunData.runResetEventHandlers = (RunData.RunEvent)Delegate.Remove(RunData.runResetEventHandlers, new RunData.RunEvent(OnRunReset));
		RunData.runResetEventHandlers = (RunData.RunEvent)Delegate.Combine(RunData.runResetEventHandlers, new RunData.RunEvent(OnRunReset));
		damageMod = new NumVarStatMod(ID, 0f, 10, VarStatModType.Multiplicative);
	}

	private static void OnRunReset()
	{
		idDamageDict.Clear();
	}

	public override void Activate()
	{
		SetModStatus(givenStatus: true);
	}

	public override void Deactivate()
	{
		SetModStatus(givenStatus: false);
	}

	private void SetModStatus(bool givenStatus)
	{
		if (SetParentAsPlayer())
		{
			parentID = parentPlayer.playerID;
			ApplyDamageMod(givenStatus);
			Player player = parentPlayer;
			player.skillEnterEventHandlers = (Player.SkillEventHandler)Delegate.Remove(player.skillEnterEventHandlers, new Player.SkillEventHandler(OnSkillUse));
			if (givenStatus)
			{
				Player player2 = parentPlayer;
				player2.skillEnterEventHandlers = (Player.SkillEventHandler)Delegate.Combine(player2.skillEnterEventHandlers, new Player.SkillEventHandler(OnSkillUse));
			}
		}
	}

	private void OnSkillUse(Player.SkillState givenSkill)
	{
		if (givenSkill != null && givenSkill.isUltimate && Globals.InGameScene(string.Empty))
		{
			ApplyDamageMod(givenStatus: true, givenSkill, 0.01f);
		}
	}

	private void ApplyDamageMod(bool givenStatus, Player.SkillState givenSkill = null, float givenValue = 0f)
	{
		if (signatureSkill != null)
		{
			signatureSkill.skillData.RemoveMod(StatData.odDmgStr, damageMod.ID);
		}
		if (givenSkill != null)
		{
			signatureSkill = givenSkill;
			if (givenStatus)
			{
				DamageModValue = Mathf.Min(DamageModValue + givenValue, 0.5f);
				signatureSkill.skillData.AddMod(StatData.odDmgStr, damageMod);
			}
		}
	}
