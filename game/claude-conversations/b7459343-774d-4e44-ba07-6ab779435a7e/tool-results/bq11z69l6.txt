	private float GetPhysicalAttackVal(Particle particle, int type, float elementalRingFac, Player p)
	{
		switch (particle.flags)
		{
		case 37:
			if (type == 0)
			{
				return 1f * elementalRingFac;
			}
			return 0f;
		case 235:
			return GetShieldAttackVal(p, type) * elementalRingFac;
		case 189:
		case 190:
		case 218:
		case 247:
			return GetFistAttackVal(particle, type, p) * elementalRingFac;
		default:
			return GetDefaultAttackVal(p, particle, type, elementalRingFac);
		}
	}
	private float ConvertRangedDamage(InvLoot weap, int dType, float aVal, Particle particle)
	{
		int flags = particle.flags;
		switch (flags)
		{
		case 11:
			return aVal *= 1.1f;
		case 12:
		case 13:
		case 14:
		case 16:
		case 17:
			aVal /= 2f;
			break;
		}
		if ((flags == 12 && dType == 2) || (flags == 13 && dType == 3) || (flags == 14 && dType == 5) || (flags == 17 && dType == 9) || (flags == 16 && dType == 8))
		{
			aVal += GetBaseDamage(weap) / 2f;
		}
		return aVal;
	}
	public float GetCharmAdjustedPhysicalAttackVal(float aVal, int type, Player p, InvLoot weap)
	{
		if (!GameModeMgr.IsGameEnhanced())
		{
			aVal = GetGlobalModifiedDamage(aVal, p, weap);
			aVal += GetCharmOnlyDamage(weap, type);
		}
		else
		{
			aVal += GetCharmOnlyDamage(weap, type);
			aVal = GetGlobalModifiedDamage(aVal, p, weap);
		}
		return aVal;
	}
	public float GetBaseDamage(InvLoot weap)
	{
		if (c.GetPlayer() == null || weap == null || weap.category < 0 || weap.catalogIdx < 0)
		{
			return 0f;
		}
		float num = weap.GetScaledValue(0, c.GetPlayer(), c.GetPlayer().stats, c.equipment.twoHanded[c.equipment.loadoutIdx], null);
		switch (LootCatalog.category[weap.category].loot[weap.catalogIdx].flags)
		{
		case 1:
			num += weap.GetScaledValue(6, c.GetPlayer(), c.GetPlayer().stats, c.equipment.twoHanded[c.equipment.loadoutIdx], null);
			break;
		case 2:
			num += weap.GetScaledValue(7, c.GetPlayer(), c.GetPlayer().stats, c.equipment.twoHanded[c.equipment.loadoutIdx], null);
			break;
		case 5:
			num += weap.GetScaledValue(12, c.GetPlayer(), c.GetPlayer().stats, c.equipment.twoHanded[c.equipment.loadoutIdx], null);
			break;
		case 6:
			num += weap.GetScaledValue(13, c.GetPlayer(), c.GetPlayer().stats, c.equipment.twoHanded[c.equipment.loadoutIdx], null);
			break;
		case 3:
			num += weap.GetScaledValue(9, c.GetPlayer(), c.GetPlayer().stats, c.equipment.twoHanded[c.equipment.loadoutIdx], null);
			break;
		}
		return num;
	}

		{
			aVal *= 1.1f;
		}

		if (c.stats.GetCharmFlag(15) && p.salt > 0)

			aVal *= 1.25f;
		}
		if (c.stats.GetCharmFlag(13) && c.hp <= c.stats.GetMaxHP() * 0.25f)
		{
			aVal *= 1.5f;
		}
		if (c.buffs.GetGoldenBuff())
		{
			float num = 1.5f;
			if (GameModeMgr.IsGameEnhanced())
			{
				num = 1.35f;
				float num2 = 1.42f;
				if (weap == null)
				{
					num = num2;
				}
				else
				{
					LootDef lootDef = LootCatalog.category[weap.category].loot[weap.catalogIdx];
					if (lootDef.flags < 1 || lootDef.flags > 6)
					{
						num = num2;

				}
			}
			aVal *= num;
		}
