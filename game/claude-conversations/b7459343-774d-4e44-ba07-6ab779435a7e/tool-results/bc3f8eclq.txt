def fp_paths(*paths) -> str:
    h = hashlib.sha1()
    for p in paths:
        p = Path(p) if p else None
        if p is None or not p.exists():
            h.update(b"-")
            continue
        if p.is_file():
            s = p.stat()
            h.update(f"{p.name}:{s.st_size}:{s.st_mtime_ns}".encode())
        else:
            n = tot = mx = 0
            for f in p.rglob("*"):
                if f.is_file() and ".gen_state" not in f.name and f.suffix != ".tmp":
                    s = f.stat()
                    n += 1; tot += s.st_size; mx = max(mx, s.st_mtime_ns)
            h.update(f"{p.name}:{n}:{tot}:{mx}".encode())
    return h.hexdigest()[:16]
