    74		public float GetMaxHP()
    75		{
    76			if (c.playerIdx > -1)
    77			{
    78				float hP = c.GetPlayer().stats.GetHP();
    79				if (c.hpCapDmg > hP / 2f)
    80				{
    81					c.hpCapDmg = hP / 2f;
    82				}
    83				return hP - c.hpCapDmg;
    84			}
    85			MonsterDef obj = MonsterCatalog.catalog[c.monsterIdx];
    86			float num = obj.hp;
    87			Player mainPlayer = PlayerMgr.GetMainPlayer();
    88			if (mainPlayer != null && mainPlayer.IsScaledRandomizer())
    89			{
    90				num = RandomizerMgr.GetTierAdjustHp(c);
    91			}
    92			else if (GameModeMgr.IsGameEnhanced())
    93			{
    94				num = GetPlaythroughAdjustedBaseHp(c);
    95			}
    96			if (GameStateManager.coopPlayerIdx > -1)
    97			{
    98				num *= 1.5f;
    99			}
   100			if (GameStateManager.playthroughCount > 0 && !GameModeMgr.IsGameEnhanced())
   101			{
   102				num = num + num * 0.5f + num * 0.1f * (float)GameStateManager.playthroughCount + (float)(50 * GameStateManager.playthroughCount) + 80f;
   103				if (c.boss)
   104				{
   105					num += 2000f + (float)GameStateManager.playthroughCount * 500f;
   106				}
   107			}
   108			else if (GameStateManager.playthroughCount > 0 && GameModeMgr.IsGameEnhanced())
   109			{
   110				if (c.boss)
   111				{
   112					if (GameStateManager.playthroughCount > 1)
   113					{
   114						num += num * 0.07f * ((float)GameStateManager.playthroughCount - 1f) + ((float)GameStateManager.playthroughCount - 1f) * 400f;
   115					}
   116				}
   117				else
   118				{
   119					num = num + num * 0.5f + num * 0.1f * (float)GameStateManager.playthroughCount + (float)(50 * GameStateManager.playthroughCount) + 50f;
   120					if (GameStateManager.playthroughCount > 1)
   121					{
   122						num += 50f;
   123					}
   124				}
   125			}
   126			if (obj.name == "saltbat")
   127			{
   128				Player mainPlayer2 = PlayerMgr.GetMainPlayer();
   129				if (mainPlayer2 != null)
   130				{
   131					return mainPlayer2.stats.level;
   132				}
   133			}
   134			return num;
   135		}

   137		public float GetAdjustedDamage(float damage, int damageType)
   138		{
   139			float num = 0f;
   140			switch (damageType)
   141			{
   142			case 0:
   143				num = GetDefense(0);
   144				break;
   145			case 2:
   146				num = GetDefense(1);
   147				break;
   148			case 3:
   149				num = GetDefense(2);
   150				break;
   151			case 6:
   152				num = GetDefense(3);
   153				break;
   154			case 5:
   155				num = GetDefense(4);
   156				break;
   157			case 8:
   158				num = GetDefense(5);
   159				break;
   160			case 9:
   161				num = GetDefense(6);
   162				break;
   163			}
   164			if (c.equipment.active && c.buffs.GetProtect())
   165			{
   166				num *= 2f;
   167			}
   168			if (c.buffs.GetBloodied() && damageType != 8)
   169			{
   170				if (num > 0f)
   171				{
   172					num = 0f;
   173				}
   174				if (!GameModeMgr.IsGameEnhanced())
   175				{
   176					damage *= 1.25f;
   177				}
   178			}
   179			float num2 = 0f;
   180			num2 = num / (num + 200f);
   181			return damage - damage * num2;
   182		}

   184		public float GetDefense(int stat, PlayerStats stats)
   185		{
   186			float num = 0f;
   187			if (c.playerIdx > -1)
   188			{
   189				if (c.equipment.active)
   190				{
   191					Player player = c.GetPlayer();
   192					if (c.equipment.helm.catalogIdx > -1 && c.equipment.helm.catalogIdx < LootCatalog.category[2].loot.Length && c.equipment.helm.invIdx > -1 && stats.Equippable(LootCatalog.category[2].loot[c.equipment.helm.catalogIdx], -1) == 2)
   193					{
   194						num += player.playerInv.inventory[c.equipment.helm.invIdx].GetValue(stat, player);
   195					}
   196					if (c.equipment.armor.catalogIdx > -1 && c.equipment.armor.catalogIdx < LootCatalog.category[2].loot.Length && c.equipment.armor.invIdx > -1 && stats.Equippable(LootCatalog.category[2].loot[c.equipment.armor.catalogIdx], -1) == 2)
   197					{
   198						num += player.playerInv.inventory[c.equipment.armor.invIdx].GetValue(stat, player);
   199					}
   200					if (c.equipment.gloves.catalogIdx > -1 && c.equipment.gloves.catalogIdx < LootCatalog.category[2].loot.Length && c.equipment.gloves.invIdx > -1 && stats.Equippable(LootCatalog.category[2].loot[c.equipment.gloves.catalogIdx], -1) == 2)
   201					{
   202						num += player.playerInv.inventory[c.equipment.gloves.invIdx].GetValue(stat, player);
   203					}
   204					if (c.equipment.boots.catalogIdx > -1 && c.equipment.boots.catalogIdx < LootCatalog.category[2].loot.Length && c.equipment.boots.invIdx > -1 && stats.Equippable(LootCatalog.category[2].loot[c.equipment.boots.catalogIdx], -1) == 2)
   205					{
   206						num += player.playerInv.inventory[c.equipment.boots.invIdx].GetValue(stat, player);
   207					}
   208					num += GetPlayerLevelDefenseBonus(player, stat);
   209					if (stat == 7)
   210					{
   211						num *= 0.5f;
   212					}
   213					num += GetRingDefense(stat);
   214					if (c.buffs.GetWillBuff() > 0f && GameModeMgr.IsGameEnhanced() && stat == 7)
   215					{
   216						num *= 1.5f;
   217					}
   218				}
   219				switch (stat)
   220				{
   221				case 0:
   222				case 3:
   223				case 4:
   224					num += c.buffs.GetPhysDefBuff(23);
   225					if (c.buffs.GetGoldenBuff())
   226					{
   227						num = ((!GameModeMgr.IsGameEnhanced() || stat == 4 || c.playerIdx <= -1) ? (num + 200f) : (num + 200f));
   228					}
   229					break;
   230				case 1:
   231				case 2:
   232				case 5:
   233				case 6:
   234					num += c.buffs.GetPhysDefBuff(24);
   235					if (c.buffs.GetGoldenBuff())
   236					{
   237						num += 100f;
   238					}
   239					break;
   240				}
   241				if (c.buffs.GetProtect() && stat != 7)
   242				{
   243					num *= 2f;
   244				}
   245			}
   246			return num;
   247		}
