   252			private Vector2 GetStrafeVector(Vector2 givenTargetLoc)
   254				useStrafe = true;
   255				strafeRight = Globals.RandBool;
   256				currentPosition = parent.transform.position;
   257				calcDist = Vector2.Distance(currentPosition, givenTargetLoc);
   258				if (calcDist > 2.5f)
   260					calcVector = givenTargetLoc - currentPosition;
   261					perpendVec = Globals.GetPerpendicular(calcVector.normalized, strafeRight);
   262					calcPosition = givenTargetLoc + perpendVec * calcVector.magnitude;
   263					calcVector = (calcPosition - currentPosition).normalized;
   264					calcPosition = currentPosition + calcVector * dashRange;
   265					if (Globals.CheckLine(currentPosition, calcPosition, ChaosCollisions.layerAllWallAndObst))
   267						return calcVector;
   269					strafeRight = !strafeRight;
   270					calcPosition = givenTargetLoc - perpendVec * calcVector.magnitude;
   271					calcVector = (calcPosition - currentPosition).normalized;
   272					calcPosition = currentPosition + calcVector * dashRange;
   273					if (Globals.CheckLine(currentPosition, calcPosition, ChaosCollisions.layerAllWallAndObst))
   275						return calcVector;
   278				useStrafe = false;
   279				calcVector = (currentPosition - givenTargetLoc).normalized;
   280				if (Globals.CheckCircle(currentPosition + calcVector * 3f, 0.25f, ChaosCollisions.layerAllWallAndObst))
   282					return calcVector;
   284				return -calcVector;
