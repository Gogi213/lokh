using UnityEngine;
public class FireWall : Wave
{
private new static GameObject prefab;
private bool colliderFlip;
private bool waveActive;
[HideInInspector]
public int hitCount = 30;
[HideInInspector]
public float hitInterval = 0.2f;
private int hitStopwatch;
private int middleWave;
private float soundInterval = 1f;
private int soundStopwatchID = -1;
private AudioSource audioSource;
private ParticleSystem[] fireWallEmitters;
[HideInInspector]
public bool lowParticleMode;
private ParticleSystem.EmissionModule emitMod;
private ParticleSystem.MinMaxCurve emitValue;
private Vector2 effectVec;
public static GameObject Prefab => prefab ?? (prefab = ChaosBundle.Get("Assets/Prefabs/AreaEffects/FireWall.prefab"));
public override void Start()
{
base.Start();
hitStopwatch = ChaosStopwatch.Begin(hitInterval, useInterval: true, hitInterval, hitCount);
middleWave = Mathf.CeilToInt((float)wavePoolSize * 0.5f);
fireWallEmitters = new ParticleSystem[wavePoolSize];
for (int i = 0; i < wavePoolSize; i++)
{
fireWallEmitters[i] = wavePoolTrans[i].Find("FireEmitter").GetComponent<ParticleSystem>();
if (lowParticleMode)
{
emitMod = fireWallEmitters[i].emission;
emitValue = emitMod.rateOverTime;
emitValue.constant *= 0.6f;
emitMod.rateOverTime = emitValue;
}
}
}
public override void Update()
{
base.Update();
if (!waveActive)
{
return;
}
if (ChaosStopwatch.Check(soundStopwatchID) && (audioSource == null || !audioSource.isPlaying) && wavePoolTrans[middleWave] != null)
{
soundStopwatchID = ChaosStopwatch.Begin(soundInterval);
audioSource = SoundManager.PlayAudioWithDistance("FireWallLoop", wavePoolTrans[middleWave].position);
}
switch (ChaosStopwatch.CheckInterval(hitStopwatch))
{
case StopwatchState.Running:
if (!colliderFlip)
{
colliderFlip = true;
EnableAttackColliders(colliderFlip);
}
break;
case StopwatchState.Ready:
colliderFlip = false;
EnableAttackColliders(colliderFlip);
break;
}
}
public override void ActivateWave()
{
base.ActivateWave();
if (!layerMaskHit)
{
waveActive = true;
effectVec = wavePoolTrans[waveCount].position;
FireBurst poolItem = PoolManager.GetPoolItem<FireBurst>();
int? particleCount = 3;
Vector3? position = effectVec;
Vector3? localEulerAngles = Globals.GetRotationVector(Vector2.up);
poolItem.EmitDouble(particleCount, position, null, localEulerAngles);
if (Globals.CheckCircle(effectVec, 0.5f, ChaosCollisions.layerPit))
{
PoolManager.GetPoolItem<ParticleEffect>("BurnMarkMedEmitter").Emit(1, effectVec, new ParticleSystemOverride
{
startLifetime = destroyDelay + 1.5f
});
}
}
}
public override void DestroyWave()
{
ChaosStopwatch.Stop(destroyStopwatchID);
if (setToDestroy && audioSource != null)
{
audioSource.Stop();
}
if (destroyInSequence && !destroySequenceStarted)
{
destroySequenceStopwatchID = ChaosStopwatch.Begin(0f, useInterval: true, spawnRate, wavePoolSize);
destroySequenceStarted = true;
}
switch (ChaosStopwatch.CheckInterval(destroySequenceStopwatchID))
{
case StopwatchState.Ready:
waveCount++;
waveCount %= wavePoolSize;
if (wavePoolAttacks[waveCount] != null && wavePoolAttacks[waveCount].collider != null)
{
wavePoolAttacks[waveCount].collider.gameObject.SetActive(value: false);
}
if (wavePoolTrans[waveCount] != null)
{
fireWallEmitters[waveCount].Stop();
if (waveActive)
{
PoolManager.GetPoolItem<ParticleEffect>("SmokeEmitter").Emit(4, wavePoolTrans[waveCount].position + Vector3.up * 1.5f, SmokeEmitter.SmokeSmallOverride);
}
}
if (wavePoolColDetectors[waveCount] != null)
{
wavePoolColDetectors[waveCount].SetActive(value: false);
}
break;
case StopwatchState.Done:
if (!setToDestroy && gameObject != null)
{
setToDestroy = true;
Object.Destroy(gameObject, 1f);
}
break;
}
}
public void EnableAttackColliders(bool status = true)
{
for (int i = 0; i < wavePoolSize; i++)
{
if (wavePoolAttacks[i] != null && wavePoolAttacks[i].collider != null)
{
wavePoolAttacks[i].collider.enabled = status;
}
}
}
}

using UnityEngine;
public class FireZone : MonoBehaviour
{
private static GameObject prefab;
public Attack attack;
public CircleCollider2D attackCol;
public ParticleSystem firePartSys;
private ParticleSystem.MainModule fireMainMod;
private ParticleSystem.ShapeModule fireShapeMod;

private bool expandCompleted;
private float expandLerpVal;
private float lerpedRadius;
private int attackStopwatchID;
private int attackCount = 8;
private float attackInterval = 0.125f;
[HideInInspector]
public bool durationCompleted;
private bool createBurst = true;
public static GameObject Prefab => prefab ?? (prefab = ChaosBundle.Get("Assets/Prefabs/AreaEffects/FireZone.prefab"));
private void Awake()
{
if (firePartSys != null)
{
fireMainMod = firePartSys.main;
fireShapeMod = firePartSys.shape;
}
}
private void Start()
{
SoundManager.PlayAudioWithDistance("FlameSlam", transform.position);
attackStopwatchID = ChaosStopwatch.Begin(0f, useInterval: true, attackInterval, attackCount + 1);
expandStopwatch.IsRunning = true;
}
private void FixedUpdate()
{
if (durationCompleted)
{
return;
}
switch (ChaosStopwatch.CheckInterval(attackStopwatchID))
{
case StopwatchState.Running:
attackCol.enabled = false;
break;
case StopwatchState.Ready:
if (ChaosStopwatch.CheckInterval(attackStopwatchID, getCharge: false) != StopwatchState.Done)
{
attackCol.enabled = true;
}
break;
case StopwatchState.Done:
{
attackCol.enabled = false;
durationCompleted = true;
fireShapeMod.radius = 0.25f;
fireShapeMod.arcMode = ParticleSystemShapeMultiModeValue.Random;
fireMainMod.startSpeed = 3f;
firePartSys.Stop();
if (createBurst && attack.atkInfo != null)
{
firePartSys.Emit(50);
FlameBurst.CreateBurst(transform.position, attack.atkInfo.skillCategory, attack.atkInfo.skillID, 2, maxRadius + 0.25f);
PoolManager.GetPoolItem<ParticleEffect>("BurnMarkLargeEmitter").Emit(1, transform.position, ParticleEffectOverrides.StartLifeTimeTwo);
CameraController.ShakeCamera(0.3f);
}
for (int i = 0; i < 12; i++)
{
PoolManager.GetPoolItem<ParticleEffect>("SmokeEmitter").Emit(3, transform.position + (Vector3)Globals.GetCirclePositionVector(12, i) * maxRadius * 0.9f);
}
Object.Destroy(gameObject, 1f);
break;
}
}
if (!expandCompleted)
{
expandLerpVal = expandStopwatch.TimePercentage;
fireShapeMod.arcSpeed = Mathf.Lerp(0.5f, 2f, expandLerpVal);
lerpedRadius = Mathf.Lerp(minRadius, maxRadius, expandLerpVal);
fireShapeMod.radius = lerpedRadius - 0.5f;
attackCol.radius = lerpedRadius;
expandCompleted = expandLerpVal >= 1f;
}
}
public void Reset()
{
ChaosStopwatch.Stop(attackStopwatchID);
createBurst = false;
expandCompleted = true;
}
}

##### FlakField
using System;
using UnityEngine;
public class FlakField : MonoBehaviour
{
private static GameObject prefab;
public SpriteRenderer[] flakSprites;
private Vector2[] flakPositions = new Vector2[8];
public Attack attack;
public float atkRadius = 2f;
public bool isEmpowered;
private int maxHitCount = 5;
private float hitInterval = 0.125f;
private int hitStopwatchID;
private int hitIndex;
private bool destroyQueued;
private Vector2 currentPos;
public static GameObject Prefab
{
get
{
if (prefab == null)
{
prefab = ChaosBundle.Get("Assets/Prefabs/AreaEffects/FlakField.prefab");
}
return prefab;
}
}
private void Start()
{
currentPos = transform.position;
for (int i = 0; i < maxHitCount && i < flakPositions.Length; i++)
{
ref Vector2 reference = ref flakPositions[i];
reference = currentPos + UnityEngine.Random.insideUnitCircle * (atkRadius * 0.75f);
flakSprites[i].transform.position = flakPositions[i];
flakSprites[i].enabled = true;
}
this.attack.transform.localScale = new Vector3(atkRadius, atkRadius, 1f);
Attack attack = this.attack;
attack.attackCollisionEventHandlers = (Attack.AttackCollisionEventHandler)Delegate.Remove(attack.attackCollisionEventHandlers, new Attack.AttackCollisionEventHandler(OnColHit));
Attack attack2 = this.attack;
attack2.attackCollisionEventHandlers = (Attack.AttackCollisionEventHandler)Delegate.Combine(attack2.attackCollisionEventHandlers, new Attack.AttackCollisionEventHandler(OnColHit));
hitStopwatchID = ChaosStopwatch.Begin(hitInterval, useInterval: true, hitInterval, maxHitCount);
hitIndex = 0;
FlameBurst.CreateBurst(currentPos, string.Empty, string.Empty, 1, 2f);
PoolManager.GetPoolItem<ParticleEffect>("FloorCraterLarge").Emit(1, currentPos);
SoundManager.PlayWithDistAndSPR("VacuumExplosionForm", currentPos, 1.1f);
CameraController.ShakeCamera(0.4f);
}
private void FixedUpdate()
{
if (!destroyQueued)
{
switch (ChaosStopwatch.CheckInterval(hitStopwatchID))
{
case StopwatchState.Running:
attack.collider.enabled = true;
break;
case StopwatchState.Ready:
flakSprites[hitIndex].enabled = false;
attack.collider.enabled = false;
PlayFX(flakPositions[hitIndex]);
hitIndex++;
break;
case StopwatchState.Done:
FlameBurst.CreateBurst(currentPos, attack.GetAtkInfoAtLevel((!isEmpowered) ? 2 : 3), atkRadius + 0.5f);
SoundManager.PlayAudioWithDistance("VacuumExplosion", currentPos);
Disable();
break;
}
}
}
private void OnColHit(Collider2D col)
{
attack.knockbackOverwriteVector = UnityEngine.Random.insideUnitCircle;
}
private void PlayFX(Vector2 givenPos)
{
FireExplosion.SpawnExplosion(givenPos, 1f, 1f, 12, 8);
PoolManager.GetPoolItem<ParticleEffect>("SmokeEmitter").Emit(10, givenPos);
PoolManager.GetPoolItem<ParticleEffect>("BurnMarkMedEmitter").Emit(1, givenPos + Vector2.down * 0.25f, ParticleEffectOverrides.StartLifeTimeTwo);
SoundManager.PlayWithDistAndSPR("ExplosionPop", givenPos, 0.9f);
}
private void Disable()
{
if (!destroyQueued)
{
destroyQueued = true;
Attack attack = this.attack;
attack.attackCollisionEventHandlers = (Attack.AttackCollisionEventHandler)Delegate.Remove(attack.attackCollisionEventHandlers, new Attack.AttackCollisionEventHandler(OnColHit));
UnityEngine.Object.Destroy(gameObject, hitInterval);
}
}
}

public bool quickArc;

InitArcVars();
Attack attack = attackBox;
attack.attackCollisionEventHandlers = (Attack.AttackCollisionEventHandler)Delegate.Remove(attack.attackCollisionEventHandlers, new Attack.AttackCollisionEventHandler(OnTargetHit));
Attack attack2 = attackBox;

{
return;

{

{
boomerangReturning = true;
}
if (parentTrans == null || Vector2.Distance(transform.position, parentTrans.position) < returningSnapDistance)
{
ResetProjectile();

}

{
attackBox.knockbackOverwriteVector = moveVector;

if (!ignoreArc)
{
calcVector = initMoveVector * moveBias + Vector2.Lerp(leftBound, rightBound, arcLerpValue) * arcBias;
moveVector = (moveVector + calcVector).normalized;

arcStopwatch.IsRunning = true;
initMoveVector *= -1f;

attackBox.ChangeAttackInfoLevel(attackBox.atkInfo.skillLevel + 1);

ignoreArc = false;

else if (parentTrans != null)

calcVector = (parentTrans.position - transform.position).normalized * returnBias;
prevVector = moveVector;

}

roteAngle += (float)Math.PI;
leftBound = new Vector2(Mathf.Cos(roteAngle), Mathf.Sin(roteAngle)).normalized;
}
else
{
leftBound = new Vector2(Mathf.Cos(roteAngle), Mathf.Sin(roteAngle)).normalized;
roteAngle += (float)Math.PI;
rightBound = new Vector2(Mathf.Cos(roteAngle), Mathf.Sin(roteAngle)).normalized;
}

using UnityEngine;
public class FlameSwirl : MonoBehaviour
{
private static GameObject prefab;
private Rigidbody2D rigidbody2D;
[HideInInspector]
public int swirlCount = 1;
[HideInInspector]
public float swirlTime = 0.5f;
private ChaosQuickStopwatch swirlStopwatch;
[HideInInspector]
public Attack attack;
private Transform attackTrans;
private BoxCollider2D attackCollider;
public ParticleSystem[] fireEmitters;
public ParticleSystem trailEmitter;
private ParticleSystem.ShapeModule[] fireEmitterShapeMods;
private bool emittersActive = true;
private Vector3 rotationVector;
private float startRoteValue;
private int lastRoteCount;
private int lastRoteThreshold;
[HideInInspector]
public Transform followTrans;
private float lerpValue;
private float lerpBound;
[HideInInspector]
public bool isEmpowered;
private bool attackCompleted;
private bool destroyQueued;
public bool destroyImmediate;
private static Vector3 empoweredEmitterOffset = new Vector3(0f, -1.75f, 0f);
private static Vector3 empoweredTrailOffset = new Vector3(0f, -3.75f, 0f);
public static GameObject Prefab => prefab ?? (prefab = ChaosBundle.Get("Assets/Prefabs/AreaEffects/FlameSwirl.prefab"));
private void Awake()
{
rigidbody2D = GetComponent<Rigidbody2D>();
attackTrans = transform.Find("Attack");
attack = attackTrans.GetComponent<Attack>();
attackCollider = attackTrans.GetComponent<BoxCollider2D>();
}
private void Start()
{
rotationVector = Vector3.zero;
startRoteValue = transform.localEulerAngles.z;
lerpBound = 360f * (float)swirlCount + startRoteValue;
(swirlStopwatch = new ChaosQuickStopwatch(swirlTime * (float)swirlCount)).IsRunning = true;
if (!isEmpowered)
{
return;
}
attackCollider.size = new Vector2(0.5f, 3.5f);
attackCollider.offset = new Vector2(0f, -1.75f);
fireEmitterShapeMods = new ParticleSystem.ShapeModule[fireEmitters.Length];
for (int i = 0; i < fireEmitters.Length; i++)
{
ParticleSystem particleSystem = fireEmitters[i];
if (i == 2)
{
particleSystem.transform.localPosition = empoweredTrailOffset;
continue;
}
particleSystem.transform.localPosition = empoweredEmitterOffset;
ref ParticleSystem.ShapeModule reference = ref fireEmitterShapeMods[i];
reference = particleSystem.shape;
fireEmitterShapeMods[i].radius = 1.75f;
}
}
private void FixedUpdate()
{
if (followTrans != null)
{
rigidbody2D.MovePosition(followTrans.position);
}
if (attackCompleted)
{
return;
}
lerpValue = Globals.GetSmoothLerpValue(swirlStopwatch.TimePercentage);
rotationVector.z = Mathf.Lerp(-30f, 30f, lerpValue);
attackTrans.localEulerAngles = rotationVector;
rotationVector.z = Mathf.Lerp(startRoteValue, lerpBound, lerpValue);
if (lastRoteCount < swirlCount && (float)lastRoteThreshold < rotationVector.z)
{
lastRoteThreshold += 360;
lastRoteCount++;
SoundManager.PlayAudioWithDistance("FireBlast", transform.position);
SoundManager.PlayAudioWithDistance("StandardHeavySwing", transform.position);
}
transform.localEulerAngles = rotationVector;
trailEmitter.Emit(5);
if (emittersActive && lerpValue >= 0.9f)
{
emittersActive = false;
for (int i = 0; i < fireEmitters.Length; i++)
{
fireEmitters[i].Stop();
}
}
if (lerpValue >= 1f)
{
attackCompleted = true;
EndFlameSwirl();
}
}
public void EndFlameSwirl()
{
if (destroyImmediate)
{
Object.Destroy(gameObject);
}
else
{
if (destroyQueued)
{
return;
}
destroyQueued = true;
attackCollider.enabled = false;
if (emittersActive)
{
for (int i = 0; i < fireEmitters.Length; i++)
{
fireEmitters[i].Stop();
}
}
followTrans = null;
Object.Destroy(gameObject, 1f);
PoolManager.GetPoolItem<ParticleEffect>("SmokeEmitter").Emit(4, transform.TransformPoint(Vector3.down * 2f + Vector3.right), SmokeEmitter.SmokeMedOverride);
PoolManager.GetPoolItem<ParticleEffect>("SmokeEmitter").Emit(4, transform.TransformPoint(Vector3.down * 4f + Vector3.right), SmokeEmitter.SmokeMedOverride);
}
}
