public class BlizzardIcicle : MonoBehaviour
{
	private static GameObject _prefab;
	public Animator anim;
	private SpriteRandomizer spriteRandomizer;
	private bool playingFlashEffect;
	private float cachedAnimSpeed;
	private float flashDuration = 0.1f;
	private ChaosQuickStopwatch flashStopwatch = new ChaosQuickStopwatch();
	private Material spriteMaterial;
	public static GameObject Prefab
	{
		get
		{
			if (_prefab == null)
			{
				_prefab = ChaosBundle.Get("Assets/Prefabs/Effects/Ice/BlizzardIcicle.prefab");
			}
			return _prefab;
		}
	}
	private void Awake()
	{
		anim = GetComponent<Animator>();
		Transform transform = base.transform.Find("Sprite");
		spriteRandomizer = transform.GetComponent<SpriteRandomizer>();
		spriteMaterial = transform.GetComponent<SpriteRenderer>().material;
	}
	private void Update()
	{
		if (playingFlashEffect)
		{
			spriteMaterial.SetFloat("_FlashAmount", 1f - flashStopwatch.TimePercentage + 0.1f);
			if (!flashStopwatch.IsRunning)
			{
				playingFlashEffect = false;
				anim.speed = cachedAnimSpeed;
				cachedAnimSpeed = 0f;
			}
		}
	}
	public void Emit(Vector3 position, float distance, bool playFlashEffect = true)
	{
		gameObject.SetActive(value: true);
		transform.position = position;
		spriteRandomizer.Randomize();
		anim.speed = Random.Range(2f, 5f);
		if (playFlashEffect)
		{
			cachedAnimSpeed = anim.speed;
			anim.speed = 0f;
			playingFlashEffect = true;
			flashStopwatch.SetDelay(flashDuration, stopCurrent: false, autoStart: true);
		}
		anim.Play("Fall", -1, 1f - Mathf.Clamp01(distance * 0.09f));
	}
	public void PlayBreakEffect()
	{
		PoolManager.GetPoolItem<ParticleEffect>("IcicleFloorBreakEffect").Emit(15, transform.position);
	}
	public void DisableGameObject()
	{
		gameObject.SetActive(value: false);
	}
}

	{
		markDict = new Dictionary<int, GameObject>();
		attackBox = transform.Find("AttackBox").GetComponent<Attack>();

	}
	private void Start()
	{
		lifeTimeStopwatchID = ChaosStopwatch.Begin(lifeTime);
	}
	private void Update()
	{
		if (ChaosStopwatch.Check(lifeTimeStopwatchID))
		{
			if (gameObject != null)
			{
				UnityEngine.Object.Destroy(gameObject);
			}
		}
		else if (parentTrans != null)
		{
			transform.position = parentTrans.position;
		}
	}
	private void AddMark(Collider2D col)
	{
		currentInstID = col.GetInstanceID();
		if (!markDict.ContainsKey(currentInstID))
		{
			markDict[currentInstID] = UnityEngine.Object.Instantiate(MarkPrefab, col.transform.position, Quaternion.identity);
			markDict[currentInstID].transform.SetParent(col.transform);
		}
	}
	private void OnDestroy()
	{
		foreach (GameObject value in markDict.Values)
		{
			if (value != null)
			{
				CreateBurst(value.transform.position);
				UnityEngine.Object.Destroy(value);
			}
		}
	}
	private void CreateBurst(Vector3 position)
	{
		GameObject gameObject = UnityEngine.Object.Instantiate(BurstPrefab, position, Quaternion.identity);
		gameObject.transform.Find("Attack").GetComponent<Attack>().SetAttackInfo(attackBox.atkInfo);
		gameObject.transform.Find("Attack").GetComponent<Attack>().ChangeAttackInfoLevel(burstSkillLevel);
		if (UnityEngine.Random.value > 0.5f)
		{
			gameObject.transform.localScale = new Vector3(-1f, 1f, 1f);
		}
		gameObject.transform.localEulerAngles = new Vector3(0f, 0f, UnityEngine.Random.Range(0, 360));
	}

public class BoulderDrop : MonoBehaviour
{
	private static GameObject prefab;
	private Vector2 currentPosition;
	private Transform spriteTrans;
	private SpriteRenderer sprite;
	[HideInInspector]
	public Attack attack;
	public Sprite altSprite;
	public bool useAltSprite;
	private CircleCollider2D atkCollider;
	private ParticleSystem explosionEmitter;
	public float riseHeight = 4f;
	private float startHeight;
	public float waitTime = 0.15f;
	public bool skipFirstAttack;
	private ChaosQuickStopwatch riseStopwatch = new ChaosQuickStopwatch(0.25f);
	private ChaosQuickStopwatch waitStopwatch;
	private ChaosQuickStopwatch fallStopwatch = new ChaosQuickStopwatch(0.0625f);
	private ChaosQuickStopwatch atkStopwatch = new ChaosQuickStopwatch(0.1f);
	private bool riseComplete;
	private bool waitComplete;
	private bool fallComplete;
	private bool atkComplete;
	private Vector2 lerpedPosition = Vector2.zero;
	private float lerpValue;
	public static GameObject Prefab => prefab ?? (prefab = ChaosBundle.Get("Assets/Prefabs/AreaEffects/BoulderDrop.prefab"));
	private void Awake()
	{
		attack = transform.Find("Attack").GetComponent<Attack>();
		atkCollider = attack.transform.GetComponent<CircleCollider2D>();
		spriteTrans = transform.Find("Sprite");
		sprite = spriteTrans.GetComponent<SpriteRenderer>();
		startHeight = spriteTrans.localPosition.y;
		explosionEmitter = transform.Find("ExplosionEmitter").GetComponent<ParticleSystem>();
	}
	private void Start()
	{
		currentPosition = transform.position;
		riseStopwatch.IsRunning = true;
		waitStopwatch = new ChaosQuickStopwatch(waitTime);
		string audioID = "SummonRock";
		Vector2? soundOrigin = currentPosition;
		float standardPitchRange = SoundManager.StandardPitchRange;
		SoundManager.PlayAudioWithDistance(audioID, soundOrigin, null, 24f, -1f, standardPitchRange);
		PoolManager.GetPoolItem<FloorCrackEmitter>().Emit(1, currentPosition);
		if (useAltSprite)
		{
			sprite.sprite = altSprite;
		}
		if (!skipFirstAttack)
		{
			ActivateAttack(1);
		}
	}
	private void FixedUpdate()
	{
		if (!atkComplete && !atkStopwatch.IsRunning)
		{
			atkComplete = true;
			atkCollider.enabled = false;
		}
		if (!riseComplete)
		{
			lerpValue = riseStopwatch.TimePercentage;
			lerpValue = lerpValue * lerpValue * (3f - 2f * lerpValue);
			lerpedPosition.y = Mathf.Lerp(startHeight, riseHeight, lerpValue);
			spriteTrans.localPosition = lerpedPosition;
			if (lerpValue > 0.75f)
			{
				sprite.sortingLayerName = "UI";
			}
			if (lerpValue >= 1f)
			{
				riseComplete = true;
				waitStopwatch.IsRunning = true;
			}
		}
		else if (!waitComplete)
		{
			lerpValue = waitStopwatch.TimePercentage;
			if (lerpValue >= 1f)
			{
				waitComplete = true;
				fallStopwatch.IsRunning = true;
			}
		}
		else
		{
			if (fallComplete)
			{
				return;
			}
			lerpValue = fallStopwatch.TimePercentage;
			lerpedPosition.y = Mathf.Lerp(riseHeight, startHeight, lerpValue);
			spriteTrans.localPosition = lerpedPosition;
			if (lerpValue >= 1f)
			{
				fallComplete = true;
				ActivateAttack(2);
				spriteTrans.gameObject.SetActive(value: false);
				explosionEmitter.Play();
				DustEmitter poolItem = PoolManager.GetPoolItem<DustEmitter>();
				int particleCount = 100;
				Vector3? emitPosition = currentPosition;
				poolItem.EmitCircle(particleCount, 1f, -1f, -1f, emitPosition);
				string audioID = "GroundSlam";
				Vector2? soundOrigin = currentPosition;
				float standardPitchRange = SoundManager.StandardPitchRange;
				SoundManager.PlayAudioWithDistance(audioID, soundOrigin, null, 24f, -1f, standardPitchRange);
				PoolManager.GetPoolItem<ParticleEffect>("FloorCrater").Emit(1, currentPosition);
				CameraController.ShakeCamera(0.3f);
				if (gameObject != null)
				{
					Object.Destroy(gameObject, 1f);
				}
			}
		}
	}
	private void ActivateAttack(int givenLevel)
	{
		atkCollider.radius = ((givenLevel != 1) ? 2f : 1f);
		atkComplete = false;
		atkCollider.enabled = true;
		atkStopwatch.IsRunning = true;
		attack.ChangeAttackInfoLevel(givenLevel);
	}
}

public class BubbleProjectile : Projectile
{

	public override void Start()
	{
		base.Start();
		initialVelocity = moveVector.normalized * moveSpeed;
		initialMagnitude = initialVelocity.magnitude;
		rigidbody2D.velocity = initialVelocity;
		if (isLarge)
		{
			anim.Play("IdleLarge", -1, 0f);
			animRand.enabled = false;
			attack.transform.localScale *= 1.5f;
		}
	}
	public override void FixedUpdate()
	{
		base.FixedUpdate();
		if (!isMinVelocity && rigidbody2D.velocity.magnitude < minVelocity)
		{
			isMinVelocity = true;
			rigidbody2D.velocity = rigidbody2D.velocity.normalized * minVelocity;
			rigidbody2D.drag = 0f;
		}
	}
	public void OnCollisionEnter2D(Collision2D col)
	{
		if (!markedForDestroy)
		{
			anim.Play("Bounce" + ((!isLarge) ? string.Empty : "Large"), -1, 0f);
			transform.localEulerAngles = Globals.GetRotationVector(-Globals.GetPrimaryCardinalVec(((Vector2)transform.position - col.contacts[0].point).normalized));
			if (Projectile.globalProjectileWallCollisionEvents != null)
			{
				Projectile.globalProjectileWallCollisionEvents(this);
			}
		}
	}
	public override bool OnReflect(Vector2 givenVec, float givenMod)
	{
		if (!base.OnReflect(givenVec, givenMod))
		{
			return false;
		}
		rigidbody2D.velocity = givenVec.normalized * moveSpeed;
		return true;
	}
	public override void ResetProjectile()
	{
		if (!markedForDestroy)
		{
			PoolManager.GetPoolItem<WaterDropEmitter>().Emit((!isLarge) ? 6 : 3, transform.position, WaterDropEmitter.LargeWaterExplosionOverride);
			string audioID = "BubblePopAlt";
			Vector2? soundOrigin = transform.position;
			float standardPitchRange = SoundManager.StandardPitchRange;
			SoundManager.PlayAudioWithDistance(audioID, soundOrigin, null, 24f, -1f, standardPitchRange);
			if (isLarge)
			{
				markedForDestroy = true;
				anim.Play("PopLarge", -1, 0f);
			}
			else
			{
				PoolManager.GetPoolItem<ParticleEffect>("BubblePopEffect").Emit(1, transform.position);
				base.ResetProjectile();
			}
		}
	}
