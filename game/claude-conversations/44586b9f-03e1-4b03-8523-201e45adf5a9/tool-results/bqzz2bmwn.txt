	public class ChargeState : State<Laser>
		public new Laser parent;
		private int chargeStopwatchID = -1;
			parent = parentEntity;
		public override void OnEnter()
			if (parent.targetLineIgnoresEntities)
				parent.targetLine0.collisionLayerMask = ChaosCollisions.layerUpperWall;
				parent.targetLine1.collisionLayerMask = ChaosCollisions.layerUpperWall;
				parent.targetLine2.collisionLayerMask = ChaosCollisions.layerUpperWall;
			parent.flashRed = false;
			parent.flashStarted = false;
			if (parent.chargeTime <= 0f)
				fsm.ChangeState("Shoot");
			parent.SetTargetLineVars();
			chargeStopwatchID = ChaosStopwatch.Begin(parent.chargeTime);
		public override void Update()
			if (ChaosStopwatch.Check(chargeStopwatchID))
				fsm.ChangeState("Shoot");
			else
				if (!parent.flashDuringCharge)
				if (!parent.flashStarted)
					if (parent.flashThreshold != 0f && !(ChaosStopwatch.GetTimePercentage(chargeStopwatchID) >= parent.flashThreshold))
					parent.flashStarted = true;
				if (!parent.flashStarted)
				if (parent.flashTimer == null)
					parent.flashTimer = new ChaosQuickStopwatch(0.025f);
					parent.flashTimer.IsRunning = true;
					parent.curFlashColor.a = 1f;
				else if (!parent.flashTimer.IsRunning)
					if (parent.flashRed)
						parent.curFlashColor = parent.flashColorOG;
						parent.curFlashColor.a = 1f;
						parent.flashRed = false;
					else
						parent.flashRed = true;
						parent.curFlashColor = parent.flashColorWhite;
					parent.flashTimer.IsRunning = true;
				parent.targetLine0.line.SetColors(parent.curFlashColor, parent.curFlashColor);
				parent.targetLine1.line.SetColors(parent.curFlashColor, parent.curFlashColor);
				parent.targetLine2.line.SetColors(parent.curFlashColor, parent.curFlashColor);
		public override void OnExit()
			if (parent.targetLineIgnoresEntities)
				parent.targetLine0.collisionLayerMask = parent.initialCollisionMask;
				parent.targetLine1.collisionLayerMask = parent.initialCollisionMask;
				parent.targetLine2.collisionLayerMask = parent.initialCollisionMask;
	public class ShootState : State<Laser>
		public new Laser parent;
		private AudioSource beamAudio;
		private int atkStopwatchID;
		private bool atkEnabled;
		private bool targetCompleted;
			parent = parentEntity;
		public override void OnEnter()
			parent.SetTargetLineVars();
			CameraController.ShakeCamera(1f, ignoreShakeCount: true);
			parent.attack.SetAttackInfo(parent.skillCategory, parent.skillID, parent.skillLevel, parent.skillIsUltimate);
			InitShootEffects();
			if (parent.timeTilMaxRange != 0f)
				parent.rangeStopwatchID = ChaosStopwatch.Begin(parent.timeTilMaxRange);
			targetCompleted = false;
		public override void FixedUpdate()
			if (targetCompleted)
				if (!atkEnabled)
					atkEnabled = true;
					parent.attackCollider.gameObject.SetActive(value: true);
				if (ChaosStopwatch.Check(atkStopwatchID))
					atkStopwatchID = ChaosStopwatch.Begin(parent.hitCooldownTime);
					parent.attackCollider.gameObject.SetActive(value: false);
					atkEnabled = false;
		public override void Update()
			UpdateTargetLines();
			parent.attackCollider.offset = new Vector2(parent.targetLine0.range * 0.5f + parent.attackOriginOffset, 0f);
			parent.attackCollider.size = new Vector2(Mathf.Clamp(parent.targetLine0.range - parent.attackOriginOffset * 2f, parent.minAttackLen, 999f
			RandomizeShootThickness();
			UpdateShootEffects();
			if (beamAudio == null || !beamAudio.isPlaying)
			targetCompleted = true;
		public override void OnExit()
			if (beamAudio != null)
				beamAudio.Stop();
				beamAudio = null;
			if (parent.impactEffect != null)
				parent.impactEffect.Stop();
			if (parent.shootEffect != null)
				parent.shootEffect.Stop();
			parent.impactEffect = null;
			parent.shootEffect = null;
		public virtual void UpdateShootEffects()
			if (parent.impactEffect != null)
				parent.impactEffect.transform.position = parent.targetLine0.transform.position + (Vector3)parent.targetLine0.aimVector * parent.targetLi
			if (parent.shootEffect != null)
				parent.shootEffect.transform.position = parent.targetLine0.transform.position;
		public virtual void InitShootEffects()
		public void RandomizeShootThickness()
			parent.targetLine0.thickness = parent.shootThickness0 + UnityEngine.Random.Range(parent.randomSizeRange0.x, parent.randomSizeRange0.y);
			parent.targetLine1.thickness = parent.shootThickness1 + UnityEngine.Random.Range(parent.randomSizeRange1.x, parent.randomSizeRange1.y);
			parent.targetLine2.thickness = parent.shootThickness2 + UnityEngine.Random.Range(parent.randomSizeRange2.x, parent.randomSizeRange2.y);
		public virtual void UpdateTargetLines()
			float num = ChaosStopwatch.GetTimePercentage(parent.rangeStopwatchID);
			if (num < 1f)
				if (parent.targetLine0.collisionOccurred || parent.targetLine1.collisionOccurred || parent.targetLine2.collisionOccurred)
					num = 1f;
					ChaosStopwatch.Stop(parent.rangeStopwatchID);
				float maxRange = Mathf.Lerp(0f, parent.maxRange, num);
				parent.targetLine0.maxRange = maxRange;
				parent.targetLine1.maxRange = maxRange;
				parent.targetLine2.maxRange = maxRange;
			if (parent.targetTransform != null)
				parent.targetVector = parent.targetTransform.position;
			Vector2 vector = parent.targetLine0.transform.position;
			if (parent.lockOn)
				parent.targetLine0.aimVector = ((Vector2)parent.targetVector - vector).normalized;
			else
				float num2 = parent.turnSpeed * Time.deltaTime;
				float num3 = Globals.Vector3AngleWithPivot(vector + parent.targetLine0.aimVector, vector, parent.targetVector);
				Vector3 vector2 = Vector3.Cross(parent.targetLine0.aimVector, parent.targetVector - (Vector3)vector);
				if (num3 <= 2f && num3 >= -2f)
					num2 = 0f;
				else if (vector2.z < 0f)
					num2 = 0f - num2;
				parent.targetLine0.aimVector = Globals.PointFromRotation(Vector2.zero, parent.targetLine0.aimVector, num2);
			parent.targetLine1.aimVector = parent.targetLine0.aimVector;
			parent.targetLine2.aimVector = parent.targetLine0.aimVector;

		targetLine1.line.sortingLayerName = sortingLayerName;
		targetLine2.line.sortingLayerName = sortingLayerName;
