using System;
using UnityEngine;

public class HealOverdriveItem : Item
{
	public static string staticID = "HealOverdriveItem";

	public HealOverdriveItem()
	{
		ID = staticID;
		category = Category.Misc;
	}

	public override void Activate()
	{
		SetModStatus(givenStatus: true);
	}

	public override void Deactivate()
	{
		SetModStatus(givenStatus: false);
	}

	private void SetModStatus(bool givenStatus)
	{
		if (SetParentAsPlayer())
		{
			Health health = parentPlayer.health;
			health.healEventHandlers = (Health.HealEventHandler)Delegate.Remove(health.healEventHandlers, new Health.HealEventHandler(OnHealEvent));
			if (givenStatus)
			{
				Health health2 = parentPlayer.health;
				health2.healEventHandlers = (Health.HealEventHandler)Delegate.Combine(health2.healEventHandlers, new Health.HealEventHandler(OnHealEvent));
			}
		}
	}

	private void OnHealEvent(int givenValue, int overhealAmount = 0)
	{
		if (parentPlayer != null && !parentPlayer.overdriveReady)
		{
			parentPlayer.OverdriveProgress += Mathf.Min(35f, givenValue);
		}
	}
}

using System.Collections.Generic;
using System.Linq;
using Chaos.ArrayExtensions;
using Chaos.ListExtensions;
using UnityEngine;

public class Inventory
{
	public delegate void ItemEventHandler(Item givenItem, bool givenStatus);

	public delegate void SingleEventHandler(Item givenItem);

	public delegate bool SkipConditional(Item givenItem);

	public const int MaxItemCount = 12;

	private int count;

	private Entity parentEntity;

	public Dictionary<string, Item> itemDict;

	private Item currentItem;

	private string currentItemID = string.Empty;

	public ItemEventHandler inventoryChangeEventHandler;

	public SingleEventHandler itemAddEventHandler;

	private List<Item> tempRandList = new List<Item>(12);

	public int Count => count;

	public bool IsFull => Count >= 12;

	public int CursedCount
	{
		get
		{
			int num = 0;
			foreach (Item value in itemDict.Values)
			{
				if (value != null && value.isCursed)
				{
					num++;
				}
			}
			return num;
		}
	}

	public Inventory(Entity newParentEntity)
	{
		parentEntity = newParentEntity;
		itemDict = new Dictionary<string, Item>();
	}

	public static string GetNpcTokenID()
	{
		string text = string.Empty;
		Player[] activePlayers = GameController.activePlayers;
		foreach (Player player in activePlayers)
		{
			text = player.inventory.GetNpcTokenItemID();
			if (text != string.Empty)
			{
				break;
			}
		}
		return text;
	}

	public static bool EitherPlayerHasItem(string givenItemID)
	{
		Player[] activePlayers = GameController.activePlayers;
		foreach (Player player in activePlayers)
		{
			if (player != null && player.inventory.ContainsItem(givenItemID))
			{
				return true;
			}
		}
		return false;
	}

	public static bool BothPlayersHaveItem(string givenItemID)
	{
		if (GameController.activePlayers.Length <= 1)
		{
			return false;
		}
		Player[] activePlayers = GameController.activePlayers;
		foreach (Player player in activePlayers)
		{
			if (player != null && !player.inventory.ContainsItem(givenItemID))
			{
				return false;
			}
		}
		return true;
	}

	public static string GetExtraInfo(int givenPlayerID, string givenItemID)
	{
		Player[] activePlayers = GameController.activePlayers;
		foreach (Player player in activePlayers)
		{
			if (player != null && player.playerID == givenPlayerID && player.inventory.ContainsItem(givenItemID))
			{
				return player.inventory.GetItem(givenItemID).FormattedExtraInfo;
			}
		}
		return string.Empty;
	}

	public bool CheckForMutallyExclusiveItems(string givenItemID)
	{
		foreach (List<string> mutuallyExclusiveItem in LootManager.mutuallyExclusiveItems)
		{
			if (!mutuallyExclusiveItem.Contains(givenItemID))
			{
				continue;
			}
			for (int i = 0; i < mutuallyExclusiveItem.Count; i++)
			{
				if (ContainsItem(mutuallyExclusiveItem[i]) && itemDict[mutuallyExclusiveItem[i]].isCursed)
				{
					return false;
				}
			}
			for (int j = 0; j < mutuallyExclusiveItem.Count; j++)
			{
				DropItem(mutuallyExclusiveItem[j]);
			}
		}
		return true;
	}

	public void AnnounceItemEvent(Item givenItem, bool status)
	{
		if (inventoryChangeEventHandler != null)
		{
			inventoryChangeEventHandler(givenItem, status);
		}
		if (itemAddEventHandler != null)
		{
			itemAddEventHandler(givenItem);
		}
	}

	public bool CheckItemCombine(Item newItem)
	{
		if (newItem.parentCombineID != string.Empty)
		{
			List<string> list = itemDict.Keys.ToList();
			list.Add(newItem.ID);
			if (ItemRecipe.RecipeComplete(newItem.parentCombineID, list))
			{
				return true;
			}
		}
		if (newItem.parentGroupID != string.Empty && (itemDict.ContainsKey(newItem.parentGroupID) || (GroupItemManager.groupsDict.ContainsKey(newItem.parentGroupID) && GroupItemManager.groupsDict[newItem.parentGroupID].Any((string x) => itemDict.Keys.Any((string y) => y == x)))))
		{
			return true;
		}
		return false;
	}

	public bool AddItem(Item givenItem, bool showNotice = false, bool ignoreMax = false)
	{
		currentItemID = givenItem.ID;
		Player.newItems[givenItem.category].Remove(currentItemID);
		if (ContainsItem(currentItemID))
		{
			return false;
		}
		if (!CheckForMutallyExclusiveItems(currentItemID))
		{
			return false;
		}
		if (itemDict.Count >= 12 && !CheckItemCombine(givenItem) && !ignoreMax && !DropItem(string.Empty))
		{
			return false;
		}
		givenItem.parentEntity = parentEntity;
		givenItem.SetParentSkillCategory();
		itemDict[currentItemID] = givenItem;
		givenItem.Activate();
		if (showNotice && parentEntity is Player)
		{
			((Player)parentEntity).newItemNoticeUI.Display(TextManager.GetItemName(currentItemID), IconManager.GetItemIcon(currentItemID));
			if (givenItem.isCursed)
			{
				SoundManager.PlayAudio("BuyCursedRelic");
			}
			else
			{
				SoundManager.PlayAudio("Powerup");
			}
			if (givenItem.isCursed)
			{
				PoolManager.GetPoolItem<ParticleEffect>("CursedAuraBurstEffect").Play(null, parentEntity.transform.position, null, null, null, null, 0f, loop: false);
			}
		}
		count++;
		AnnounceItemEvent(givenItem, status: true);
		if (givenItem.isCursed)
		{
			AchievementManager.UnlockAchievement(Achievement.HoldFourCursed);
		}
		return true;
	}

	public bool AddItem(string givenItemID, bool showNotice = false, bool ignoreMax = false)
	{
		return LootManager.completeItemDict.ContainsKey(givenItemID) && AddItem(Item.CreateItem(givenItemID), showNotice, ignoreMax);
	}

	public bool RemoveItem(string givenItemID, bool forceOverride = false, bool showNotice = false)
	{
		if (!ContainsItem(givenItemID))
		{
			return false;
		}
		currentItem = itemDict[givenItemID];
		if (!forceOverride && currentItem.isCursed)
		{
			SoundManager.PlayErrorAudio();
			return false;
		}
		if (showNotice && parentEntity is Player)
		{
			((Player)parentEntity).newItemNoticeUI.Display(TextManager.GetItemName(givenItemID), IconManager.GetItemIcon(givenItemID), null, isSkill: false, isSignature: false, isEmpowered: false, acquired: false);
		}
		currentItem.Deactivate();
		currentItem.parentEntity = null;
		currentItem.parentPlayer = null;
		currentItem.parentSkillCategory = string.Empty;
		itemDict.Remove(givenItemID);
		count--;
		AnnounceItemEvent(currentItem, status: false);
		return true;
	}

	public bool DropItem(string givenItemID = "")
	{
		string loadedLevelName = Application.loadedLevelName;
		if (loadedLevelName == "PlayerRoom" || loadedLevelName == "Hub")
		{
			return false;
		}
		if (givenItemID == string.Empty)
		{
			foreach (string itemID in GetItemIDList())
			{
				currentItem = itemDict[itemID];
				if (currentItem == null || currentItem.isCursed || !RemoveItem(itemID))
				{
					continue;
				}
				if (!currentItem.destroyOnDrop)
				{
					LootManager.DropItem(parentEntity.transform.position, 1, itemID, randomLoc: false);
				}
				return true;
			}
		}
		else if (RemoveItem(givenItemID))
		{
			if (!currentItem.destroyOnDrop)
			{
				LootManager.DropItem(parentEntity.transform.position, 1, givenItemID, randomLoc: false);
			}
			return true;
		}
		return false;
	}

	public List<string> GetItemIDList()
	{
		return new List<string>(itemDict.Keys);
	}

	public bool ContainsItem(string givenItemID)
	{
		return itemDict.ContainsKey(givenItemID);
	}

	public Item GetItem(string givenItemID)
	{
		return (!ContainsItem(givenItemID)) ? null : itemDict[givenItemID];
	}

	public ResurrectItem GetResurrectItem()
	{
		ResurrectItem resurrectItem = null;
		foreach (Item value in itemDict.Values)
		{
			if (value is ResurrectItem { IsAvailable: not false } resurrectItem2 && (resurrectItem == null || resurrectItem.priority > resurrectItem2.priority))
			{
				resurrectItem = resurrectItem2;
			}
		}
		return resurrectItem;
	}

	public string GetNpcTokenItemID()
	{
		foreach (Item value in itemDict.Values)
		{
			if (value != null && value is NpcTokenItem)
			{
				return value.ID;
			}
		}
		return string.Empty;
	}

	public void ClearInventory(bool keepHubOnly = false, bool keepCursed = false, bool keepCombine = false, bool keepRecipe = false)
	{
		List<string> list = new List<string>();
		foreach (Item value in itemDict.Values)
		{
			if (value != null && (!keepHubOnly || !value.hubOnly) && (!keepCursed || !value.isCursed) && (!keepCombine || !value.combineOnly) && (!keepRecipe || !(value.parentCombineID != string.Empty)))
			{
				list.Add(value.ID);
				if (value.isGroupItem)
				{
					(value as GroupItem).HandleDrop(ignoreDrop: true);
				}
				value.Deactivate();
				value.parentEntity = null;
				value.parentPlayer = null;
				value.parentSkillCategory = string.Empty;
			}
		}
		for (int i = 0; i < list.Count; i++)
		{
			itemDict.Remove(list[i]);
		}
		count -= list.Count;
	}

	public void ClearInventory(SkipConditional GivenConditional)
	{
		List<string> list = new List<string>();
		foreach (Item value in itemDict.Values)
		{
			if (value != null && GivenConditional != null && !GivenConditional(value))
			{
				list.Add(value.ID);
				value.Deactivate();
				value.parentEntity = null;
				value.parentPlayer = null;
				value.parentSkillCategory = string.Empty;
			}
		}
		for (int i = 0; i < list.Count; i++)
		{
			itemDict.Remove(list[i]);
		}
		count -= list.Count;
	}

	public Item GetRandomItem(bool excludeCursed = true)
	{
		if (count == 0)
		{
			return null;
		}
		if (!excludeCursed)
		{
			return itemDict.Values.ToArray().GetRandomValue();
		}
		tempRandList.Clear();
		foreach (Item value in itemDict.Values)
		{
			if (!value.isCursed)
			{
				tempRandList.Add(value);
			}
		}
		return (tempRandList.Count != 0) ? tempRandList.GetRandomValue() : null;
	}
}

using System;
using UnityEngine;

public class MagicianHat : HeadgearItem
{
	public static string staticID = "MagicianHat";

	protected float cdrValue = 0.5f;

	protected float empowerChance = 0.16f;

	protected BoolVarStatMod boolMod;

	protected bool skillAltered;

	protected Vector2 effectPosition;

	protected override string ExtraInfo => PercentToStr(empowerChance, string.Empty);

	public MagicianHat()
	{
		ID = staticID;
		category = Category.Offense;
		parentCombineID = MagicianSet.staticID;
		boolMod = new BoolVarStatMod(ID, newModValue: true, 40);
		headgearPrefab = ChaosBundle.Get("Assets/Prefabs/Headgear/TopHat.prefab");
	}

	public override void Activate()
	{
		base.Activate();
		SetModStatus(givenStatus: true);
	}

	public override void Deactivate()
	{
		base.Deactivate();
		SetModStatus(givenStatus: false);
	}

	public virtual void SetModStatus(bool givenStatus)
	{
		if (!SetParentAsPlayer())
		{
			return;
		}
		if (givenStatus && ID != MagicianSet.staticID && parentPlayer.inventory.ContainsItem(MagicianWand.staticID))
		{
			parentPlayer.inventory.AddItem(MagicianSet.staticID, showNotice: false, ignoreMax: true);
			parentPlayer.inventory.RemoveItem(MagicianWand.staticID, forceOverride: true);
			parentPlayer.inventory.RemoveItem(staticID, forceOverride: true);
			return;
		}
		Player player = parentPlayer;
		player.skillEnterEventHandlers = (Player.SkillEventHandler)Delegate.Remove(player.skillEnterEventHandlers, new Player.SkillEventHandler(OnSkillEnter));
		Player player2 = parentPlayer;
		player2.skillExitEventHandlers = (Player.SkillEventHandler)Delegate.Remove(player2.skillExitEventHandlers, new Player.SkillEventHandler(OnSkillExit));
		if (givenStatus)
		{
			Player player3 = parentPlayer;
			player3.skillEnterEventHandlers = (Player.SkillEventHandler)Delegate.Combine(player3.skillEnterEventHandlers, new Player.SkillEventHandler(OnSkillEnter));
			Player player4 = parentPlayer;
			player4.skillExitEventHandlers = (Player.SkillEventHandler)Delegate.Combine(player4.skillExitEventHandlers, new Player.SkillEventHandler(OnSkillExit));
		}
	}

	private void OnSkillEnter(Player.SkillState givenSkill)
	{
		if (givenSkill != null && CheckForActivation(givenSkill))
		{
			skillAltered = true;
			givenSkill.SetEmpowered(givenStatus: true, boolMod);
			SpawnEffect();
		}
	}

	private void OnSkillExit(Player.SkillState givenSkill)
	{
		if (givenSkill != null && skillAltered)
		{
			skillAltered = false;
			givenSkill.SetEmpowered(givenStatus: false, boolMod);
			if (!givenSkill.isBasic)
			{
				parentPlayer.hud.MiscPing("HourglassUI");
				givenSkill.cooldownRef.ReduceRemainingTime(givenSkill.cooldownRef.RemainingTime * cdrValue);
			}
		}
	}

	public virtual bool CheckForActivation(Player.SkillState givenSkill)
	{
		if (givenSkill == null || givenSkill.isBasic)
		{
			return false;
		}
		if (givenSkill.isDash && givenSkill is Player.BaseDashState && !((Player.BaseDashState)givenSkill).cooldownReady)
		{
			return false;
		}
		return UnityEngine.Random.value < empowerChance;
	}

	public virtual void SpawnEffect()
	{
		effectPosition = parentPlayer.attackOriginTrans.position;
		PoolManager.GetPoolItem<ParticleEffect>("SparkleAura").Emit(7, effectPosition);
		SoundManager.PlayAudioWithDistance("MagicianSparkle", effectPosition);
	}
}

using System;
using System.Collections.Generic;
using UnityEngine;

public class MaxHealthUpOnHeal : Item
{
	public static string staticID = "MaxHealthUpOnHeal";

	private static Dictionary<int, float> idHealthDict = new Dictionary<int, float>();

	private NumVarStatMod healthMod;

	private int maxIncrease = 75;

	private int healthIncrease = 3;

	private int parentID = -1;

	private float HealthModValue
	{
		get
		{
			return (!idHealthDict.ContainsKey(parentID)) ? 0f : idHealthDict[parentID];
		}
		set
		{
			idHealthDict[parentID] = value;
			healthMod.modValue = value;
		}
	}

	protected override string ExtraInfo => "+ " + (int)HealthModValue;

	public MaxHealthUpOnHeal()
	{
		ID = staticID;
		category = Category.Defense;
		healthMod = new NumVarStatMod(ID, 0f);
		RunData.runResetEventHandlers = (RunData.RunEvent)Delegate.Remove(RunData.runResetEventHandlers, new RunData.RunEvent(OnRunReset));
		RunData.runResetEventHandlers = (RunData.RunEvent)Delegate.Combine(RunData.runResetEventHandlers, new RunData.RunEvent(OnRunReset));
	}

	private static void OnRunReset()
	{
		idHealthDict.Clear();
	}

	public override void Activate()
	{
		SetEventHandlers(givenStatus: true);
	}

	public override void Deactivate()
	{
		SetEventHandlers(givenStatus: false);
	}

	private void SetEventHandlers(bool givenStatus)
	{
		if (SetParentAsPlayer())
		{
			parentID = parentPlayer.playerID;
			ApplyHealthMod(givenStatus);
			Health health = parentPlayer.health;
			health.healEnterEventHandlers = (Health.HealEnterEventHandler)Delegate.Remove(health.healEnterEventHandlers, new Health.HealEnterEventHandler(OnHealEvent));
			if (givenStatus)
			{
				Health health2 = parentPlayer.health;
				health2.healEnterEventHandlers = (Health.HealEnterEventHandler)Delegate.Combine(health2.healEnterEventHandlers, new Health.HealEnterEventHandler(OnHealEvent));
			}
		}
	}

	private void ApplyHealthMod(bool givenStatus, int givenValue = 0)
	{
		if (givenStatus)
		{
			HealthModValue = Mathf.Min(HealthModValue + (float)givenValue, maxIncrease);
		}
		parentPlayer.health.healthStat.Modify(healthMod, givenStatus);
	}

	private void OnHealEvent(ref int givenValue)
	{
		if (parentPlayer != null && givenValue > 0)
		{
			ApplyHealthMod(givenStatus: true, healthIncrease);
		}
	}
}
