   210				{

   215							wordsExitFrame = 1f;

   222						{
   223							wordsExitFrame = 2f;

   241					}

   246				{
   247					Sound.PlaySnd("omiloop");
   248				}

   250				{
   251					if (wordsFrame > 15f)
   252					{
   253						QuakeManager.SetQuake(1f, ScrollManager.scroll);

   256					{

   259						if (arenaIdx > -1)

   263								int num2 = MapMgr.map.triggerMgr.arena[arenaIdx].bossCharIdx[i];

   265								{
   266									CharMgr.character[num2].update.death.KillMe();
   267								}
   268							}
   269						}

   271						{
   272							mainCharacter.update.death.Respawn(mainPlayer);
   273						}
   274					}
   275				}
   276			}
   277			simulationGravity = new Vector2(0f, 10f);
   278			upsideDown = false;
   279			for (int j = 0; j < CharMgr.maxChars; j++)
   280			{
   281				if (CharMgr.character[j].buffs.GetUpsideDown())
   282				{
   283					upsideDown = true;
   284					break;
   285				}
   286			}
   287			if (coopPlayerIdx > -1)
   288			{
   289				Player mainPlayer2 = PlayerMgr.GetMainPlayer();
   290				Player coopPlayer = PlayerMgr.GetCoopPlayer();
   291				if (mainPlayer2 != null && coopPlayer != null && mainPlayer2.charIdx > -1 && coopPlayer.charIdx > -1 && CharMgr.character[mainPlayer2.charIdx].dyingFrame > 1f && CharMgr.character[coopPlayer.charIdx].dyingFrame > 1f && uberDeadFrame < 1f)
   292				{
   293					CoopObliteratedMessage();

   295					Sound.PlaySnd("obliterated");
   296					RandomWisps();
   297				}
   298			}
   299			else
   300			{
   301				Character mainCharacter2 = GetMainCharacter();
   302				if (mainCharacter2.dyingFrame > 1f && uberDeadFrame < 1f)
   303				{
   304					if (mainCharacter2.playerIdx > -1)
   305					{
   306						Player p = PlayerMgr.player[mainCharacter2.playerIdx];
   307						int monsterIdx = -1;
   308						if (mainCharacter2.lastHitBy > -1)
   309						{
   310							Character character = CharMgr.character[mainCharacter2.lastHitBy];
   311							if (character.origVec.X > 0f && character.origVec.Y > 0f && MonsterCatalog.catalog[character.monsterIdx].ai != 30 && character.hp > 0f)
   312							{
   313								monsterIdx = character.monsterIdx;

   317					}
   318					uberDeadFrame = 1f;
   319					Sound.PlaySnd("obliterated");
   320					RandomWisps();
   321				}
   322				if (mainCharacter2.loc.Y > 49900f)
   323				{
   324					float num3 = (mainCharacter2.loc.Y - 49900f) / 2100f;
   325					if (num3 > 1f)
   326					{
   327						num3 = 1f;
   328					}
   329					simulationGravity.Y -= num3 * 20f;
   330				}
   331			}
   332			if (vanquishedFrame >= 1f)
   333			{
   334				vanquishedFrame += Game1.frameTime * 0.5f;
   335				if (vanquishedFrame > 5f)
   336				{
   337					vanquishedFrame = 0f;
   338				}
   339			}
   340			if (uberDeadFrame >= 1f)
   341			{
   342				uberDeadFrame += Game1.frameTime * 0.5f;
   343				if (coopPlayerIdx > -1 && uberDeadFrame > 5f)
   344				{
   345					if (uberDeadFrame > 4.5f)
   346					{
   347						Player mainPlayer3 = PlayerMgr.GetMainPlayer();
   348						Player coopPlayer2 = PlayerMgr.GetCoopPlayer();
   349						if (mainPlayer3 != null && coopPlayer2 != null && mainPlayer3.charIdx > -1 && coopPlayer2.charIdx > -1)
   350						{
   351							Character character2 = CharMgr.character[mainPlayer3.charIdx];
   352							Character character3 = CharMgr.character[coopPlayer2.charIdx];
   353							character2.update.CheckDropTorch(mainPlayer3);
   354							character3.update.CheckDropTorch(coopPlayer2);
   355							character2.update.death.Respawn(mainPlayer3);
   356							character3.update.death.Respawn(coopPlayer2);
   357							mainPlayer3.camMgr.scroll = character2.loc;
   358							coopPlayer2.camMgr.scroll = character3.loc;
   359							MapMgr.SanctuaryRefresh();
   360							character2.update.death.DraggingRevive(mainPlayer3);
   361							character3.update.death.DraggingRevive(coopPlayer2);
   362						}
   363					}
   364					uberDeadFrame = 0f;
   365				}
   366			}
   367			if (!(quitGameFrame > 0f))
   368			{
   369				return;
   370			}
   371			_ = quitGameFrame;
   372			if (demoMode)
   373			{
   374				quitGameFrame -= Game1.frameTime * 0.08f;
   375			}
   376			else
   377			{
   378				quitGameFrame -= Game1.frameTime;
   379			}
   380			if (!(quitGameFrame <= 0f))
   381			{
   382				return;
   383			}
   384			Player mainPlayer4 = PlayerMgr.GetMainPlayer();
   385			Game1.charSaveList.Init(mainPlayer4);
   386			gameState = 2;
   387			uberDeadFrame = 0f;
   388			upsideDown = false;
   389			vanquishedFrame = 0f;
   390			LogoRender.drawFrame = 0f;
   391			LogoRender.frame = 0f;

   393			if (coopPlayerIdx > -1)
   394			{
   395				coopPlayerIdx = -1;

   397			}
   398			for (int k = 0; k < PlayerMgr.player.Length; k++)
   399			{
   400				Player player = PlayerMgr.player[k];
   401				if (player != null)
   402				{
   403					player.menuMgr.curLev = 0;
   404					player.menuMgr.transMode = 2;

   412		}
