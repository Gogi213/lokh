using System.Collections.Generic;
using System.Linq;
using Chaos.ListExtensions;
using UnityEngine;

public class EnemyDecorator : Decorator
{
	public class EnemyGroup
	{
		public const int maxCenterEnemies = 4;

		private List<Enemy.EName> enemyList;

		public bool allowAdds;

		public int addsCount;

		public bool isSurvivalRoom;

		public int centerCount;

		public List<Enemy.EName> EnemyList => enemyList;

		public int GroupCount => enemyList.Count;

		public float GroupValue
		{
			get
			{
				float num = 0f;
				for (int i = 0; i < enemyList.Count; i++)
				{
					num += enemyValuesDict[enemyList[i]];
				}
				return num;
			}
		}

		public EnemyGroup(int centerEnemies = 0, bool addsAllowed = true, bool survivalRoom = false)
		{
			enemyList = new List<Enemy.EName>();
			centerCount = centerEnemies;
			allowAdds = addsAllowed;
			isSurvivalRoom = survivalRoom;
		}

		public EnemyGroup(Enemy.EName givenEnemy, int centerEnemies = 0, bool addsAllowed = true, bool survivalRoom = false)
		{
			enemyList = new List<Enemy.EName>();
			enemyList.Add(givenEnemy);
			centerCount = centerEnemies;
			allowAdds = addsAllowed;
			isSurvivalRoom = survivalRoom;
		}

		public EnemyGroup(List<Enemy.EName> givenList, int centerEnemies = 0, bool addsAllowed = true, bool survivalRoom = false)
		{
			enemyList = givenList ?? new List<Enemy.EName>();
			centerCount = centerEnemies;
			allowAdds = addsAllowed;
			isSurvivalRoom = survivalRoom;
		}

		public EnemyGroup(EnemyGroup givenGroup)
		{
			enemyList = givenGroup.EnemyList.ToList();
			centerCount = givenGroup.centerCount;
			allowAdds = givenGroup.allowAdds;
			isSurvivalRoom = givenGroup.isSurvivalRoom;
		}

		public EnemyGroup Add(Enemy.EName givenEnemy, bool isAdd = false)
		{
			enemyList.Add(givenEnemy);
			if (isAdd)
			{
				addsCount++;
			}
			return this;
		}

		public EnemyGroup Add(List<Enemy.EName> givenList, bool isAdd = false)
		{
			if (givenList == null)
			{
				return this;
			}
			enemyList.AddRange(givenList);
			if (isAdd)
			{
				addsCount += givenList.Count;
			}
			return this;
		}

		public EnemyGroup Remove(Enemy.EName givenEnemy)
		{
			enemyList.Remove(givenEnemy);
			return this;
		}

		public EnemyGroup Remove(List<Enemy.EName> givenList)
		{
			if (givenList == null)
			{
				return this;
			}
			for (int i = 0; i < givenList.Count; i++)
			{
				enemyList.Remove(givenList[i]);
			}
			return this;
		}

		public bool Contains(Enemy.EName givenEnemy)
		{
			return enemyList.Contains(givenEnemy);
		}

		public bool IsAllowed(List<Enemy.EName> givenList)
		{
			if (givenList != null)
			{
				for (int i = 0; i < enemyList.Count; i++)
				{
					if (!givenList.Contains(enemyList[i]))
					{
						return false;
					}
				}
			}
			return true;
		}
	}

	public void UpdateEnemyDictWithElements(Dictionary<EnemyDecoration, List<EnemyGroup>> givenGroupDict)
	{
		int num = 0;
		int num2 = 0;
		foreach (List<EnemyGroup> value in givenGroupDict.Values)
		{
			foreach (EnemyGroup item in value)
			{
				List<Enemy.EName> enemyList = item.EnemyList;
				num = enemyList.Count;
				num2 = 0;
				for (num2 = 0; num2 < num; num2++)
				{
					if (eeDict.ContainsKey(enemyList[num2]) && (useRandomElementalEnemies || Random.value < 0.55f))
					{
						enemyList[num2] = GetRandomFromEEDict(enemyList[num2]);
					}
				}
			}
		}
	}

	private Enemy.EName GetRandomFromEEDict(Enemy.EName key)
	{
		if (!eeDict.ContainsKey(key))
		{
			return Enemy.EName.Ghoul;
		}
		return eeDict[key][Random.Range(0, eeDict[key].Count)];
	}

		{

		}

				wRandomizer.Add(new WeightedItem<EnemyDecoration>(enemyDecoration, givenDict[enemyDecoration].Count, string.Empty));

			foreach (EnemyGroup item in givenDict[availableDecorationsList[0]])
			{

	public virtual bool DecoIsAvailable(EnemyDecoration givenDeco, TileGrid givenTileGrid)

	public virtual float GetRoomValue(TileGrid tileGrid)
	{
		float num = 0f;
		for (int i = 0; i < tileGrid.RowCount; i++)

			for (int j = 0; j < tileGrid.ColCount; j++)
			{
				if (tileGrid[i, j] != null && tileGrid[i, j].terrain == Tile.TerrainType.Floor)
				{
					num++;
				}
			}
		}

	public virtual bool PlaceEnemyGroup(EnemyDecoration givenDeco, EnemyGroup givenGroup, TileGrid givenTileGrid, Room givenRoom, bool spawnAtCenter)
	{
		IntVec2 validPosition = GetValidPosition(givenTileGrid, givenDeco, spawnAtCenter);
		if (validPosition == IntVec2.zero)
		{
			return false;
		}
		if (givenGroup.isSurvivalRoom)
		{
			Vector2 worldOrigin = givenRoom.GetWorldOrigin(givenTileGrid);
			SpawnSurvivalRoom(worldOrigin, givenTileGrid, givenGroup, givenRoom);
		}
		else
		{
			givenDeco.Place(givenTileGrid, validPosition.x, validPosition.y);
			Vector2 givenLoc = givenRoom.GetWorldOrigin(givenTileGrid) + new Vector2(validPosition.y, -validPosition.x);
			SpawnEnemies(givenLoc, givenDeco, givenGroup, givenRoom.EnemiesTrans);
		}
		return true;
	}
