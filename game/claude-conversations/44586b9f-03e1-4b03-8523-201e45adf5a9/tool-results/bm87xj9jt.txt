using UnityEngine;

public class WindSlam : MonoBehaviour
{
	private static GameObject prefab;

	private static string audioStr = "WindArrowEnd";

	public Attack attack;

	public CircleCollider2D attackCol;

	private ChaosQuickStopwatch atkStopwatch = new ChaosQuickStopwatch(0.125f);

	private bool destroyQueued;

	public static GameObject Prefab => prefab ?? (prefab = ChaosBundle.Get("Assets/Prefabs/AreaEffects/WindSlam.prefab"));

	private void Start()
	{
		atkStopwatch.IsRunning = true;
		string audioID = audioStr;
		Vector2? soundOrigin = transform.position;
		float overridePitch = Random.Range(0.7f, 0.75f);
		SoundManager.PlayAudioWithDistance(audioID, soundOrigin, null, 24f, -1f, overridePitch);
	}

	private void Update()
	{
		if (!destroyQueued && !atkStopwatch.IsRunning)
		{
			destroyQueued = true;
			attackCol.enabled = false;
			Object.Destroy(gameObject, 0.5f);
		}
	}
}

using UnityEngine;

public class WindSlashProjectile : Projectile
{
	private static GameObject _prefab;

	public bool multiHitAttack;

	public float multiHitInterval = 0.125f;

	private ChaosQuickStopwatch multiHitStopwatch;

	private bool movementStopped;

	public bool fasterOvertime;

	public float speedUpThreshold = 0.05f;

	public float speedUpPerTick = 300f;

	public bool playBlastAudio = true;

	public static GameObject Prefab
	{
		get
		{
			if (_prefab == null)
			{
				_prefab = ChaosBundle.Get("Assets/Prefabs/Projectiles/WindSlash.prefab");
			}
			return _prefab;
		}
	}

	public new void Start()
	{
		base.Start();
		if (multiHitAttack)
		{
			multiHitStopwatch = new ChaosQuickStopwatch(multiHitInterval);
			multiHitStopwatch.IsRunning = true;
		}
		if (playBlastAudio)
		{
			string audioID = "WindBlastDeep";
			Vector2? soundOrigin = transform.position;
			float standardPitchRange = SoundManager.StandardPitchRange;
			SoundManager.PlayAudioWithDistance(audioID, soundOrigin, null, 24f, 0.15f, standardPitchRange);
		}
	}

	public override void Update()
	{
		base.Update();
		if (fasterOvertime && ChaosStopwatch.GetTimePercentage(flyStopwatchID) >= speedUpThreshold)
		{
			moveSpeed += Time.deltaTime * speedUpPerTick;
		}
	}

	public new void FixedUpdate()
	{
		base.FixedUpdate();
		if (!CheckMaxLifeTime() && !ChaosStopwatch.Check(flyStopwatchID))
		{
			if (!attackBox.collider.enabled)
			{
				attackBox.collider.enabled = true;
			}
			if (multiHitAttack && !multiHitStopwatch.IsRunning)
			{
				multiHitStopwatch.IsRunning = true;
				attackBox.collider.enabled = false;
			}
		}
	}

	public override void ResetProjectile()
	{
		if (!movementStopped)
		{
			moveVector = Vector3.zero;
			rigidbody2D.velocity = Vector3.zero;
			anim.Play("Break");
			movementStopped = true;
		}
		else if (!movementStopped || anim.AnimPlayed())
		{
			base.ResetProjectile();
		}
	}
}
