   195		private void SetStageLayout()
   196		{
   197			if (availableRoomDict == null)
   198			{
   199				availableRoomDict = new Dictionary<string, GameObject>();
   200				GameObject gameObject = GameObject.Find("PvPRooms").gameObject;
   201				Room[] componentsInChildren = gameObject.GetComponentsInChildren<Room>();
   202				foreach (Room room in componentsInChildren)
   203				{
   204					availableRoomDict.Add(room.gameObject.name, room.gameObject);
   205				}
   206			}
   207			if (!availableRoomDict.ContainsKey(selectedRoomName))
   208			{
   209				selectedRoomName = "DefaultRoom";
   210			}
   211			forceEmpowerDrop = selectedRoomName.Contains("Lightning");
   212			applyAirRoomBuffs = selectedRoomName.Contains("Air");
   213			applyEarthRoomBuffs = selectedRoomName.Contains("Earth");
   214			applyFireRoomBuffs = selectedRoomName.Contains("Fire");
   215			lightStageEffects.Toggle(forceEmpowerDrop);
   216			airStageEffects.Toggle(applyAirRoomBuffs);
   217			foreach (KeyValuePair<string, GameObject> item in availableRoomDict)
   218			{
   219				if (item.Value != null)
   220				{
   221					if (item.Key == selectedRoomName)
   222					{
   223						selectedRoom = item.Value.GetComponent<Room>();
   224						item.Value.SetActive(value: true);
   225					}
   226					else
   227					{
   228						item.Value.SetActive(value: false);
   229					}
   230				}
   231			}
   232			if (selectedRoom != null)
   233			{
   234				StartCoroutine(Pathfinder.GenerateNodeMap(selectedRoom.tileGrid));
   235				switch (selectedRoom.element)
   236				{
   237				case ElementType.Neutral:
   238					stageLight.intensity = 0.6f;
   239					break;
   240				case ElementType.Fire:
   241					stageLight.intensity = 0.25f;
   242					break;
   243				case ElementType.Air:
   244					stageLight.intensity = 0.55f;
   245					break;
   246				case ElementType.Earth:
   247					stageLight.intensity = 0.55f;
   248					break;
   249				case ElementType.Lightning:
   250					stageLight.intensity = 0.35f;
   251					break;
   252				case ElementType.Water:
   253					stageLight.intensity = 0.3f;
   254					break;
   255				default:
   256					stageLight.intensity = 0.4f;
   257					break;
   258				}
   259			}
   260		}

   262		private void Update()
   263		{
   264			if (!playersReady)
   265			{
   266				TogglePlayerInvulnerable(status: true);
   267				if (BothPlayersAlive && (BothPlayersReady || matchInProgress))
   268				{
   269					ResetPlayers(fullReset: true);
   270					playersReady = true;
   271					matchInProgress = true;
   272					if (applyAirRoomBuffs)
   273					{
   274						ApplyAirBuffs();
   275					}
   276					else if (applyEarthRoomBuffs)
   277					{
   278						ApplyEarthBuffs();
   279					}
   280					else if (applyFireRoomBuffs)
   281					{
   282						ApplyFireBuffs();
   283					}
   284					StartCoroutine(StartBattle());
   285				}
   286			}
   287			else if (!matchOver)
   288			{
   289				if (!BothPlayersAlive)
   290				{
   291					StartCoroutine(MatchOver());
   292					matchOver = true;
   293					return;
   294				}
   295				HandleSkillSpawn();
   296			}
   297			Player[] activePlayers = GameController.activePlayers;
   298			foreach (Player player in activePlayers)
   299			{
   300				if (player.fsm.currentStateName.Contains("Fall"))
   301				{
   302					if (!playerFallCheck[player.playerID])
   303					{
   304						player.lastCheckpoint = respawnLocations[Random.Range(0, respawnLocations.Count)];
   305						player.fall.preFallPos = player.lastCheckpoint;
   306						playerFallCheck[player.playerID] = true;
   307					}
   308				}
   309				else if (playerFallCheck[player.playerID])
   310				{
   311					playerFallCheck[player.playerID] = false;
   312				}
   313			}
   314		}

   316		private void HandleSkillSpawn()
   317		{
   318			if (!resetSkillStopwatch.IsRunning)
   319			{
   320				resetSkillStopwatch.IsRunning = true;
   321				LootManager.ResetAvailableSkills();
   322			}
   323			if (ChaosStopwatch.Check(skillSpawnStopwatchID))
   324			{
   325				skillSpawnStopwatchID = ChaosStopwatch.Begin(skillSpawnTime + Random.Range(-1f, 1f));
   326				droppedSkillID = LootManager.GetSkillID();
   327				if (!(droppedSkillID == string.Empty))
   328				{
   329					skillSpawnLocIndex = (skillSpawnLocIndex + Random.Range(1, skillSpawnLocCount)) % skillSpawnLocCount;
   330					Vector3 vector = skillSpawnLocations[skillSpawnLocIndex];
   331					Vector3 spawnLocation = vector;
   332					int amount = 1;
   333					string skillID = droppedSkillID;
   334					bool droppedEmpowered = DroppedEmpowered;
   335					LootManager.DropSkill(spawnLocation, amount, skillID, 0f, 0f, null, useLifeTime: true, droppedEmpowered);
   336					vector.y -= 0.5f;
   337					Player.SpawnAllyPortal(vector, 0.5f);
   338				}
   339			}
   340		}

   388		private void ResetPlayers(bool fullReset = false)
   389		{
   390			Player[] activePlayers = GameController.activePlayers;
   391			foreach (Player player in activePlayers)
   392			{
   393				player.myOrigin = ((player.playerID != 0) ? p2StartPos : p1StartPos);
   394				player.lastCheckpoint = player.myOrigin;
   395				player.ResetPlayer(originPosition: true, fullyRestoreHealth: true, resetState: true, fullReset);
   396				Player.SpawnAllyPortal(player.transform.position);
   397				List<GameObject> list = new List<GameObject>();
   398				player.CleanUpPlayerSummonsList();
   399				foreach (GameObject playerSummons in player.playerSummonsList)
   400				{
   401					if (playerSummons != null && playerSummons != player.gameObject)
   402					{
   403						list.Add(playerSummons);
   404					}
   405				}
   406				foreach (GameObject item in list)
   407				{
   408					Entity component = item.GetComponent<Entity>();
   409					if (component != null && component.fsm.HasState("Dead"))
   410					{
   411						component.fsm.ChangeState("Dead");
   412					}
   413					Ward component2 = item.GetComponent<Ward>();
   414					if (component2 != null)
   415					{
   416						component2.DestroyWard();
   417					}
   418				}
   419				list.Clear();
   420				player.CleanUpPlayerSummonsList();
   421			}
   422		}

   424		private void ResetStage(bool fullReset = false)
   425		{
   426			GameUI.BroadcastNoticeMessage(" ");
   427			if (fullReset)
   428			{
   429				for (int i = 0; i < numberOfPlayers; i++)
   430				{
   431					playerWinCounts[i] = 0;
   432				}
   433				Transform[] array = winCountTransform;
   434				foreach (Transform transform in array)
   435				{
   436					Transform[] componentsInChildren = transform.GetComponentsInChildren<Transform>();
   437					foreach (Transform transform2 in componentsInChildren)
   438					{
   439						if (transform2 != transform)
   440						{
   441							Object.Destroy(transform2.gameObject);
   442						}
   443					}
   444				}
   445				buttonPrompts.gameObject.SetActive(value: false);
   446				ChaosStopwatch.Stop(skillSpawnStopwatchID);
   447				LootManager.Reset();
   448			}
   449			GameController.itemSpawner.Reset();
   450			Player[] activePlayers = GameController.activePlayers;
   451			foreach (Player player in activePlayers)
   452			{
   453				Object.DestroyImmediate(player.gameObject);
   454			}
   455			GameController.p1Joined = false;
   456			GameController.p2Joined = false;
   457			ResetPlayers(fullReset: true);
   458			playersReady = false;
   459			matchOver = false;
   460		}

   462		private void AddWinToken(int playerID, Color capeColor)
   463		{
   464			playerID = Mathf.Clamp(playerID, 0, 1);
   465			GameObject gameObject = Object.Instantiate(winCountTokenPrefab);
   466			RectTransform component = gameObject.GetComponent<RectTransform>();
   467			gameObject.transform.SetParent(winCountTransform[playerID]);
   468			float num = 0.14f * (float)(playerWinCounts[playerID] - 1);
   469			num = ((playerID != 0) ? (1f - num - 0.14f) : num);
   470			component.anchorMin = new Vector2(num, 0f);
   471			component.anchorMax = new Vector2(num + 0.14f, 1f);
   472			component.offsetMin = Vector2.zero;
   473			component.offsetMax = Vector2.zero;
   474			gameObject.transform.Find("CapeColor").GetComponent<Image>().color = capeColor * 1.4f;
   475			capeColor *= 0.4f;
   476			capeColor.a = 1f;
   477			gameObject.transform.Find("Circle").GetComponent<Image>().color = capeColor;
   478			gameObject.GetComponent<Animator>().Play("Spawn");
   479		}

   481		private IEnumerator StartBattle()
   482		{
   483			LootManager.Reset();
   484			GameUI.BroadcastNoticeMessage(" ");
   485			skillSpawnStopwatchID = ChaosStopwatch.Begin(skillSpawnTime);
   486			GameController.stageTitleUI.Announce(TextManager.GetUIText("pvp-ready"), 0f, 100, Color.white);
   487			yield return Globals.waitForOne;
   488			SoundManager.PlayBGM("Boss");
   489			if (pvpMode == PvPMode.Round)
   490			{
   491				GameController.stageTitleUI.Announce((roundCounter == maxRoundCount) ? TextManager.GetUIText("pvp-finalRound") : (TextManager.GetUIText("pvp-round") + " " + roundCounter), 0f, 100, Color.white);
   492				SoundManager.PlayAudio("Woosh");
   493				yield return Globals.waitForTwo;
   494			}
   495			GameController.stageTitleUI.AnnounceFight(TextManager.GetUIText("pvp-fight"));
   496			CameraController.ShakeCamera(0.75f);
   497			yield return Globals.waitForQuarter;
   498			SoundManager.PlayAudio("MetalBell");
   499			GameController.TogglePvpMode(pvpStatus: true);
   500			TogglePlayerInvulnerable(status: false);
   501		}

   503		private IEnumerator MatchOver()
   504		{
   505			Player winner = null;
   506			if (GameController.activePlayers.Length == 0)
   507			{
   508				yield break;
   509			}
   510			if (!GameController.AllPlayersDead())
   511			{
   512				Player[] activePlayers = GameController.activePlayers;
   513				foreach (Player player in activePlayers)
   514				{
   515					if (!player.fsm.currentStateName.Contains("Dead"))
   516					{
   517						winner = player;
   518						playerWinCounts[player.playerID]++;
   519						break;
   520					}
   521				}
   522			}
   523			if (winner != null)
   524			{
   525				roundCounter++;
   526				GameController.stageTitleUI.AnnounceVictory(TextManager.GetUIText("pvp-victory"));
   527			}
   528			else
   529			{
   530				GameController.stageTitleUI.Announce(TextManager.GetUIText("pvp-double"), 2.5f, 100, null, null, showUnderline: true, 1f, 0.5f);
   531			}
   532			TimeScaleController.StandardFreezeIntoEaseOutTimeScale();
   533			SoundManager.PlayAudio("Explosion2");
   534			SoundManager.PlayAudio("FloorVictory");
   535			yield return Globals.waitForHalf;
   536			GameController.TogglePvpMode(pvpStatus: false);
   537			yield return Globals.waitForOne;
   538			yield return Globals.waitForOne;
   539			SoundManager.PlayAudio("SwordDraw");
   540			if (winner != null)
   541			{
   542				AddWinToken(winner.playerID, winner.capeColor);
   543			}
   544			yield return Globals.waitForTwo;
   545			bool inputDetected = false;
   546			int selection = -1;
   547			if (pvpMode == PvPMode.Round)
   548			{
   549				int winnerExists = -1;
   550				for (int j = 0; j < numberOfPlayers; j++)
   551				{
   552					if (playerWinCounts[j] >= (int)((float)maxRoundCount * 0.5f + 1f))
   553					{
   554						winnerExists = j;
   555						break;
   556					}
   557				}
   558				if (winnerExists == -1)
   559				{
   560					ResetStage();
   561					yield break;
   562				}
   563				roundCounter = 1;
   564				if (winner != null)
   565				{
   566					GameController.stageTitleUI.Announce(TextManager.GetUIText((winnerExists != 0) ? "playerName-1" : "playerName-0") + " " + TextManager.GetUIText("pvp-winner"), 3f, 100, null, null, showUnderline: false);
   567					requestRestartConfirmation = true;
   568					yield return Globals.waitForTwo;
   569				}
   570			}
   571			if (requestRestartConfirmation)
   572			{
   573				requestRestartConfirmation = false;
   574				confirmImage.sprite = GameUI.GetInputSprite(winner.inputDevice.inputScheme);
   575				cancelImage.sprite = GameUI.GetInputSprite(winner.inputDevice.inputScheme, "Cancel");
   576				buttonPrompts.gameObject.SetActive(value: true);
   577				GameController.PauseAllPlayers();
   578				TogglePlayerInvulnerable(status: true);
   579				while (!inputDetected)
   580				{
   581					if (InputController.GetAnyDeviceButtonDown("Confirm") != null)
   582					{
   583						selection = 0;
   584						inputDetected = true;
   585					}
   586					else if (InputController.GetAnyDeviceButtonUp("Cancel") != null)
   587					{
   588						selection = 1;
   589						inputDetected = true;
   590					}
   591					yield return null;
   592				}
   593			}
   594			SoundManager.PlayBGM(string.Empty);
   595			matchInProgress = false;
   596			switch (selection)
   597			{
   598			case 0:
   599				ResetStage(fullReset: true);
   600				break;
   601			case 1:
   602				ResetStage(fullReset: true);
   603				GameController.LoadLevel("PvP");
   604				break;
   605			}
   606		}

   621			}
   622		}

# 13 — Кооп на одном экране и PvP-арена

## Цель среза

Два игрока на двух геймпадах проходят забег вместе: общий экран, подстройка врагов и лута под двоих, призрак и возрождение, дуэль после босса. Отдельный режим PvP-арены: раунды до трёх побед. Готово, когда два геймпада проходят этаж, один умирает, возрождается напарником и в арене всё играется без вмешательства клавиатуры.

## Поведение

### Кооп

1. **Вход.** Режимы выбираются на титуле: одиночный, кооп, PvP. Кооп и PvP требуют второго подключённого устройства, иначе сообщение и отказ. **Экрана выбора персонажа нет — есть экран присоединения:** каждый игрок нажимает «подтвердить» на своём устройстве, устройство закрепляется за слотом; облик и набор берутся из сохранённого билда и меняются в гардеробе домашней комнаты. Экран появляется сам, пока нет ни одного игрока и не идёт загрузка; доступен только в сценах: домашняя комната, обучение, город, арена и лобби PvP, концовка, эпилог, сцены боссов и тестовые; второй слот показывается только при флаге коопа; «отмена» возвращает на титул только в соло (в коопе и PvP не действует); игрок появляется у партнёра (эффект телепорта). Референсная проверка «подключено более одного устройства» считает контроллеры платформы, а клавиатура и мышь считаются контроллерами, поэтому на ПК она почти всегда проходит; реальное условие старта коопа — второй игрок должен нажать «подтвердить» на другом устройстве (29 п.6). Портал выхода из домашней комнаты и портал арены активны, только когда присоединились оба.
   *Лобби PvP* — сцена со стойками выбора комнаты: каждая стойка — портал с именем комнаты арены, активный только при флаге коопа и обоих присоединённых; имя записывается в статическое поле; неизвестное имя → нейтральная комната; арена содержит все комнаты и показывает выбранную. Флаг коопа принудительно ставится при старте PvP; индикатор золота скрыт.
2. **Общее и раздельное.** Золото и кристаллы общие на команду. Здоровье, слоты арканов, шкала ульты, реликвии и инвентарь у каждого свои. Реликвии не дублируются между игроками (исключения — реликвии «на двоих»).
3. **Камера.** Следует за средней точкой игроков с плавным догоном. Расходиться дальше поля зрения нельзя: запреты по осям при разлёте больше 26 м по горизонтали и 14 м по вертикали (23 §3). Если расстояние между ними больше 32 м, к игроку 1 ВСЕГДА переносится игрок 2 (не «отставший»), и только если игрок 2 жив и доступен; при мёртвом игроке расстояние принимается −1 и перенос не делается (23 «Телепорт»). Вертикальный предел разведения 15 м, горизонтальный 25 м (блокировка движения «от» напарника); в PvP зум ×0.42 (23). Если один мёртв, камера следует за живым.
4. **Масштабирование под двоих.**
   - Число комнат этажа: 10 → 12.
   - Бюджет врагов на комнату: коэффициент «клетки пола × k», k = 0.08 → 0.125.
   - Число «подкреплений» к ядру комнаты: случайное 0…2 → 1…4.
   - Комнаты выживания: циклов волн 2 → 3; в режиме «удвоения врагов» каждая точка даёт на одного лишнего врага.
   - Здоровье разрушаемых «пиньят»: ×1.5. Щит финального босса ×2.
   - Золото: отдельный множительный модификатор +10%. Потолок тира стихийных сундуков: `min(3, D + (соло 1 / кооп 2))`.
   - Интервал тика деградации здоровья 0.25 → 0.4 с.
   - Сердца: множитель выпадения считается от здоровья слабейшего ЖИВОГО игрока (доля здоровья > 0) и работает также в соло (там просто один игрок); формула `lerp(базовое_значение_стата, 0.5, доля)`, где базовое значение стата 1.75 (его меняют реликвии) — при доле 0 множитель 1.75, при доле 1 — 0.5.
5. **Смерть одного.** Умерший не выбывает: появляется призрак (скорость 8 м/с, управляется тем же устройством, не получает урона и не бьёт). Призрак привязан к напарнику: по горизонтали не дальше 12 м, по вертикали 8 м; выход за предел включает автодогон, возвращающий до 10 м / 6 м. Эмоции на кнопки арканов. Призрак не появляется, если все мертвы или идёт дуэль. Призрак проходит над ямами; при выпадении добычи над призраком вспыхивает эмоция «тревога»; эмоции — на 5 кнопок арканов, танец при удержании третьей кнопки (ноты 4 × 0.25 с); над призраком полоса душ.
   *Падение в яму в коопе (не PvP):* если есть живой напарник, не падающий и стоящий на клетке без ям (круг 0.25 м свободен), и он дальше 10 м (≥ 10) — возрождение В ПОЗИЦИИ напарника; иначе — ближайшая свободная клетка к позиции перед падением; при отсутствии свободной — последний чекпойнт (11 п.20). Если урон ямы убил игрока — он умирает в точке возврата. Урон ямы 25 и ожидание 1 с — в 01 п.13.
6. **Души и возрождение.** Души падают при выпадении ЛЮБОЙ добычи, если кооп включён и хотя бы один игрок мёртв (в любом состоянии смерти): запись таблицы добычи с душами выдаёт их. Количество по данным: обычные враги всех классов и их стихийные варианты — 1; большой гуль — 1; циклоп — 1; гроб — 1–2; движущаяся статуя — 2–3; сундуки — малый 1, стихийный, элитный, финальный элитный — 5; мимики — 3 (большой 5). Душа — подбираемая сущность со скоростью 9 м/с; подбирается только призраком (касание, +1, звук); непойманная исчезает с эффектом. (Прежнее «только статуи и гробы» неверно; 09 с цифрами 1 / 2–3 / 5 верно.) Призрак собирает души касанием. При 5 душах появляется подсказка; по подтверждению (срабатывает на отпускании) игрок возрождается с 25% здоровья в точке призрака, если оттуда есть прямая видимость до напарника в пределах 15 м, иначе рядом с напарником; ульта и перезарядки сброшены. Если напарник тоже мёртв и нет реликвии возрождения — призрак исчезает, забег заканчивается (11-run-loop, п. 12).
7. **Смена этажа.** Мёртвый на загрузке следующего этажа встаёт с 15% здоровья (в одиночном забеге — с полным).
8. **Итоги этажа в коопе.** Для каждого игрока: очки = урон обычными атаками + урон арканами + 25 × побеждённые враги + бонус здоровья + бонус золота (+500 за победу в дуэли после босса), не ниже 0. Бонус здоровья `ceil(500 × (1 − полученный_урон / макс_здоровье))` со знаком; бонус золота `ceil(500 × (1 − потрачено_игроком / потрачено_всего))`, при нулевых тратах всего знаменатель 1. Больше очков — «корона» и знак победы; равные — корона обоим (корона появляется через 0.5 с). Очки считаются за ЭТАЖ (счётчики игрока сбрасываются на каждой загрузке); знак победы выдаётся на КАЖДОМ показе итога (этажи, боссы, итог перед концовкой) игроку с большим счётом, обоим при равенстве. Знаки решают, чьи билд и имя попадут в запись забега и в эпилог (12 п.12): максимум знаков, при равенстве — монета (в референсе несидируемая; у нас поток `loot`). «Потрачено» = золото игрока, потраченное в этаже; бонус золота в соло всегда `ceil(500 × (1 − 1)) = 0`. Референс считает бонус здоровья только как штраф (никогда не плюс) — это ошибка округления/знака, мы делаем со знаком.
9. **Дуэль после босса.** Условия: не гонка боссов, кооп, дуэль ещё не начиналась или идёт; реликвия дружбы у любого игрока — дуэли нет; живых должно быть больше одного (иначе победителя нет). Таймлайн:
   - плашка «ждите» 2 с (красная, кегль 60); ждать окончания идущего хитстопа; пауза 3 с; плашка «готовы» 2 с;
   - камера: слежение выключено, ставится в среднюю точку стартовых позиций + 1 м вверх; каждый игрок: неуязвим, полное здоровье, призрак выключен, ульта 0, эффекты ульты выключены, перезарядки сброшены, позиция в своей точке, взгляд вправо/влево по чётности, эффект появления; игроки на паузе; пауза 3 с;
   - отсчёт **3-2-1**: «3» (1 с), «2» (0.4 с → анимация удара рукой; ещё 0.6 с), «1» (анимация обратного удара; 1 с), затем «бой» (тряска 0.75, колокол); затем 0.25 с, и неуязвимость снимается, дружественный огонь включён (прежнее «3, 2» неверно).
   Победитель = последний живой (хитстоп). Конец: плашка победы 1.5 с; мёртвый возрождён в своей позиции, здоровье всех = значение ДО дуэли, призрак включён, дружественный огонь выключен; затем сундук босса и портал (11 п.17). В финальной комнате дополнительно ждёт 0.5 с и затухание. Один живой или ничья — дуэли нет.
10. **Реликвия дружбы.** Бонус к урону (только к нестатусным записям) и скорости движения (отдельный множитель): три состояния по наличию у игроков — у обоих владельцев +15%, у одного в коопе +12.5%, в соло +5%. Реликвия открыта всегда (не покупается); её наличие у любого игрока отключает дуэль (п.9).

### PvP-арена

11. **Матч.** Выбирается комната арены (нейтральная или одна из стихийных). Игроки встают на «площадки готовности»; как только обе заняты и оба живы — всё сбрасывается: здоровье полное, позиции в углах, перезарядки и ульта нулевые, призыватели игроков уничтожены. Неуязвимость до команды «бой». Матч после первого запуска перезапускается автоматически без площадок готовности; при сбросе также уничтожаются призывы и стражи игроков; после каждого раунда игроки пересоздаются (флаги присоединения сбрасываются).
12. **Раунд.** Последовательность: «готовы» 1 с → «раунд N» (или «финальный раунд») 2 с → удар камеры, 0.25 с, звук → бой. Раунд длится, пока живы оба. Побеждает последний живой, счётчик побед +1. Оба погибли — «ничья», раунд не засчитывается и номер не растёт.
13. **Конец раунда.** Заморозка времени, 0.5 с, 1 с, 1 с, знак победы появляется у победителя, 2 с. Матч — «до трёх побед» из пяти (`floor(5 × 0.5 + 1) = 3`). Пока никто не набрал — новый раунд автоматически; иначе победитель объявляется 3 с, потом 2 с, затем подсказки: «подтвердить» (по нажатию) = реванш с полным сбросом (счёт 0, добыча перезапущена), «отмена» (по отпусканию) = полный сброс и выход в лобби (сцена PvP). Знак победы — шкала по 0.14 ширины панели на знак, у второго игрока справа налево.
14. **Подборы.** Первый подбор через 5 с после старта боя (без случайной добавки), далее каждые 5 ± 1 с; индекс точки = (прошлый + случайное 1…N−1) mod N (стартовый индекс случайный), то есть точка никогда не совпадает с прошлой; подбор появляется с порталом (на 0.5 м ниже) и звуком и живёт ограниченное время. Пул арканов пересобирается при старте боя и затем каждые 60 с; каждый выпавший удаляется из пула; если пул пуст, подбор пропускается. «Усиленный» 25%, в грозовой комнате 100%.
15. **Бонусы комнат.** Воздух: перезарядки −50%, скорость +40%. Земля: оглушение от ударов −90%. Огонь: получаемый урон ×2. Молния: все выпавшие арканы усиленные. Освещённость комнаты зависит от стихии: интенсивность света арены — нейтраль 0.6; огонь 0.25; воздух 0.55; земля 0.55; молния 0.35; вода 0.3; остальное 0.4. Комнаты без бонуса — нейтральная и водяная. Для воздуха и молнии включаются свои эффекты сцены.
16. **Падение в яму.** Игрок возрождается в случайной точке из списка возрождения, без смерти.
17. **Враги в арене.** Фабрики врагов включаются при начале боя; раз в 5 с каждая, если все её враги побеждены, перезапускается. Сложность арены (шаг этапа) растёт на 1 сразу при первом включении, затем каждые 20 с; сбрасывается в 0 при конце боя или отключении; проверка фабрик — раз в 5 с (таймер перезапускается каждую проверку).
18. **Заряд ульты в PvP.** Коэффициент заряда от урона по игроку хранится в данных. В референсе это статическое число без присвоения, то есть **0**: удары по игроку в PvP и дуэли НЕ заряжают ульту (удары по врагам: 0.25 × урон × множитель; по разрушаемым: 3 × множитель). Наш выбор по умолчанию — 1 (ульта заряжается), как явное отклонение.

## Критерии приёмки
