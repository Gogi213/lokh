52				if (!chargeCompleted)

65				{

67				}
68				releaseVarsSet = true;

73			base.Update();

75		public override void FixedUpdate()
76		{
77			base.FixedUpdate();
78			if (!targetHit)
79			{
80				return;
81			}
82			switch (ChaosStopwatch.CheckInterval(hitStopwatchID))
83			{
84			case StopwatchState.Ready:
85				currentCharge--;
86				attackBox.gameObject.SetActive(value: true);
87				if (hitWall)
88				{

94					}
95				}
96				SoundManager.PlayWithDistAndSPR("PopSnap", transform.position);
97				break;
98			case StopwatchState.Running:
99				attackBox.gameObject.SetActive(value: false);
100				break;
101			case StopwatchState.Done:
102				ResetProjectile();
103				break;
104			}
105		}

152			if (hitWall)
153			{
154				currentPosition -= releaseVector * 1.5f;
155			}

158			{
159				CreateCaltrop(currentPosition + Globals.GetRotatedCircleVector(12, i, releaseVector) * 1.5f);
160			}
161			for (int j = 0; j < 6; j++)
162			{
163				CreateCaltrop(currentPosition + Globals.GetRotatedCircleVector(6, j, releaseVector) * 2.75f);
164			}
165		}

167		{
168			if (Globals.CheckCircle(givenPosition, 0.125f, ChaosCollisions.layerAllWallAndObst))
169			{
170				currentTC = Globals.ChaosInst<ThrownCaltrop>(ThrownCaltrop.Prefab, null, currentPosition);
171				currentTC.attack.SetAttackInfo(caltropInfo);
172				currentTC.targetPosition = givenPosition;
173				currentTC.lifeTime = 3.5f;
174				currentTC.layerMask = ChaosCollisions.layerAllWallAndObst;
175				currentTC.useLineCastPosition = true;
176			}
177		}

185		public override void OnProjectileCollision(Collider2D other, Projectile otherProjectile, bool doCalculations = true)
186		{
187			if (doCalculations)
188			{
189				int num = calculatedDamage - otherProjectile.calculatedDamage;
190				otherProjectile.calculatedDamage -= calculatedDamage;
191				calculatedDamage = num;
192				otherProjectile.OnProjectileCollision(other, this, doCalculations: false);
193			}
194			if (calculatedDamage > 10)
195			{
196				currentCharge = Mathf.CeilToInt(calculatedDamage / baseDamage);
197				otherProjectile.calculatedDamage = 0;
198			}
199			else
200			{
201				currentCharge = 0;
202				calculatedDamage = 0;
203				HandleCollision(other);
204			}
205			collidedProjectiles.Add(otherProjectile);
206		}
207		public override void HandleWallCollision(Collider2D hitCol)
208		{
209			if (!hitWall)
210			{
211				hitWall = true;
212				updateVelocity = false;
213				moveVector = Vector2.zero;
214				rigidbody2D.velocity = Vector2.zero;
215				attackBox.knockbackMultiplierOverride = -1f;
216				attackBox.atkInfo.knockbackMultiplier = -5f;
217				attackBox.atkInfo.knockbackOverwrite = false;
218				OnTargetHit(null);
219				if (Projectile.globalProjectileWallCollisionEvents != null)
220				{
221					Projectile.globalProjectileWallCollisionEvents(this);
222				}
223				SoundManager.PlayWithDistAndSPR("PopSnap", transform.position);
224			}
225		}
