using System.Collections.Generic;
using Chaos.AnimatorExtensions;
using UnityEngine;
public class Arcana : MonoBehaviour, ICard
{
	[HideInInspector]

	[HideInInspector]
	public AnimEffect animEffect;
	public bool autoPlaySpawn;
	public bool useAnimEffect = true;
	public bool spawnPlayed = true;
	public bool flipPlayed = true;
	public Entity entityScript;
	public static GameObject _prefab;
	public static Color cardOutlineColor = new Color(0.41373f, 0.35882f, 0.333137f, 1f);
	public static Color cardOutlineColor2 = new Color(0.41373f, 0.35882f, 0.333137f, 0.15f);

	private Vector3 objPos;
	[HideInInspector]
	public Light light;
	[HideInInspector]
	public ParticleSystem pSys;
	[HideInInspector]
	public SpriteRenderer bgRenderer;
	[HideInInspector]
	public SpriteRenderer iconRenderer;
	[HideInInspector]
	public SpriteRenderer shadowRenderer;
	private static Dictionary<string, float> _enemyOffsets;
	public static GameObject Prefab
	{
		get
		{
			if (_prefab == null)
			{
				_prefab = ChaosBundle.Get("Assets/Prefabs/Items/Arcana.prefab");
			}
			return _prefab;
		}
	}
	private static Dictionary<string, float> EnemyOffsets
	{
		get
		{
			if (_enemyOffsets == null)
			{
				InitEnemyOffsets();
			}
			return _enemyOffsets;
		}
	}
	private void Awake()
	{
		anim = GetComponent<Animator>();
		animEffect = GetComponent<AnimEffect>();
		bgRenderer = transform.Find("BGSprite").GetComponent<SpriteRenderer>();
		iconRenderer = transform.Find("IconSprite").GetComponent<SpriteRenderer>();
		shadowRenderer = transform.Find("ShadowSprite").GetComponent<SpriteRenderer>();
		pSys = transform.Find("BGSprite/AuraEmitter").GetComponent<ParticleSystem>();
		light = transform.Find("Light").GetComponent<Light>();
	}
	private void Start()
	{
		if (autoPlaySpawn)
		{
			Spawn();
		}
	}
	public void Spawn()
	{
		if (useAnimEffect)
		{
			animEffect.ignoreAnimCheck = true;
			animEffect.PlayAnim("Spawn");
		}
		else
		{
			anim.Play("Spawn", -1, 0f);
		}
	}
	private void Update()
	{
		if (!spawnPlayed && anim.AnimPlayed())
		{
			spawnPlayed = true;
			if (useAnimEffect)
			{
				animEffect.PlayAnim("SpinToFrontEntity");
			}
			else
			{
				anim.Play("Spawn", -1, 0f);
			}
		}
		else if (!flipPlayed && anim.AnimPlayed(1.25f))
		{
			flipPlayed = true;
			if (useAnimEffect)
			{
				animEffect.ignoreAnimCheck = false;
				animEffect.PlayAnim("SummonEntity");
			}
			else
			{
				anim.Play("Spawn", -1, 0f);
			}
			SoundManager.PlayAudioWithDistance("Teleport", transform.position);
			CameraController.ShakeCamera();
			if (objRef != null)
			{
				objRef.SetActive(value: true);
			}
			pSys.Emit(Random.Range(4, 8));
		}
	}
	private static void InitEnemyOffsets()
	{
		_enemyOffsets = new Dictionary<string, float>();
		_enemyOffsets["Lancer"] = 0.125f;
	}
	public void InitCardSettings(Entity entityScript = null, GameObject objRef = null, SpriteRenderer objSpriteRenderer = null, ElementType element = ElementType.Neutral, Vector3? objPos = null, int paletteIndexOverride = -1)
	{
		if (objPos.HasValue)
		{
			transform.position = objPos.Value;
		}
		this.entityScript = entityScript;
		this.objRef = objRef;
		InitSpriteRenderer(objSpriteRenderer);
		spawnPlayed = false;
		flipPlayed = false;
	}
	public void InitSpriteRenderer(SpriteRenderer parentRenderer, Vector3? offset = null)
	{
		if (!(parentRenderer == null) && !(parentRenderer.material == null))
		{
			if (!parentRenderer.material.HasProperty("_Palette"))
			{
				iconRenderer.material = Globals.SpriteDiffuseMasked;
			}
			else
			{
				iconRenderer.material = Globals.PaletteSwapMasked;
				iconRenderer.material.SetTexture("_Palette", parentRenderer.material.GetTexture("_Palette"));
				iconRenderer.material.SetFloat("_PaletteCount", parentRenderer.material.GetFloat("_PaletteCount"));
				iconRenderer.material.SetFloat("_PaletteIndex", parentRenderer.material.GetFloat("_PaletteIndex"));
			}
			iconRenderer.sprite = parentRenderer.sprite;
			iconRenderer.transform.localScale = parentRenderer.transform.parent.transform.localScale;
			if (iconRenderer.material.HasProperty("_OutlineColor"))
			{
				iconRenderer.material.SetColor("_OutlineColor", cardOutlineColor);
			}
			if (iconRenderer.material.HasProperty("_OutlineColor2"))
			{
				iconRenderer.material.SetColor("_OutlineColor2", cardOutlineColor2);
			}
			iconRenderer.color = parentRenderer.color;
			float num = iconRenderer.sprite.pivot.y / iconRenderer.sprite.pixelsPerUnit;
			float num2 = iconRenderer.sprite.rect.size.y / iconRenderer.sprite.pixelsPerUnit;
			Vector3 localPosition = new Vector3(0f, 0.25f + num + (3.0625f - num2) / 2f, 0f);
			if (offset.HasValue)
			{
				localPosition = offset.Value;
			}
			iconRenderer.transform.localPosition = localPosition;
		}
	}
}

using Chaos.AnimatorExtensions;
using UnityEngine;
using UnityEngine.UI;
public class CastingCircle : MonoBehaviour
{
	private static Vector3 defaultUnitScale = new Vector3(4f, 4f, 1f);
	public static Color defaultColor = new Color(0.6f, 0.6f, 0.6f);
	public static Color enemyColor = new Color(1f, 0f, 0f, 0.9f);
	public static Color enemyColorDim = new Color(1f, 0f, 0f, 0.75f);
	private Animator anim;
	private Image[] castingCircleImages;
	private GameObject parentObject;
	private Transform parentTransform;
	public bool disableWithParent = true;
	public bool useCastTime;
	private int castStopwatchID = -1;
	public bool trackParent;
	public float drawTime = 1f;
	private bool disableOnNextFrame;
	public bool autoSpin;
	public Color autoSpinColor = Color.white;
	public float autoSpinSpeed = 1f;
	private void Awake()
	{
		anim = GetComponent<Animator>();
		castingCircleImages = transform.Find("CastingCircleCanvas").GetComponentsInChildren<Image>();
		if (!autoSpin)
		{
			Reset();
		}
	}
	public void Reset(bool disableImmediately = false)
	{
		parentObject = null;
		parentTransform = null;
		disableWithParent = true;
		useCastTime = false;
		ChaosStopwatch.Stop(castStopwatchID);
		trackParent = false;
		drawTime = 1f;
		anim.speed = 1f;
		anim.Play("Idle");
		disableOnNextFrame = !disableImmediately;
		if (disableImmediately)
		{
			SetColor(defaultColor);
			gameObject.SetActive(value: false);
		}
	}
	private void Start()
	{
		if (autoSpin)
		{
			string animName = "Spin";
			Vector3 position = transform.position;
			bool disableWithParentVal = false;
			Color? color = autoSpinColor;
			Activate(animName, position, disableWithParentVal, null, trackParentVal: true, useCastTimeVal: false, 0f, 0f, 0f, color);
			anim.speed = autoSpinSpeed;
		}
	}
	public void SetColor(Color? newColor = null)
	{
		if (newColor.HasValue)
		{
			Color value = newColor.Value;
			for (int i = 0; i < castingCircleImages.Length; i++)
			{
				value.a = castingCircleImages[i].color.a;
				castingCircleImages[i].color = value;
			}
		}
	}
	public void Activate(string animName, Vector3 spawnPos, bool disableWithParentVal = true, GameObject parentObj = null, bool trackParentVal = true, bool useCastTimeVal = false, float castTimeVal = 0f, float unitSizeX = 0f, float unitSizeY =
	{
		disableWithParent = disableWithParentVal;
		trackParent = trackParentVal;
		SetParent(parentObj);
		useCastTime = useCastTimeVal;
		castStopwatchID = ((!useCastTime) ? (-1) : ChaosStopwatch.Begin(castTimeVal));
		drawTime = drawTimeVal;
		if (unitSizeX > 0f && unitSizeY > 0f)
		{
			SetUnitSize(unitSizeX, unitSizeY);
		}
		else if (unitSizeX > 0f)
		{
			SetUnitSize(unitSizeX);
		}
		else if (unitSizeY > 0f)
		{
			SetUnitSize(unitSizeY);
		}
		SetColor(color);
		ActivateWithoutParams(animName, spawnPos);
	}
	public void ActivateWithoutParams(string animName, Vector3 spawnPos)
	{
		transform.position = spawnPos;
		transform.localEulerAngles = new Vector3(0f, 0f, Random.Range(0f, 360f));
		gameObject.SetActive(value: true);
		anim.Play(animName, -1, 0f);
		if (animName.Contains("Draw") && drawTime > 0f)
		{
			anim.speed = 1f / drawTime;
		}
	}
	public void ResetAnimSpeed()
	{
		anim.speed = 1f;
	}
	private void Update()
	{
		if (autoSpin)
		{
			return;
		}
		if (disableOnNextFrame && anim.AnimPlayed())
		{
			disableOnNextFrame = false;
			gameObject.SetActive(value: false);
			SetColor(defaultColor);
			return;
		}
		if (disableWithParent && (parentObject == null || !parentObject.activeSelf))
		{
			Reset();
		}
		else if (useCastTime && ChaosStopwatch.Check(castStopwatchID))
		{
			Reset();
		}
		if (trackParent && parentTransform != null)
		{
			transform.position = parentTransform.position;
		}
	}
	public void SetParent(GameObject newParentObject)
	{
		parentObject = newParentObject;
		parentTransform = ((!(parentObject == null)) ? parentObject.transform : null);
	}
	public void SetUnitSize(float x, float y)
	{
		transform.localScale = new Vector3(x / defaultUnitScale.x, y / defaultUnitScale.y, 1f);
	}
	public void SetUnitSize(float x)
	{
		SetUnitSize(x, x);
	}
	public void SetScale(float x, float y)
	{
		transform.localScale = new Vector3(x, y, 1f);
	}
	public void SetScale(float x)
	{
		SetScale(x, x);
	}

		}
		else if (unitSizeX != 0f && unitSizeY == 0f)
		{
			unitSizeY = unitSizeX;
		}
		else if (unitSizeY != 0f && unitSizeX == 0f)
		{
			unitSizeX = unitSizeY;
		}

		CastingCircle castingCircle = GetCastingCircle();
		castingCircle.Activate(animName, spawnPos);
		return castingCircle;
	}
	public static CastingCircle GetCastingCircle()
	{
		castingCircleIndex++;
		if (castingCircleIndex == castingCircleCount)
		{
			castingCircleIndex = 0;
		}
		return castingCircles[castingCircleIndex];
	}
	private void Initialize()
	{
		castingCirclePrefab = ChaosBundle.Get("Assets/Prefabs/Effects/CastingCircle.prefab");
		castingCircles = new List<CastingCircle>();
		castingCircleIndex = 0;
		castingCircleCount = 25;
		for (int i = 0; i < castingCircleCount; i++)
		{
			GameObject gameObject = Object.Instantiate(castingCirclePrefab);
			gameObject.transform.SetParent(transform);
			castingCircles.Add(gameObject.GetComponent<CastingCircle>());
		}
	}

=== Cooldown
using UnityEngine;
public class Cooldown
{
	public string ID;
	public int remainingTimeStopwatchID = -1;
	private bool isCharging;
	public bool queueCDPing;
	public bool isChargeSkill;
	public NumVarStat maxChargeStat;
	public int chargeCount;
	public float chargeDelayTime;
	public int delayStopwatchID;
	private float maxTime = 1f;
	public StatData statData;
	public Player.SkillState skillState;
	public float RemainingTime => ChaosStopwatch.GetRemainingTime(remainingTimeStopwatchID);
	public float TimePercentage => ChaosStopwatch.GetTimePercentage(remainingTimeStopwatchID);
	public bool Ready => (!isChargeSkill) ? (!IsCharging) : (!OnChargeDelay && chargeCount > 0);
	public bool IsCharging
	{
		get
		{
			return isCharging && (isCharging = !ChaosStopwatch.Check(remainingTimeStopwatchID));
		}
		set
		{
			isCharging = value;
		}
	}
	public int MaxChargeCount
	{
		get
		{
			return Mathf.CeilToInt(maxChargeStat.ModifiedValue);
		}
		set
		{
			maxChargeStat = new NumVarStat(value, syncCurrentToModMax: true);
		}
	}
	public int ChargesMissing => MaxChargeCount - chargeCount;
	public bool OnChargeDelay => isChargeSkill && !ChaosStopwatch.Check(delayStopwatchID);
	public float RemainingDelayTime => ChaosStopwatch.GetRemainingTime(delayStopwatchID);
	public float TotalChargePercentage => (TimePercentage + (float)chargeCount) / (float)MaxChargeCount;
	public float MaxTime
	{
		get
		{
			if (statData != null)
			{
				if (skillState == null)
				{
					return statData.GetValue<float>("cooldown");
				}
				return statData.GetValue<float>("cooldown", skillState.currentLevel);
			}
			return maxTime;
		}
		set
		{
			if (statData == null)
			{
				maxTime = value;
			}
		}
	}
	public Cooldown()
	{
		ID = string.Empty;
		remainingTimeStopwatchID = -1;
		delayStopwatchID = -1;
		chargeDelayTime = 0f;
		MaxTime = 1f;
	}
	public Cooldown(string givenID, float givenMaxTime, StatData givenData = null, Player.SkillState givenSkill = null)
	{
		ID = givenID;
		remainingTimeStopwatchID = -1;
		delayStopwatchID = -1;
		chargeDelayTime = 0f;
		MaxTime = givenMaxTime;
		statData = givenData;
		skillState = givenSkill;
	}
	public void Reset()
	{
		ChaosStopwatch.Stop(remainingTimeStopwatchID);
		remainingTimeStopwatchID = -1;
		ChaosStopwatch.Stop(delayStopwatchID);
		delayStopwatchID = -1;
		if (isChargeSkill)
		{
			chargeCount = MaxChargeCount;
		}
	}
	public void ReduceRemainingTime(float givenValue)
	{
		ChaosStopwatch.ReduceTimeRemaining(remainingTimeStopwatchID, givenValue);
	}
	public void ReduceTimePercentage(float givenPercent)
	{
		ChaosStopwatch.ReduceTimeRemaining(remainingTimeStopwatchID, MaxTime * givenPercent);
	}
}
=== CooldownEntry
using UnityEngine;
using UnityEngine.UI;
public class CooldownEntry
{
	private enum CooldownEntryState
	{
		Disabled,
		Ready,
		NotReady,
		Cooldown
	}
	private GameObject skillBG;
	public Image skillBorder;
	private Image skillButtonImage;
	private Image skillIcon;
	private GameObject skillBGInactive;
	private Image skillBGOverlay;
	private Player.SkillState skillState;
	private Text skillCooldownText;
	private CooldownUI cooldownUI;
	private float remainingChargeCD;
	private float remainingCD;
	private CooldownEntryState currentState;
	public CooldownEntry(Transform parentTrans, CooldownUI newCooldownUI)
	{
		cooldownUI = newCooldownUI;
		skillBG = parentTrans.Find("SkillBG").gameObject;
		skillBorder = skillBG.GetComponent<Image>();
		skillButtonImage = parentTrans.Find("SkillButton").GetComponent<Image>();
		skillIcon = parentTrans.Find("SkillIcon").GetComponent<Image>();
		skillBGInactive = parentTrans.Find("SkillBGInactive").gameObject;
		skillBGOverlay = parentTrans.Find("SkillBGOverlay").GetComponent<Image>();
		skillCooldownText = parentTrans.Find("SkillCooldown").GetComponent<Text>();
		Disable();
	}
	public void EntryUpdate()
	{
		if (currentState == CooldownEntryState.Disabled || skillState == null)
		{
			return;
		}
		if (skillState.cooldownRef.IsCharging)
		{
			if (skillState.isChargeSkill)
			{
				SetChargeCooldown();
			}
			else
			{
				SetCooldown();
			}
		}
		else if (skillState.IsReady())
		{
			SetSkillBGSprite(skillState.isSignature, skillState.IsEmpowered);
			SetReady();
		}
		else
		{
			SetNotReady();
		}
	}
	public void SetChargeCooldown()
	{
		currentState = CooldownEntryState.Cooldown;
		skillBorder.fillAmount = skillState.cooldownRef.TotalChargePercentage;
		skillBG.SetActive(value: true);
		skillBGInactive.SetActive(value: true);
		if (skillState.cooldownRef.Ready)
		{
			if (skillState.IsReady() && skillState.cooldownRef.queueCDPing)
			{
				skillState.cooldownRef.queueCDPing = false;
				cooldownUI.RequestCDPing(skillState.skillID);
			}
			if (skillState.IsReady())
			{
				skillBGOverlay.enabled = false;
				skillIcon.color = Vector4.one;
				skillButtonImage.color = CooldownUI.fullColor;
			}
			else
			{
				skillBGOverlay.enabled = true;
				skillIcon.color = CooldownUI.transparent;
				skillButtonImage.color = CooldownUI.darkened;
			}
			skillCooldownText.text = skillState.cooldownRef.chargeCount.ToString();
		}
		else
		{
			skillBGOverlay.enabled = true;
			skillIcon.color = CooldownUI.transparent;
			skillButtonImage.color = CooldownUI.darkened;
			if (skillState.cooldownRef.OnChargeDelay)
			{
				remainingChargeCD = skillState.cooldownRef.RemainingDelayTime;
				skillBGOverlay.fillAmount = remainingChargeCD / skillState.cooldownRef.chargeDelayTime;
			}
			else
			{
				remainingChargeCD = skillState.cooldownRef.RemainingTime;
				skillBGOverlay.fillAmount = remainingChargeCD / skillState.cooldownRef.MaxTime;
			}
			skillCooldownText.text = remainingChargeCD.ToString((!(remainingChargeCD < 10f)) ? "0" : "0.0");
		}
	}
	public void SetNotReady()
	{
		if (currentState != CooldownEntryState.NotReady)
		{
			skillBG.SetActive(value: false);
			skillBGInactive.SetActive(value: true);
			skillIcon.color = CooldownUI.transparent;
			skillBGOverlay.enabled = true;
			skillBGOverlay.fillAmount = 1f;
			skillCooldownText.text = string.Empty;
			skillButtonImage.color = CooldownUI.darkened;
			if (currentState == CooldownEntryState.Cooldown && skillState != null)
			{
				cooldownUI.RequestCDPing(skillState.skillID);
			}
			currentState = CooldownEntryState.NotReady;
		}
	}
	public void SetSkillBGSprite(bool isSignature, bool isEmpowered)
	{
		if (isSignature)
		{
			skillBorder.sprite = cooldownUI.skillBGSprites[4];
		}
		else if (isEmpowered)
		{
			skillBorder.sprite = cooldownUI.skillBGSprites[2];
		}
		else
		{
			skillBorder.sprite = cooldownUI.skillBGSprites[0];
		}
	}
	public void SetReady()
	{
		if (currentState != CooldownEntryState.Ready)
		{
			skillBG.SetActive(value: true);
			skillBGInactive.SetActive(value: false);
			skillIcon.color = Vector4.one;
			skillBGOverlay.enabled = false;
			skillBGOverlay.fillAmount = 0.01f;
			skillButtonImage.color = CooldownUI.fullColor;
			if (skillState.isChargeSkill)
			{
				skillBorder.fillAmount = skillState.cooldownRef.TotalChargePercentage;
				skillCooldownText.text = skillState.cooldownRef.MaxChargeCount.ToString();
			}
			else
			{
				skillCooldownText.text = string.Empty;
				skillBorder.fillAmount = 1f;
			}
			if (currentState == CooldownEntryState.Cooldown)
			{

using System.Collections;
using System.Collections.Generic;
public class CooldownManager
{
	public delegate void StartCDEventHandler(string cooldownID);
	public delegate void StartChargeEventHandler(string cooldownID);
	public Dictionary<string, Cooldown> cooldowns;
	public StartCDEventHandler startCDEventHandlers;
	public StartChargeEventHandler startChargeEventHandlers;
	public CooldownManager()
	{
		cooldowns = new Dictionary<string, Cooldown>();
	}
	public void ResetCooldown(string cooldownID)
	{
		cooldowns[cooldownID].Reset();
	}
	public void ResetCooldowns()
	{
		foreach (Cooldown value in cooldowns.Values)
		{
			value.Reset();
		}
	}
	public bool CheckCooldown(string cooldownID)
	{
		return cooldowns.ContainsKey(cooldownID) && cooldowns[cooldownID].Ready;
	}
	public float GetRemainingTime(string cooldownID)
	{
		return (!cooldowns.ContainsKey(cooldownID)) ? 0f : cooldowns[cooldownID].RemainingTime;
	}
	public float GetMaxTime(string cooldownID)
	{
		return (!cooldowns.ContainsKey(cooldownID)) ? 0f : cooldowns[cooldownID].MaxTime;
	}
	public float GetTotalChargeSkillProgress(string cooldownID)
	{
		if (!cooldowns.ContainsKey(cooldownID) || !cooldowns[cooldownID].isChargeSkill)
		{
			return 0f;
		}
		return (cooldowns[cooldownID].chargeCount != cooldowns[cooldownID].MaxChargeCount) ? cooldowns[cooldownID].TotalChargePercentage : 1f;
	}
	public IEnumerator StartChargeCounter(string cooldownID)
	{
		if (!cooldowns.ContainsKey(cooldownID))
		{
			yield break;
		}
		Cooldown cd = cooldowns[cooldownID];
		float remainingTime = ((!cd.IsCharging) ? cd.MaxTime : ChaosStopwatch.GetRemainingTime(cd.remainingTimeStopwatchID));
		ChaosStopwatch.Stop(cd.remainingTimeStopwatchID);
		cd.remainingTimeStopwatchID = ChaosStopwatch.Begin(remainingTime, useInterval: true, cd.MaxTime, cd.ChargesMissing);
		cd.IsCharging = true;
		if (cd.chargeCount > 0 && cd.chargeDelayTime > 0f)
		{
			cd.delayStopwatchID = ChaosStopwatch.Begin(cd.chargeDelayTime);
		}
		int currentStopwatchID = cd.remainingTimeStopwatchID;
		while (currentStopwatchID == cd.remainingTimeStopwatchID)
		{
			switch (ChaosStopwatch.CheckInterval(cd.remainingTimeStopwatchID))
			{
			case StopwatchState.Ready:
				if (cd.chargeCount == 0)
				{
					cd.queueCDPing = true;
				}
				cd.chargeCount++;
				break;
			case StopwatchState.Done:
				yield break;
			}
			yield return null;
		}
	}
	public void LowerCoolDownTimer(string cooldownID, float reductionAmount, Player player = null, bool showCooldownPing = true, bool useNormalizedVal = false)
	{
		if (cooldowns.ContainsKey(cooldownID))
		{
			if (useNormalizedVal)
			{
				cooldowns[cooldownID].ReduceRemainingTime(cooldowns[cooldownID].RemainingTime / (1f - cooldowns[cooldownID].TimePercentage) * reductionAmount);
			}
			else
			{
				cooldowns[cooldownID].ReduceRemainingTime(reductionAmount);
			}
			if (player != null && showCooldownPing)
			{
				player.hud.MiscPing("HourglassUI");
			}
		}
	}
	public void LowerAllCoolDownTimers(float reductionAmount, Player player = null, bool showCooldownPing = true, bool useNormalizedVal = false)
	{
		foreach (string key in cooldowns.Keys)
		{
			string cooldownID = key;
			bool useNormalizedVal2 = useNormalizedVal;
			LowerCoolDownTimer(cooldownID, reductionAmount, null, showCooldownPing: true, useNormalizedVal2);
		}
		if (showCooldownPing && player != null)
		{
			player.hud.MiscPing("HourglassUI");
		}
	}
	public void StartCooldownTimer(string cooldownID, float cooldownOverride = -1f)
	{
		if (cooldowns.ContainsKey(cooldownID))
		{
			cooldowns[cooldownID].remainingTimeStopwatchID = ChaosStopwatch.Begin((!(cooldownOverride > 0f)) ? cooldowns[cooldownID].MaxTime : cooldownOverride);
			cooldowns[cooldownID].IsCharging = true;
			AnnounceStartCDEvent(cooldownID);
		}
	}
	public void Add(string givenID, float givenTime, StatData givenData = null, Player.SkillState givenSS = null)
	{
		cooldowns[givenID] = new Cooldown(givenID, givenTime, givenData, givenSS);
	}
	public void AnnounceStartCDEvent(string cooldownID)
	{
		if (startCDEventHandlers != null)
		{
			startCDEventHandlers(cooldownID);
		}
	}
	public void AnnounceStartChargeEvent(string cooldownID)
	{
		if (startChargeEventHandlers != null)
		{
			startChargeEventHandlers(cooldownID);
		}
	}

	{
		List<CooldownEntry> list = new List<CooldownEntry>();
		int num = 0;
		Transform transform = base.transform.Find("Skill" + num);
		while (transform != null)
		{
			CooldownEntry item = new CooldownEntry(transform, this);

		cdPingStopwatchID = ChaosStopwatch.Begin(cooldownPingDelay);
		Player player = this.player;
		player.assignSkillSlotEventHandlers = (Player.AssignSkillSlotEventHandler)Delegate.Combine(player.assignSkillSlotEventHandlers, new Player.AssignSkillSlotEventHandler(AssignSkillEvent));

	{
		if (this.player != null)
		{
			Player player = this.player;
			player.assignSkillSlotEventHandlers = (Player.AssignSkillSlotEventHandler)Delegate.Remove(player.assignSkillSlotEventHandlers, new Player.AssignSkillSlotEventHandler(AssignSkillEvent));
		}
		this.player = null;
		if (gameObject != null)
		{
			gameObject.SetActive(value: false);
		}
	}

	{
		if (!(player == null) && !GameController.hideUI && GameController.activePlayers.Contains(player))
		{

				Color.black

				new Color(1f, 0.5f, 0f)

				Color.white

			{
				ElementType.Earth,
				new Color(0.45f, 0.75f, 0.5f)
			},
			{
				ElementType.Lightning,
				Color.yellow
			},
			{
				ElementType.Water,
				new Color(0.4f, 0.6f, 1f)
			}
		};
		statusModifier = 1f;
		empowerDuration = 2.5f;

	{
		if (followTrans != null)
		{
			transform.position = followTrans.position;
		}

	public void Empower(string skillID, ElementType element, StatData skillData)
	{
		Reset(skillID, skillData);
		if (queuedElement != ElementType.Neutral && queuedElement != ElementType.Air && queuedElement != ElementType.Earth)
		{
			for (int i = 0; i < maxLevel; i++)
			{
				skillData.SetValue(queuedStatusString, statusModifier, i);
			}
		}
		if (element == ElementType.Neutral || element == ElementType.Air || element == ElementType.Earth)
		{
			queuedElement = ElementType.Neutral;
			return;
		}
		queuedElement = element;
		queuedStatusString = elementStatusDictionary[queuedElement];

		if (overrideElement != ElementType.Neutral)
		{
			StartCoroutine(UpdateEffect(elementColorDictionary[overrideElement]));
		}
		else if (queuedElement != ElementType.Neutral)
		{
			StartCoroutine(UpdateEffect(elementColorDictionary[queuedElement]));
		}
	}
	private IEnumerator UpdateEffect(Color color)
	{

		particleSys.Emit(40);

			while (ChaosStopwatch.Check(empowerStopwatchID) && queuedElement != ElementType.Neutral)
			{
				yield return null;
			}
			queuedElement = ElementType.Neutral;
			isEmpowered = false;
			particleSys.Stop();
		}
	}

			skillData.SetValue("freezeChance", StatManager.GetSkillData("Player", skillID).GetValue<float>("freezeChance", i), i);
			skillData.SetValue("shockChance", StatManager.GetSkillData("Player", skillID).GetValue<float>("shockChance", i), i);
		}
	}
