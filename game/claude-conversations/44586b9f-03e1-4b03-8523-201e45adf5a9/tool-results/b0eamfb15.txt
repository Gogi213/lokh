using UnityEngine;
public class PowerUpEffect : MonoBehaviour
{
	private static GameObject prefab;
	private ParticleSystem partSys;
	public Transform parentTransform;
	public static GameObject Prefab => prefab ?? (prefab = ChaosBundle.Get("Assets/Prefabs/Effects/PowerupEffect.prefab"));
	private void Awake()
	{
		partSys = GetComponent<ParticleSystem>();
		partSys.Stop(withChildren: true);
	}
	private void LateUpdate()
	{
		if (parentTransform != null)
		{
			transform.position = parentTransform.position;
		}
		else
		{
			Deactivate();
		}
	}
	public void Activate(Transform parentTrans, Color color)
	{
		parentTransform = parentTrans;
		partSys.startColor = color;
		gameObject.SetActive(value: true);
		partSys.Play(withChildren: true);
	}
	public void Deactivate()
	{
		parentTransform = null;
		partSys.Stop(withChildren: true);
		gameObject.SetActive(value: false);
	}
}

=== SectionedTrailEmitter
using System.Collections;
using Chaos.ParticleSystemExtensions;
using UnityEngine;
public class SectionedTrailEmitter : ParticleEffect
{
	private ParticleSystem.EmitParams emitParams;
	public string endEmitterName = string.Empty;
	public float unitLength = 1f;
	private float halfUnitLength;
	private int maxEmits = 50;
	public float rotationOffset;
	private Vector2 emitOffset;
	private Vector2 nextEmitPos;
	private Vector2 trailVector;
	private float trailDist;
	private float halfTrailDist;
	private float fullTrailDist;
	private int emitCount;
	private float curRotation;
	private bool snapped;
	private void Start()
	{
		emitParams = default;
		halfUnitLength = unitLength * 0.5f;
	}
	public IEnumerator EmitDelayed(Vector2 startPos, Vector2 endPos, int particleCountOverride = -1, bool emitEveryOther = false, float roundUpValue = -1f, bool snapToFit = true, float startLifetime = -1f, float delayTime =
	{
		delayRunning = true;
		yield return new WaitForSeconds(delayTime);
		Emit(startPos, endPos, particleCountOverride, emitEveryOther, roundUpValue, snapToFit, startLifetime, 0f, startColor, useEndEmitter, boxThickness, sortingLayer);
		delayRunning = false;
	}
	public void Emit(Vector2 startPos, Vector2 endPos, int particleCountOverride = -1, bool emitEveryOther = false, float roundUpValue = -1f, bool snapToFit = true, float startLifetime = -1f, float delayTime = 0f, Color32? 
	{
		gameObject.SetActive(value: true);
		if (delayTime > 0f)
		{
			StartCoroutine(EmitDelayed(startPos, endPos, particleCountOverride, emitEveryOther, roundUpValue, snapToFit, startLifetime, delayTime, startColor, useEndEmitter, boxThickness, sortingLayer));
			return;
		}
		emitOffset = ((effectEmitter.simulationSpace != ParticleSystemSimulationSpace.Local) ? Vector3.zero : transform.position);
		fullTrailDist = Vector2.Distance(startPos, endPos);
		halfTrailDist = trailDist * 0.5f;
		trailVector = (endPos - startPos).normalized;
		curRotation = 0f - Globals.GetRotationVector((startPos - endPos).normalized).z + rotationOffset;
		snapped = false;
		for (emitCount = 0; emitCount < maxEmits; emitCount++)
		{
			if (!emitEveryOther || emitCount % 2 == 0)
			{
				nextEmitPos = endPos - trailVector * halfUnitLength * (emitCount + 1);
				trailDist = Vector2.Distance(nextEmitPos, startPos);
				if (Vector2.Distance(endPos, nextEmitPos) > fullTrailDist || (roundUpValue > 0f && trailDist < roundUpValue))
				{
					break;
				}
				if (trailDist < halfUnitLength)
				{
					if (!snapToFit)
					{
						break;
					}
					snapped = true;
					nextEmitPos = startPos + trailVector * halfUnitLength;
				}
				emitParams.position = nextEmitPos - emitOffset;
				emitParams.rotation = curRotation;
				if (startColor.HasValue)
				{
					emitParams.startColor = startColor.Value;
				}
				else
				{
					emitParams.ResetStartColor();
				}
				if (startLifetime >= 0f)
				{
					emitParams.startLifetime = startLifetime;
				}
				else
				{
					emitParams.ResetStartLifetime();
				}
				if (boxThickness.HasValue)
				{
					shapeMod.box = boxThickness.Value;
				}
				else
				{
					shapeMod.box = initialSettings.boxThickness;
				}
				if (sortingLayer != null)
				{
					ParticleSystemExtension.SetSortingLayer(effectEmitter, sortingLayer);
				}
				else
				{
					ParticleSystemExtension.SetSortingLayer(effectEmitter, initialSettings.sortingLayer);
				}
				if (emitCount == 0 && useEndEmitter && endEmitterName != null && endEmitterName != string.Empty)
				{
					PoolManager.GetPoolItem<ParticleEffect>(endEmitterName).Emit(1, nextEmitPos, localEulerAngles: new Vector3(0f, 0f, 0f - (curRotation + rotationOffset)), overrides: (startColor.HasValue || !(startLifetime < 0f)) ? ne
					{
						startColor = startColor,
						startLifetime = startLifetime
					} : null);
				}
				else
				{

	{
		for (int i = 0; i < iListCount; i++)
		{
			psRenderer = (Renderer)iList[i];
			if (psRenderer != null)
			{
				psRenderer.sortingOrder = GetSortingOrder(psRenderer.transform.position.y);
			}
		}

	public override void Awake()
	{
		if (sortingLayer == string.Empty)
		{
			sortingLayer = "Default";
			Debug.Log("Sorting Layer not set for: " + gameObject.name + ". Setting to Default.");
		}
		SetType();
		CreateList();
		baseTransform = transform;
		ParticleSystem[] componentsInChildren = GetComponentsInChildren<ParticleSystem>();
		List<Renderer> list = new List<Renderer>();
		ParticleSystem[] array = componentsInChildren;
		foreach (ParticleSystem particleSystem in array)
		{
			Renderer component = particleSystem.GetComponent<Renderer>();
			if (component.sortingLayerName == sortingLayer)
			{
				list.Add(component);
			}
		}
		iList = list;
		childSorterList = GetComponentsInChildren<SortGenericLayer>().ToList();
		childSorterList.Remove(this);
	}
	public override void Start()
	{
		base.Start();
		for (int num = iList.Count - 1; num >= 0; num--)
		{
			psRenderer = (Renderer)iList[num];
			if (psRenderer.sortingLayerName != sortingLayer)
			{
				iList.RemoveAt(num);
			}
		}
		iListCount = iList.Count;
	}

using System.Collections.Generic;
using UnityEngine;
[AddComponentMenu("Chaos/Components/SpriteTrail")]
public class SpriteTrail : MonoBehaviour
{
	private static GameObject prefab;
	private FixedSizeQueue<SpriteTrailSnapshot> trailLocations;
	private SpriteTrailSnapshot[] trailLocArray;
	private SpriteTrailSnapshot tempSTS;
	public SpriteRenderer spriteRenderer;
	private SpriteRenderer[] trailRenderers;
	private bool initialized;
	public new bool enabled;
	public int trailCount = 5;
	public float trailInterval = 0.02f;
	private int trailIntervalStopwatchID = -1;
	public bool useParentRenderer;
	public Color trailColor;
	private SpriteRenderer tempRenderer;
	public static GameObject Prefab
	{
		get
		{
			if (prefab == null)
			{
				prefab = ChaosBundle.Get("Assets/Prefabs/Effects/SpriteTrail.prefab");
			}
			return prefab;
		}
	}
	public void InitRenderers(SpriteRenderer parentRenderer)
	{
		this.spriteRenderer = parentRenderer;
		SpriteRenderer[] array = trailRenderers;
		foreach (SpriteRenderer spriteRenderer in array)
		{
			spriteRenderer.material = this.spriteRenderer.material;
			spriteRenderer.sortingLayerName = "Trail";
			spriteRenderer.sortingOrder = this.spriteRenderer.sortingOrder;
			spriteRenderer.color = trailColor;
		}
		initialized = true;
	}
	public void Awake()
	{
		enabled = false;
		initialized = false;
		List<SpriteRenderer> list = new List<SpriteRenderer>();
		for (int i = 0; i < trailCount; i++)
		{
			GameObject gameObject = new GameObject();
			gameObject.name = "SpriteTrail" + i;
			gameObject.transform.SetParent(transform);
			SpriteRenderer item = gameObject.AddComponent<SpriteRenderer>();
			list.Add(item);
		}
		trailRenderers = list.ToArray();
		trailLocations = new FixedSizeQueue<SpriteTrailSnapshot>(trailRenderers.Length);
		trailLocArray = trailLocations.GetArray();
		if (useParentRenderer)
		{
			SpriteRenderer component = GetComponent<SpriteRenderer>();
			if (component != null)
			{
				InitRenderers(component);
			}
			else
			{
				InitRenderers(GetComponentInChildren<SpriteRenderer>());
			}
		}
	}
	public void Update()
	{
		if (!initialized || !enabled)
		{
			return;
		}
		if (ChaosStopwatch.Check(trailIntervalStopwatchID))
		{
			trailIntervalStopwatchID = ChaosStopwatch.Begin(trailInterval);
			trailLocations.Enqueue(new SpriteTrailSnapshot(spriteRenderer.transform.position, spriteRenderer.transform.lossyScale, spriteRenderer.sprite, spriteRenderer.sortingOrder));
		}
		for (int i = 0; i < trailLocArray.Length; i++)
		{
			tempSTS = trailLocArray[i];
			if (tempSTS != null)
			{
				tempRenderer = trailRenderers[i];
				tempRenderer.sprite = tempSTS.sprite;
				tempRenderer.transform.position = tempSTS.position;
				tempRenderer.transform.localScale = tempSTS.scale;
				tempRenderer.sortingOrder = tempSTS.sortingOrder;
				tempRenderer.enabled = tempRenderer.transform.position != spriteRenderer.transform.position;
			}
		}
	}
	public void Toggle(bool enableTrails = true)
	{
		if (!enableTrails)
		{
			enabled = false;
			SpriteRenderer[] array = trailRenderers;
			foreach (SpriteRenderer spriteRenderer in array)
			{
				spriteRenderer.enabled = false;
			}
			trailLocations.Clear();
		}
		else
		{
			enabled = true;
			SpriteRenderer[] array2 = trailRenderers;
			foreach (SpriteRenderer spriteRenderer2 in array2)
			{
				spriteRenderer2.transform.position = this.spriteRenderer.transform.position;
			}
		}
	}
}
