public class XTexture
{
	public XSprite[] cell;

	public Texture2D texture;

	public string name = "";

	public const int TYPE_NORMAL = 0;

	public const int TYPE_CLOTHES = 1;

	public const int TYPE_MAP = 2;

	public const int TYPE_CHARACTER = 3;

	public const int TYPE_CHAR = 4;

	public const int TYPE_CHAR_VAR = 5;

	public const int MAX_CELLS = 128;

	public const string VERSION_TREE = "Tree 1.0";

	public int ugcPackageIdx = -1;

	public int ugcSheetIdx = -1;

	private bool isLoading;

	private bool isLoaded;

	public bool needsUnload;

	public int type;

	public XTexture()
	{
		cell = new XSprite[128];
		type = 0;
	}

	public XTexture(Texture2D t, string[] chars)
	{
		texture = t;
		cell = new XSprite[chars.Length];
		type = 3;
		for (int i = 0; i < chars.Length; i++)
		{
			cell[i] = new XSprite();
			XSprite xSprite = cell[i];
			xSprite.name = chars[i];
			xSprite.srcRect = new Rectangle(0, 0, 200, 400);
			xSprite.origin = new Vector2((float)xSprite.srcRect.Width / 2f, (float)xSprite.srcRect.Height - 20f);
		}
	}

	public XTexture(ContentManager Content, string path)
	{
		texture = Content.Load<Texture2D>(path);
		Read(path + ".zsx");
	}

	public XTexture(ContentManager Content, string filePath, string texPath)
	{
		texture = Content.Load<Texture2D>(texPath);
		Read(filePath);
	}

	public XTexture(ContentManager Content, string filePath, Texture2D texture)
	{
		this.texture = texture;
		Read(filePath);
	}

	public void Read(string path)
	{
		FileMgr.Open(path);
		ReadData(FileMgr.reader);
		FileMgr.Close();
	}

	public void ReadData(BinaryReader reader)
	{
		type = reader.ReadInt32();
		int num = reader.ReadInt32();
		cell = new XSprite[num];
		for (int i = 0; i < num; i++)
		{
			if (reader.ReadBoolean())
			{
				cell[i] = new XSprite();
				XSprite xSprite = cell[i];
				xSprite.name = reader.ReadString();
				xSprite.srcRect = new Rectangle(reader.ReadInt32(), reader.ReadInt32(), reader.ReadInt32(), reader.ReadInt32());
				xSprite.origin = new Vector2(reader.ReadSingle(), reader.ReadSingle());
				switch (type)
				{
				case 1:
					xSprite.char_ref = reader.ReadInt32();
					xSprite.flags = reader.ReadInt32();
					break;
				case 2:
					xSprite.flags = reader.ReadInt32();
					break;
				case 4:
					xSprite.flags = reader.ReadInt32();
					break;
				}
			}
		}
	}

	public Rectangle GetSpriteRect(int idx)
	{
		if (idx < cell.Length && cell[idx] != null)
		{
			if (ugcPackageIdx > -1 && ugcSheetIdx > -1 && idx < UGCManager.packages[ugcPackageIdx].sheets[ugcSheetIdx].cell.Length)
			{
				return UGCManager.packages[ugcPackageIdx].sheets[ugcSheetIdx].cell[idx].srcRect;
			}
			return cell[idx].srcRect;
		}
		return default;
	}

	public string GetSpriteName(int idx)
	{
		if (idx < cell.Length && cell[idx] != null)
		{
			return cell[idx].name;
		}
		return "";
	}

	public Vector2 GetSpriteOrigin(int idx)
	{
		if (idx < cell.Length)
		{
			if (ugcPackageIdx > -1 && ugcSheetIdx > -1 && idx < UGCManager.packages[ugcPackageIdx].sheets[ugcSheetIdx].cell.Length)
			{
				return UGCManager.packages[ugcPackageIdx].sheets[ugcSheetIdx].cell[idx].origin;
			}
			if (cell[idx] != null)
			{
				return cell[idx].origin;
			}
		}
		return default;
	}

	public void SetSpriteRect(int idx, Rectangle rect)
	{
		if (idx < cell.Length)
		{
			if (cell[idx] == null)
			{
				cell[idx] = new XSprite();
			}
			cell[idx].srcRect = rect;
		}
	}

	public void SetSpriteOrigin(int idx, Vector2 origin)
	{
		if (idx < cell.Length)
		{
			if (cell[idx] == null)
			{
				cell[idx] = new XSprite();
			}
			cell[idx].origin = origin;
		}
	}

	public virtual void Draw(Vector2 loc, int idx, Vector2 scale, float rotation, float r, float g, float b, float a)
	{
		if (idx < 0 || idx >= cell.Length || texture == null || texture == null)
		{
			return;
		}
		XSprite xSprite = GetCell(idx);
		if (xSprite != null)
		{
			if (float.IsNaN(rotation))
			{
				rotation = 0f;
			}
			SpriteTools.sprite.Draw(texture, loc, xSprite.srcRect, new Color(r, g, b, a), rotation, xSprite.origin - new Vector2(xSprite.srcRect.X, xSprite.srcRect.Y), scale, SpriteEffects.None, 0f);
		}
	}

	public XSprite GetCell(int idx)
	{
		if (ugcPackageIdx > -1 && ugcSheetIdx > -1 && idx < UGCManager.packages[ugcPackageIdx].sheets[ugcSheetIdx].cell.Length)
		{
			return UGCManager.packages[ugcPackageIdx].sheets[ugcSheetIdx].cell[idx];
		}
		return cell[idx];
	}

	public XSprite GetOriginalCell(int idx)
	{
		return cell[idx];
	}

	public int GetCellCount()
	{
		if (ugcPackageIdx > -1 && ugcSheetIdx > -1)
		{
			return UGCManager.packages[ugcPackageIdx].sheets[ugcSheetIdx].cell.Length;
		}
		return cell.Length;
	}

	public int GetOriginalCellCount()
	{
		return cell.Length;
	}

	public virtual void DrawFlipped(Vector2 loc, int idx, Vector2 scale, float rotation, float r, float g, float b, float a)
	{
		XSprite xSprite = GetCell(idx);
		try
		{
			if (float.IsNaN(rotation))
			{
				rotation = 0f;
			}
			float num = scale.X;
			if (scale.Y > scale.X)
			{
				num = scale.Y;
			}
			Vector2 vector = (xSprite.origin - new Vector2(xSprite.srcRect.Left, xSprite.srcRect.Top)) * num * 1.4f;
			Vector2 vector2 = (new Vector2(xSprite.srcRect.Right, xSprite.srcRect.Bottom) - xSprite.origin) * num * 1.4f;
			if (vector2.X > vector.X)
			{
				vector.X = vector2.X;
			}
			if (vector2.Y > vector.Y)
			{
				vector.Y = vector2.Y;
			}
			if (!(loc.X + vector.X < 0f) && !(loc.Y + vector.Y < 0f) && !(loc.X - vector.X > ScrollManager.totalScreenSize.X) && !(loc.Y - vector.Y > ScrollManager.totalScreenSize.Y))
			{
				Vector2 origin = xSprite.origin - new Vector2(xSprite.srcRect.X, xSprite.srcRect.Y);
				origin.X = (float)xSprite.srcRect.Width - origin.X;
				SpriteTools.sprite.Draw(texture, loc, xSprite.srcRect, new Color(r, g, b, a), rotation, origin, scale, SpriteEffects.FlipHorizontally, 0f);
			}
		}
		catch
		{
		}
	}

	public XTexture(BinaryReader reader, ContentManager Content)
	{
		name = reader.ReadString();
		ReadData(reader);
		switch (type)
		{
		case 1:
		case 4:
			isLoaded = false;
			isLoading = false;
			break;
		case 0:
			switch (name)
			{
			case "sprites":
			case "swooshes":
			case "logosegs":
			case "smashables":
			case "smashbits":
				texture = Content.Load<Texture2D>("gfx/" + name);
				isLoaded = true;
				break;
			default:
				isLoaded = false;
				isLoading = false;
				break;
			}
			break;
		default:
			texture = Content.Load<Texture2D>("gfx/" + name);
			isLoaded = true;
			break;
		}
	}

	public bool IsLoaded()
	{
		if (needsUnload)
		{
			if (isLoading)
			{
				return false;
			}
			needsUnload = false;
			isLoaded = false;
		}
		if (!isLoaded && !isLoading)
		{
			Point point = UGCManager.CheckReplace(name);
			ugcPackageIdx = point.X;
			ugcSheetIdx = point.Y;
		}
		if (isLoaded)
		{
			return true;
		}
		if (isLoading)
		{
			return false;
		}
		if (StorageJobMgr.busy)
		{
			return false;
		}
		if (StorageJobMgr.timeOut > 0f)
		{
			return false;
		}
		isLoading = true;
		isLoaded = false;
		Thread thread = new Thread(ThreadedLoader);
		try
		{
			thread.Start();
		}
		catch
		{
		}
		return false;
	}

	private void ThreadedLoader()
	{
		while (Textures.loading)
		{
			Thread.Sleep(Rand.GetRandomInt(100, 1000));
		}
		while (StorageJobMgr.jobWorking)
		{
			Thread.Sleep(Rand.GetRandomInt(100, 1000));
		}
		if (!Monitor.TryEnter(Textures.textureSemaphore, 1000))
		{
			return;
		}
		try
		{
			Textures.loading = true;
			if (ugcPackageIdx > -1 && ugcSheetIdx > -1)
			{
				FileStream fileStream = new FileStream(Path.Combine(UGCManager.packages[ugcPackageIdx].directory, UGCManager.packages[ugcPackageIdx].sheets[ugcSheetIdx].name + ".PNG"), FileMode.Open);
				texture = Texture2D.FromStream(Game1.graphics.GraphicsDevice, fileStream);
				fileStream.Close();
			}
			else
			{
				texture = Textures.StreamingContent.Load<Texture2D>("gfx/" + name);
			}
			isLoaded = true;
			isLoading = false;
			Textures.loading = false;
		}
		finally
		{
			Monitor.Exit(Textures.textureSemaphore);
		}
	}

	public XTexture(string name, BinaryReader reader, Texture2D tex)
	{
		this.name = name;
		ReadData(reader);
		texture = tex;
	}

	public Texture2D GetTexture()
	{
		return texture;
	}

	{

	}

	public void Init(string directory, string filename, string name, string description, string tags, string meta, string previewPath)
	{
		this.directory = directory;
		fileName = filename;
		this.name = name;
		this.description = description;
		this.tags = tags;
		this.meta = meta;
		this.previewPath = previewPath;
		sheets = new List<XTexture>();
	}

	public string GetFullPath()
	{
		return Path.Combine(directory, fileName + ".sltpkg");
	}

	public void RefreshTextures(GraphicsDevice dev)
	{
		for (int i = 0; i < sheets.Count; i++)
		{
			if (sheets[i].texture == null)
			{
				FileStream fileStream = new FileStream(Path.Combine(directory, sheets[i].name + ".PNG"), FileMode.Open);
				sheets[i].texture = Texture2D.FromStream(dev, fileStream);
				fileStream.Close();
			}
		}
	}

	public void Read(string path)
	{
		directory = Path.GetDirectoryName(path);
		fileName = Path.GetFileNameWithoutExtension(path);
		BinaryReader binaryReader = new BinaryReader(File.Open(GetFullPath(), FileMode.Open, FileAccess.Read));
		userNeedsToAcceptWorkshopLegalAgreement = binaryReader.ReadBoolean();
		publishedFileId = binaryReader.ReadUInt64();
		name = binaryReader.ReadString();
		description = binaryReader.ReadString();
		tags = binaryReader.ReadString();
		meta = binaryReader.ReadString();
		previewPath = binaryReader.ReadString();
		previewPath = previewPath.Substring(previewPath.LastIndexOf('\\') + 1);
		int num = binaryReader.ReadInt32();
		sheets = new List<XTexture>();
		for (int i = 0; i < num; i++)
		{
			XTexture xTexture = new XTexture();
			xTexture.name = binaryReader.ReadString();
			if (File.Exists(Path.Combine(directory, xTexture.name + ".zsx")) && File.Exists(Path.Combine(directory, xTexture.name + ".PNG")))
			{
				xTexture.Read(Path.Combine(directory, xTexture.name + ".zsx"));
				sheets.Add(xTexture);
			}
		}
		binaryReader.Close();
	}

	internal bool IsValid()
	{
		if (directory == null || directory == "" || fileName == null || fileName == "")
		{
			return false;
		}
		if (Directory.Exists(directory) && File.Exists(GetFullPath()))
		{
			return true;
		}
		return false;
	}

	internal void UpdateAffectedTextures()
	{
		for (int i = 0; i < sheets.Count; i++)
		{
			int textureIdx = Textures.GetTextureIdx(sheets[i].name);
			if (textureIdx > -1)
			{
				Textures.tex[textureIdx].needsUnload = true;
				GameConsole.WriteLine("Unloading: " + sheets[i].name);
			}
		}
	}
