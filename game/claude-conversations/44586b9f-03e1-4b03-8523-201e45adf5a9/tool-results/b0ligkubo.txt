public class RelicDrop : MonoBehaviour
{
	private ItemDrop currentItem;

	private ChaosQuickStopwatch spawnStopwatch = new ChaosQuickStopwatch(0.5f);

	private bool spawnCompleted;

	public string itemID = string.Empty;

	public float cursedChance = 0.1f;

	private void Start()
	{
		currentItem = Globals.ChaosInst<ItemDrop>(ItemDrop.Prefab, transform.parent, transform.position);
	}

	private void Update()
	{
		if (!spawnCompleted)
		{
			if (!spawnStopwatch.IsRunning)
			{
				spawnCompleted = true;
				if (currentItem != null)
				{
					currentItem.Reset();
					currentItem.itemID = ((!(itemID != string.Empty)) ? LootManager.GetAvailableItemID(cursedChance) : itemID);
					currentItem.useLifeTime = false;
					currentItem.finishedMoving = true;
					currentItem.Activate(disableActivationSFX: true);
					currentItem.transform.position = transform.position;
				}
			}
		}
		else if (gameObject != null)
		{
			Object.Destroy(gameObject);
		}
	}
}

public class RelicWell : MonoBehaviour
{
	public struct RelicInRange
	{
		public ItemDrop item;

		public Vector2 startPos;
	}

	public delegate void CollectRelicEventHandler(int givenCost);

	private static GameObject prefab;

	private Dictionary<RelicInRange, ChaosQuickStopwatch> relicTimers;

	private Dictionary<RelicInRange, float> relicRangeDuration;

	private List<RelicInRange> itemsInRange;

	private List<RelicInRange> itemsToRemove;

	private const float attractDelay = 1f;

	private const float attractTime = 1f;

	private Vector2 lootDropLocation;

	private float costModifier = 0.4f;

	private int goldValue;

	public CollectRelicEventHandler collectRelicEventHandler;

	public static GameObject Prefab => prefab ?? (prefab = ChaosBundle.Get("Assets/Prefabs/AreaEffects/RelicWell.prefab"));

	private void Start()
	{
		lootDropLocation = transform.position;
		itemsInRange = new List<RelicInRange>(20);
		itemsToRemove = new List<RelicInRange>(20);
		relicTimers = new Dictionary<RelicInRange, ChaosQuickStopwatch>(20);
		relicRangeDuration = new Dictionary<RelicInRange, float>(20);
	}

	public void Update()
	{
		if (itemsInRange.Count == 0)
		{
			return;
		}
		itemsInRange.RemoveNullValues();
		for (int i = 0; i < itemsInRange.Count; i++)
		{
			RelicInRange relicInRange = itemsInRange[i];
			if (relicRangeDuration[relicInRange] < 1f)
			{
				relicRangeDuration[relicInRange] += Time.deltaTime;
				continue;
			}
			if (!relicTimers[relicInRange].IsRunning)
			{
				relicTimers[relicInRange].IsRunning = true;
				continue;
			}
			if (Vector2.Distance(relicInRange.item.transform.position, transform.position) <= 1f)
			{
				CollectRelic(relicInRange.item);
				itemsToRemove.Add(relicInRange);
			}
			else
			{
				itemsInRange[i].item.rigidbody2D.MovePosition(Vector2.Lerp(itemsInRange[i].startPos, transform.position, Globals.GetSmoothLerpValue(relicTimers[relicInRange].TimePercentage)));
			}
			relicRangeDuration[relicInRange] += Time.deltaTime;
		}
		if (itemsToRemove.Count > 0)
		{
			for (int j = 0; j < itemsToRemove.Count; j++)
			{
				itemsInRange.Remove(itemsToRemove[j]);
				relicRangeDuration.Remove(itemsToRemove[j]);
				relicTimers.Remove(itemsToRemove[j]);
			}
			itemsToRemove.Clear();
		}
	}

	public void OnTriggerEnter2D(Collider2D col)
	{
		if (!(col.name != "ItemHitBox") && !(col.transform.parent == null))
		{
			ItemDrop component = col.transform.parent.GetComponent<ItemDrop>();
			if (!(component == null))
			{
				RelicInRange relicInRange = new RelicInRange
				{
					item = component,
					startPos = component.transform.position
				};
				itemsInRange.Add(relicInRange);
				relicRangeDuration[relicInRange] = 0f;
				relicTimers[relicInRange] = new ChaosQuickStopwatch();
			}
		}
	}

	private void CollectRelic(ItemDrop givenID)
	{
		if (!(givenID == null) && !(givenID.itemID == string.Empty))
		{
			goldValue = Mathf.CeilToInt((float)LootManager.GetPrice(givenID.itemID) * costModifier);
			if (givenID.itemID == "TokenCollector")
			{
				goldValue *= 3;
			}
			else if (Inventory.EitherPlayerHasItem("TokenCollector"))
			{
				goldValue *= 2;
			}
			LootManager.DropGold(lootDropLocation, goldValue);
			SoundManager.PlayAudio("MenuBuy");
			givenID.Disable();
			if (collectRelicEventHandler != null)
			{
				collectRelicEventHandler(goldValue);
			}
		}
	}
}

public class SkillDrop : PickupItem
{
	public string skillID;

	public bool startActivated;

	private float spawnEnableTime = 0.25f;

	private int enableStopwatchID;

	private bool noPickUpMode = true;

	private SpriteRenderer skillIcon;

	private CircleCollider2D floorCollider;

	public SpriteRenderer standardSR;

	public SpriteRenderer empoweredSR;

	public SpriteRenderer sigSR;

	public bool isSignature;

	public bool isEmpowered;

	private Light itemLight;

	private float itemLightInitIntensity;

	private float lightFadeoutIncrement;

	private Color spriteColor;

	private float spriteInitAlpha;

	private float spriteAlphaFadeoutIncrement;

	private float fadeoutStartVal = 0.75f;

	private CompassIndicatorTrigger compassTrigger;

	[HideInInspector]
	public bool lockedSkillSelected;

	private GameObject newSkillIndicator;

	public override void Awake()
	{
		base.Awake();
		overheadPromptHeight = 2.5f;
		autoPickupDelay = 0.3f;
		skillIcon = transform.Find("ItemIcon").GetComponent<SpriteRenderer>();
		floorCollider = transform.Find("ItemFloorCollider").GetComponent<CircleCollider2D>();
		floorCollider.enabled = false;
		newSkillIndicator = transform.Find("Canvas/TopText").gameObject;
		fadeoutStartVal *= lifeTime;
		itemLight = transform.Find("ItemLight").GetComponent<Light>();
		itemLightInitIntensity = itemLight.intensity;
		lightFadeoutIncrement = itemLightInitIntensity / fadeoutStartVal * 0.5f;
		spriteColor = skillIcon.color;
		spriteInitAlpha = spriteColor.a;
		spriteAlphaFadeoutIncrement = spriteInitAlpha / fadeoutStartVal;
		if (startActivated && skillID != string.Empty)
		{
			Activate();
		}
		else
		{
			Reset();
		}
	}

	public override void Activate(bool disableActivationSFX = false)
	{
		base.Activate(disableActivationSFX);
		skillIcon.sprite = IconManager.GetSkillIcon(skillID);
		enableStopwatchID = ChaosStopwatch.Begin(spawnEnableTime);
		pickupCollider.enabled = false;
		noPickUpMode = true;
		itemLight.intensity = itemLightInitIntensity * ((!isEmpowered) ? 1f : 2f);
		spriteColor.a = spriteInitAlpha;
		skillIcon.color = spriteColor;
		newSkillIndicator.SetActive(lockedSkillSelected);
		sigSR.enabled = false;
		empoweredSR.enabled = false;
		standardSR.enabled = false;
		if (isSignature)
		{
			sigSR.enabled = true;
		}
		else if (isEmpowered)
		{
			empoweredSR.enabled = true;
		}
		else
		{
			standardSR.enabled = true;
		}
		if (compassTrigger == null)
		{
			compassTrigger = CompassIndicatorTrigger.Spawn(Compass.CompassIndicatorType.Arcana, transform.position, transform, 8f);
		}
		compassTrigger.iconOverride = skillIcon.sprite;
		compassTrigger.Activate();
	}

	public override void Disable()
	{
		lockedSkillSelected = false;
		base.Disable();
	}

	public override void Update()
	{
		base.Update();
		if (!pickupCollider.enabled && ChaosStopwatch.Check(enableStopwatchID))
		{
			pickupCollider.enabled = true;
			noPickUpMode = false;
		}
		if (useLifeTime && LifeTimeRemaining < fadeoutStartVal)
		{
			itemLight.intensity -= lightFadeoutIncrement * Time.deltaTime;
			spriteColor.a -= spriteAlphaFadeoutIncrement * Time.deltaTime;
			spriteColor.a = Mathf.Max(spriteColor.a, 0.2f);
			skillIcon.color = spriteColor;
		}
		if (finishedMoving && !floorCollider.enabled)
		{
			floorCollider.enabled = true;
		}
	}

	private void OnTriggerStay2D(Collider2D col)
	{
		if (!(col.gameObject.tag == "Player") || noPickUpMode)
		{
			return;
		}
		Player component = col.transform.parent.GetComponent<Player>();
		if (component == null)
		{
			return;
		}
		if (requireInteract)
		{
			CurrentOverheadPrompt.Initialize(transform, component.inputDevice.inputScheme, overheadPromptHeight);
			CurrentOverheadPrompt.DisplayPrompt("Interact");
			if (!players.Contains(component))
			{
				players.Add(component);
			}
		}
		else
		{
			OnInteract(component);
		}
	}

	private void OnTriggerExit2D(Collider2D col)
	{
		BaseOnTriggerExit2D(col);
	}

	public override void OnInteract(Player player)
	{
		if (!(player == null))
		{
			player.PickUpSkill(skillID, isSignature, isEmpowered);
			Disable();
		}
	}

	public override void Reset()
	{
		base.Reset();
		skillID = string.Empty;
		isSignature = false;
		isEmpowered = false;
		lockedSkillSelected = false;
	}
}

[Serializable]
public class Wallet
{
	public delegate void WithdrawEventHandler(int amount);

	public delegate void DepositEventHandler(int amount);

	public delegate void PreDepositEventHandler(ref int amount);

	public int balance;

	public int maxBalance;

	public WithdrawEventHandler withdrawEventHandlers;

	public DepositEventHandler depositEventHandlers;

	public PreDepositEventHandler preDepositEventHandlers;

	public Wallet(int newBalance, int newMaxBalance)
	{
		balance = newBalance;
		maxBalance = newMaxBalance;
	}

	public virtual bool Withdraw(int withdrawAmount)
	{
		if (balance - withdrawAmount < 0)
		{
			return false;
		}
		balance -= withdrawAmount;
		AnnounceWithdrawEvent(withdrawAmount);
		return true;
	}

	public virtual bool WithdrawIfAny(int withdrawAmount)
	{
		if (balance > 0)
		{
			withdrawAmount = Mathf.Min(balance, withdrawAmount);
			balance -= withdrawAmount;
			AnnounceWithdrawEvent(withdrawAmount);
			return true;
		}
		return false;
	}

	public virtual bool Deposit(int depositAmount)
	{
		AnnouncePreDepositEvent(ref depositAmount);
		if (balance == maxBalance)
		{
			return false;
		}
		if (balance + depositAmount > maxBalance)
		{
			depositAmount -= balance + depositAmount - maxBalance;
		}
		balance += depositAmount;
		AnnounceDepositEvent(depositAmount);
		return true;
	}

	public void AnnounceWithdrawEvent(int amount)
	{
		if (withdrawEventHandlers != null)
		{
			withdrawEventHandlers(amount);
		}
	}

	public void AnnounceDepositEvent(int amount)
	{
		if (depositEventHandlers != null)
		{
			depositEventHandlers(amount);
		}
	}

	public void AnnouncePreDepositEvent(ref int amount)
	{
		if (preDepositEventHandlers != null)
		{
			preDepositEventHandlers(ref amount);
		}
	}
}

# 09 — Лут, валюты, магазины

## Цель

Убийства и сундуки дают золото, кристаллы, сердца, реликвии и арканы по таблице данных; награда незаметно подстраивается под ход забега; магазины показывают честный выбор без повторов. Игрок не видит переключателей, но чувствует «справедливость»: потратил всё — дальше щедрее, лежит на золоте — скупее, на грани смерти — сердца падают чаще.

Отличия от референса (по одной строке):
- Референс бросает глобальный рандом — у нас каждый дроп считается из потока `hash(floor_seed, source_id, drop_index)`: тот же сид + те же убийства = тот же лут, лишний бросок где-то ещё его не сдвигает.
- Пул реликвий в референсе при пустом списке молча ничего не даёт (и 10% выпавшего «проклятого» броска пропадает) — у нас запасной пул.
- Поле «шанс заблокированной арканы» в референсе перевёрнуто (1 = не искать новую) — у нас поле `prefer_new_spell` читается прямо.
- Цены, тиры, группы исключений и константы лежат в данных (`data/loot.json`, `data/prices.json`, `data/relics.json`), не в коде.

## Поведение

### 1. Валюты

| Валюта | Живёт | Назначение | Предел |
|---|---|---|---|
| Золото | один забег (счётчик «собрано на этаже» сбрасывается каждый этаж) | магазины забега | 9999 (лишнее теряется) |
| Кристаллы | между забегами | разблокировка реликвий и арканов в хабе; проклятая сделка | 999 |
| Сердца | подбираемое | лечение: один подобранный предмет лечит 25 (Источник: HealthPowerup.cs) | — |
| Души | только кооп, пока партнёр мёртв | оживление партнёра | — |

Подбираемое золото сначала гасит долг (отдельный кошелёк долга, 18 §7): остаток после погашения идёт в кошелёк, и только эта чистая сумма попадает в счётчик «золото этажа» (§4). Меняла и банкир зачисляют прямо в кошелёк (до предела 9999) и «золото этажа» не увеличивают.
Источник: Wallet.cs:Deposit; GoldWallet.cs; Gold.cs:OnTriggerStay2D; JewelerNpc.cs.

Золото падает монетами 1/5/10: пока остаток >0, монета выбирается равновероятно из {1}, {1 или 5, если остаток ≥5}, {1, 5 или 10 по остатку}.

### 2. Схема строки таблицы дропа

Одна строка на источник (класс врага, тип сундука, разрушаемый предмет). Все поля, кроме имени, необязательны.

| Поле | По умолчанию | Смысл |
|---|---|---|
| `gold_rate` | 1 | шанс, что золото вообще выпадет |
| `gold_min`, `gold_max` | 0 | равномерное целое включительно |
| `gold_compensate` | true | применять подстройку (§4) |
| `crystal_rate`, `crystal_min`, `crystal_max` | 0 | см. §3 |
| `heart_rate`, `heart_min`, `heart_max` | 0 | число бросков §3 |
| `relic_rate`, `relic_min`, `relic_max` | 0 | число бросков |
| `spell_rate`, `spell_min`, `spell_max` | 0 | число бросков |
| `prefer_new_spell` | 1 | шанс сначала искать ещё не открытую аркану |
| `signature_rate` | 0 | шанс дать усиленную версию арканы вместо новой |
| `soul_min`, `soul_max` | 0 | души (кооп) |
| `spell_list`, `relic_list` | пусто | сценарный дроп: выбор из списка |
| `spell_element_list` | пусто | ограничение арканы по стихиям: при заданном списке выбирается случайная из перечисленных стихий |

Золото округляется вверх после всех множителей; для кристаллов при базе ровно 1 — плоский бонус + 1 с шансом M (§3).
Источник: DropTableEntry.cs.

Нормализация при загрузке: `max`=0 при `min`>0 → `max`=`min`; для кристаллов, сердец, реликвий, арканов при `rate`>0 и `min`=0 → `max`=1.

### 3. Расчёт количества

- **Золото.** С шансом `gold_rate`: целое из [min, max] × глобальный множитель (кооп +10%) × подстройка (если включена) → округление вверх. Золото без подстройки добавляется в счётчик `uncompensated`.
- **Кристаллы.** База = `crystal_min` (гарантия). С шансом `crystal_rate` база заменяется равномерным целым из [max(min,1), max]. Затем множитель сложности M = min(0.25 × D, 1), где D = петля×3 + этап (0–2) + стартовая прибавка (11 §7). Если количество >1 → floor(n × (1+M)); если =1 → 1 + плоский бонус + (1 с шансом M).
- **Сердца.** `heart_min` гарантированных + по одному броску на каждую единицу от min до max с шансом `heart_rate` × подстройка здоровья (§4).
- **Реликвии и арканы.** `min` гарантированных + по броску на каждую единицу до max с шансом `rate`. Число реликвий умножается на глобальный модификатор (1.0) с отбрасыванием дроби.

### 4. Подстройка наград

**Золото.** Работает, только когда побеждено ≥50% врагов этажа, иначе множитель 1. Пусть d = 1 − (золото_этажа − uncompensated) × 0.005. Если d ≥ 0: множитель = 1 + 0.5·d; иначе = 1 − 0.5·min(1, −d). Примеры: собрано 0 → ×1.5; 100 → ×1.25; 200 → ×1.0; 300 → ×0.75; ≥400 → ×0.5.

**Здоровье.** Берётся наименьшая доля здоровья среди живых игроков h; множитель шанса сердца = линейно от 1.75 (h=0) до 0.5 (h=1): 1.75 − 1.25·h. Гарантированные сердца множитель не трогает.

### 6. Реликвии: пул без возврата

Пул строится при каждой загрузке этажа заново (не раз на забег): из полного списка убираются стартовый, «победный», сборные (только по рецепту), нерозничные, уже имеющиеся у любого игрока, их исключения (§7) и компоненты уже собранных групповых реликвий.

Разблокировка: в пул входят открытые реликвии плюс до `locked_cap` (24) ещё не открытых, набранных по тирам от нижнего вверх (внутри тира случайно) — игрок видит новое, но не всё сразу. Тир 0–4 влияет на цену разблокировки, не на шанс выпадения; шанс равномерный.

Предел `locked_cap` проверяется ПОСЛЕ обработки целого тира: в списках остаются все закрытые записи уже обработанных тиров (их может быть больше 24, например 23 + тир с 30 закрытыми → 53). Внутри тира порядок перемешивается на месте. Для арканы пул — все не надетые аркана обработанных тиров (в том числе открытые); аркана более высоких тиров в пул не попадают; исключаются только 6 назначенных слотов, а запасной (седьмой) слот не исключается, поэтому аркана из запаса может снова выпасть и появиться в магазине. Для реликвий открытые входят из всех тиров, ограничение касается только закрытых.
Источник: LootManager.cs:ResetAvailableItems, ResetAvailableSkills.

Выбор: бросок против `curse_chance` (0.1 по умолчанию, одинаково для дропа и сундуков) → проклятый пул, иначе обычный; элемент берётся равновероятно и вычёркивается. Предел инвентаря — 12 реликвий, 12 проклятых.

Проклятый пул отдельно: проклятые реликвии минус нерозничные и только-хабовые. Особый «хаб-пул» проклятых служит сделке в хабе.

Отличие: референс при пустом выбранном пуле ничего не выдаёт; у нас пустой пул → запасной (другой пул), и только если оба пусты — нет дропа.

### 7. Группы исключений

- **Вытеснение:** владение реликвией A убирает из пула список B(A) (составная реликвия вытесняет свои части; сет вытесняет компоненты; комбинированная стихийная — обе родительские).
- **Взаимоисключение:** в каждом множестве можно держать только одну; поднятие второй сбрасывает остальные; если среди имеющихся есть проклятая — подъём отклоняется.
- **Группа:** в инвентаре компоненты группы считаются одним предметом; их по отдельности в пул не возвращают.
- **Сборная:** выдаётся только как недостающая часть рецепта.
